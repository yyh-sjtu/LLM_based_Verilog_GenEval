module top_module (
	input [7:0] a,
	input [7:0] b,
	output [7:0] s,
	output overflow
);

	reg [7:0] sum;
	reg c;

	// Add the two numbers
	always @* begin
		{c, sum} = a + b;
	end

	// Assign the sum to s
	assign s = sum;

	// Detect overflow
	always @* begin
		if ((a[7] == b[7]) && (a[7] != sum[7])) begin
			overflow = 1;
		end else begin
			overflow = 0;
		end
	end

endmodule
