module and_gate_output(
	input a,
	input b,
	output out
);

	assign out = a & b;

endmodule

module and_gate_alwaysblock(
	input a,
	input b,
	output reg out
);

	always @(*) begin
		out = a & b;
	end

endmodule

module top_module(
	input a,
	input b,
	output out_assign,
	output reg out_alwaysblock
);

	and_gate_output and_gate1(
		.a(a),
		.b(b),
		.out(out_assign)
	);

	and_gate_alwaysblock and_gate2(
		.a(a),
		.b(b),
		.out(out_alwaysblock)
	);

endmodule
