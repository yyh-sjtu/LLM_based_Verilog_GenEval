module d_ff_posedge_ar (
	input clk,
	input d,
	input ar,
	output logic q
);
	
	always_ff @(posedge clk or posedge ar) begin
		if (ar)
			q <= 0;
		else
			q <= d;
	end
	
endmodule

module top_module (
	input clk,
	input d,
	input ar,
	output logic q
);
	
	d_ff_posedge_ar d_ff_inst (
		.clk(clk),
		.d(d),
		.ar(ar),
		.q(q)
	);
	
endmodule
