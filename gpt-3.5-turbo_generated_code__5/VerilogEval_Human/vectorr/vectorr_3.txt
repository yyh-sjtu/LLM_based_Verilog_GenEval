module top_module (
	input [7:0] in,
	output [7:0] out
);

	// Create a temporary variable to store the reversed bits
	reg [7:0] reversed_bits;

	// Reverse the bit ordering
	assign reversed_bits = {in[0], in[1], in[2], in[3], in[4], in[5], in[6], in[7]};

	// Assign the reversed bits to the output
	assign out = reversed_bits;

endmodule
