module top_module(
	input [2:0] a, 
	input [2:0] b, 
	output [2:0] out_or_bitwise,
	output out_or_logical,
	output [5:0] out_not
);

	// Declare intermediate signals
	wire [2:0] inverse_a;
	wire [2:0] inverse_b;

	// Compute inverse of a and b
	assign inverse_a = ~a;
	assign inverse_b = ~b;

	// Compute bitwise-OR of a and b
	assign out_or_bitwise = a | b;

	// Compute logical-OR of a and b
	assign out_or_logical = a || b;

	// Place inverse of b in upper half of out_not and inverse of a in lower half
	assign out_not = {inverse_b, inverse_a};

endmodule
