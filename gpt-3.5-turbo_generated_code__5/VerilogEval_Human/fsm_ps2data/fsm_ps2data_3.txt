```verilog
module top_module (
	input clk,
	input [7:0] in,
	input reset,
	output [23:0] out_bytes,
	output done
);

reg [1:0] state;
reg [7:0] byte1, byte2, byte3;
reg [23:0] out_bytes_reg;
reg done_reg;

always @(posedge clk) begin
	if (reset) begin
		state <= 2'b00;
		byte1 <= 8'b0;
		byte2 <= 8'b0;
		byte3 <= 8'b0;
		out_bytes_reg <= 24'b0;
		done_reg <= 1'b0;
	end else begin
		case(state)
			2'b00: begin // State 0 - Searching for start of message
				if (in[3]) begin
					state <= 2'b01;
					byte1 <= in;
				end
			end
			2'b01: begin // State 1 - Receiving second byte
				if (in[3]) begin
					state <= 2'b10;
					byte2 <= in;
				end
			end
			2'b10: begin // State 2 - Receiving third byte
				state <= 2'b00;
				byte3 <= in;
				out_bytes_reg <= {byte1, byte2, byte3};
				done_reg <= 1'b1;
			end
		endcase
	end
end

assign out_bytes = done_reg ? out_bytes_reg : 24'b0;
assign done = done_reg;

endmodule
```
