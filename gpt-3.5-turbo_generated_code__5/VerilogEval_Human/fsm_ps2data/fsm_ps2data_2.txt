module top_module (
    input clk,
    input [7:0] in,
    input reset,
    output [23:0] out_bytes,
    output done
);

reg [2:0] state;
reg [23:0] temp_bytes;
assign out_bytes = done ? temp_bytes : 24'b0;

always @(posedge clk) begin
    if (reset) begin
        state <= 3'b0;
        temp_bytes <= 24'b0;
    end else begin
        case (state)
            3'b000: begin
                if (in[3]) begin
                    state <= 3'b001;
                    temp_bytes[23:16] <= in;
                end
            end
            3'b001: begin
                state <= 3'b010;
                temp_bytes[15:8] <= in;
            end
            3'b010: begin
                state <= 3'b011;
                temp_bytes[7:0] <= in;
            end
            3'b011: begin
                state <= 3'b100;
            end
            3'b100: begin
                if (in[3]) begin
                    state <= 3'b001;
                    temp_bytes[23:16] <= in;
                    done <= 1'b1;
                end else begin
                    state <= 3'b011;
                end
            end
        endcase
    end
end

endmodule
