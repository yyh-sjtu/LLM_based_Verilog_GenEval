module top_module (
	input clk,
	input [7:0] in,
	input reset,
	output [23:0] out_bytes,
	output done
);

reg [7:0] byte1;
reg [7:0] byte2;
reg [7:0] byte3;
reg [7:0] temp;

reg start_flag;
reg [1:0] state;

always @(posedge clk or posedge reset) begin
	if (reset) begin
		state <= 0;
		start_flag <= 0;
		byte1 <= 0;
		byte2 <= 0;
		byte3 <= 0;
		out_bytes <= 0;
		done <= 0;
	else begin
		case(state)
			0: begin // Discarding bytes until in[3] = 1
				if (in[3] == 1) begin
					start_flag <= 1;
				end 
				else begin
					start_flag <= 0;
				end
				
				if (start_flag == 1) begin
					state <= 1;
				end
			end
			
			1: begin // Receiving byte 1 of the message
				byte1 <= in;
				state <= 2;
			end
			
			2: begin // Receiving byte 2 of the message
				byte2 <= in;
				state <= 3;
			end
			
			3: begin // Receiving byte 3 of the message
				byte3 <= in;
				state <= 4;
			end
			
			4: begin // Signaling the receipt of a message
				temp[23:16] <= byte1;
				temp[15:8] <= byte2;
				temp[7:0] <= byte3;
				out_bytes <= temp;
				done <= 1;
				state <= 0;
			end
		endcase
	end
end

endmodule
