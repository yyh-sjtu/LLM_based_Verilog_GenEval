module top_module (
	input clk,
	input [7:0] in,
	input reset,
	output [23:0] out_bytes,
	output done
);
    reg [2:0] state;
    reg [7:0] byte1;
    reg [7:0] byte2;
    reg [7:0] byte3;
    wire [7:0] next_byte;
    
    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= 0;
            byte1 <= 0;
            byte2 <= 0;
            byte3 <= 0;
            out_bytes <= 0;
            done <= 0;
        end else begin
            case (state)
                0: begin // Idle state, waiting for byte 1
                    if (in[3] === 1) begin
                        byte1 <= in;
                        state <= 1;
                    end
                end
                1: begin // Waiting for byte 2
                    if (in[3] === 0) begin
                        byte2 <= in;
                        state <= 2;
                    end
                end
                2: begin // Waiting for byte 3
                    if (in[3] === 0) begin
                        byte3 <= in;
                        state <= 3;
                    end
                end
                3: begin // Message received, output bytes
                    out_bytes <= {byte1, byte2, byte3};
                    done <= 1;
                    state <= 0;
                end
                default: state <= 0;
            endcase
        end
    end
    
    assign next_byte = (state == 0) ? in : 0;
endmodule
