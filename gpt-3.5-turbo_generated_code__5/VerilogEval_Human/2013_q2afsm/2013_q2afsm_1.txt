```verilog
module fsm_module (
    input clk,
    input resetn,
    input [3:1] r,
    output reg [3:1] g
);

// State parameters
parameter A = 2'b00;
parameter B = 2'b01;
parameter C = 2'b10;
parameter D = 2'b11;

// State flip-flops
reg [1:0] state, next_state;

always @(posedge clk or negedge resetn) begin
    if (~resetn) begin
        state <= A;
    end
    else begin
        state <= next_state;
    end
end

// State table
always @(*) begin
    case (state)
        A: begin
            if (r[1] == 1)
                next_state = B;
            else if (r[2] == 1)
                next_state = C;
            else if (r[3] == 1)
                next_state = D;
            else
                next_state = A;
        end
        B: begin
            if (r[1] == 1)
                next_state = B;
            else
                next_state = A;
        end
        C: begin
            if (r[2] == 1)
                next_state = C;
            else
                next_state = A;
        end
        D: begin
            if (r[1] == 1)
                next_state = B;
            else if (r[2] == 1)
                next_state = C;
            else if (r[3] == 1)
                next_state = D;
            else
                next_state = A;
        end
    endcase
end

// FSM outputs
always @(*) begin
    case (state)
        A: g = 3'b000;
        B: g = 3'b100;
        C: g = 3'b010;
        D: g = 3'b001;
    endcase
end

endmodule
