module top_module (
    input in,
    input [3:0] state,
    output reg [3:0] next_state,
    output out
);

    always @(*) begin
        case(state)
            4'b0001: // State A
                begin
                    if(in == 1'b0) begin
                        next_state = 4'b0001; // Next state in=0 -> A
                        out = 1'b0; // Output is 0
                    end
                    else begin
                        next_state = 4'b0010; // Next state in=1 -> B
                        out = 1'b0; // Output is 0
                    end
                end
            4'b0010: // State B
                begin
                    if(in == 1'b0) begin
                        next_state = 4'b0100; // Next state in=0 -> C
                        out = 1'b0; // Output is 0
                    end
                    else begin
                        next_state = 4'b0010; // Next state in=1 -> B
                        out = 1'b0; // Output is 0
                    end
                end
            4'b0100: // State C
                begin
                    if(in == 1'b0) begin
                        next_state = 4'b0001; // Next state in=0 -> A
                        out = 1'b0; // Output is 0
                    end
                    else begin
                        next_state = 4'b1000; // Next state in=1 -> D
                        out = 1'b0; // Output is 0
                    end
                end
            4'b1000: // State D
                begin
                    if(in == 1'b0) begin
                        next_state = 4'b0100; // Next state in=0 -> C
                        out = 1'b1; // Output is 1
                    end
                    else begin
                        next_state = 4'b0010; // Next state in=1 -> B
                        out = 1'b0; // Output is 0
                    end
                end
            default:
                begin
                    next_state = 4'bxxxx; // Invalid state, next state is undefined
                    out = 1'bx; // Output is undefined
                end
        endcase
    end

endmodule
