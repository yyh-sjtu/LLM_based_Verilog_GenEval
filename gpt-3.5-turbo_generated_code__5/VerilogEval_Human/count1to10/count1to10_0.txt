module top_module(
    input clk,
    input reset,
    output reg [3:0] q);
    
    // Internal wire for holding the intermediate count value
    reg [3:0] count;

    // Always block for synchronous reset
    always @(posedge clk) begin
        if (reset)
            count <= 1;
        else if (count < 10)
            count <= count + 1;
        else
            count <= 1;
    end

    // Assign value of count to output q
    assign q = count;

endmodule
