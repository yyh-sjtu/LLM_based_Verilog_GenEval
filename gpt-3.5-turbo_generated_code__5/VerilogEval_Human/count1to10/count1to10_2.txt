```verilog
module top_module(
    input clk,
    input reset,
    output reg [3:0] q
);

	reg [3:0] count;

	always @(posedge clk or posedge reset) begin
		if(reset)
			count <= 4'b0001;
		else if(count == 4'b1010)
			count <= 4'b0001;
		else
			count <= count + 1;
	end
	
	assign q = count;
	
endmodule
```
