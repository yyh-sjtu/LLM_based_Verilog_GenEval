// Verilog code for a decade counter that counts from 1 to 10

module decade_counter(
    input clk,
    input reset,
    output reg [3:0] q
);

// Define state register to hold the current count value
reg [3:0] count_reg;

// Always block to handle synchronous reset and count logic
always @ (posedge clk or posedge reset)
begin
    if (reset) begin
        count_reg <= 4'b0001; // Reset the counter to 1
    end
    else begin
        if (count_reg == 4'b1010) // If count reaches 10, reset to 1
            count_reg <= 4'b0001;
        else
            count_reg <= count_reg + 4'b0001; // Increment count by 1
    end
end

// Assign output to the count register
assign q = count_reg;

endmodule
