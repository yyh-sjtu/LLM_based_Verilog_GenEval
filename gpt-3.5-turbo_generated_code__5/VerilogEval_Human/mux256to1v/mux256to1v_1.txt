module top_module (
	input [1023:0] in,
	input [7:0] sel,
	output [3:0] out
);
	// Generate 256-bit vectors for each sel value
	reg [255:0] sel0 = in[3:0];
	reg [255:0] sel1 = in[7:4];
	reg [255:0] sel2 = in[11:8];
	// ...

	// Select the appropriate 4-bit value based on sel input
	always @(*) begin
		case (sel)
			0: out = sel0;
			1: out = sel1;
			2: out = sel2;
			// ...
			default: out = 4'b0; // Default case
		endcase
	end
endmodule
