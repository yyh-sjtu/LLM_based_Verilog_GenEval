module top_module (
    input [1023:0] in,
    input [7:0] sel,
    output [3:0] out
);

// Declare intermediate wire variables
wire [7:0] select_start = sel * 4;  // Calculate the starting index of each select range
wire [10:0] sel_range = {select_start + 3'b111, select_start};  // Calculate the range of each select
wire [3:0] mux_output;  // Store the output of the multiplexer

// Implement the multiplexer
always @(*) begin
    case (sel)
        0: mux_output = in[sel_range[10:7]];
        1: mux_output = in[sel_range[10:7]];
        2: mux_output = in[sel_range[10:7]];
        3: mux_output = in[sel_range[10:7]];
        // Add more cases for sel values 4 to 255 if needed
        default: mux_output = 4'b0;  // Default value when sel is out of range
    endcase
end

// Assign output
assign out = mux_output;

endmodule
