module top_module (
    input [1023:0] in,
    input [7:0] sel,
    output [3:0] out
);

reg [255:0] in_vect [0:255];
reg [3:0] out;

generate
    genvar i;
    for (i = 0; i < 256; i = i + 1) begin : PACK_IN
        always @(in[i*4 +: 4]) begin
            in_vect[i] <= in[i*4 +: 4];
        end
    end

    always @(sel) begin
        case (sel)
            0: out <= in_vect[sel];
            1: out <= in_vect[sel];
            2: out <= in_vect[sel];
            // Add more cases for additional sel values if needed
            default: out <= 4'b0;
        endcase
    end
endgenerate

endmodule
