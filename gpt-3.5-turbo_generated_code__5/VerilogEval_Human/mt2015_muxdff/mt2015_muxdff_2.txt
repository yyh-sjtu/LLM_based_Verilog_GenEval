// Define the submodule called "flipflop_mux"
module flipflop_mux(
	input clk,
	input L,
	input q_in,
	input r_in,
	output reg Q
    );

	reg [2:0] q;

	always @(posedge clk) begin
		if (L) begin
			q <= r_in;
		end else begin
			q <= {q[1]^q[2], q[0], q[2]};
		end
	end

	assign Q = q_in;

endmodule

// Define the top module
module top_module(
	input clk,
	input L,
	input q_in,
	input r_in,
	output reg Q
    );

	wire mux_out;

	// Instantiate the submodule three times
	flipflop_mux ff1 (
		.clk(clk),
		.L(L),
		.q_in(q_in),
		.r_in(r_in),
		.Q(mux_out)
	);

	flipflop_mux ff2 (
		.clk(clk),
		.L(L),
		.q_in(q_in),
		.r_in(r_in),
		.Q(mux_out)
	);

	flipflop_mux ff3 (
		.clk(clk),
		.L(L),
		.q_in(q_in),
		.r_in(r_in),
		.Q(mux_out)
	);

	assign Q = mux_out;

endmodule
