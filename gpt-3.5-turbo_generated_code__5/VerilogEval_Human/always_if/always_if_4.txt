// Using assign statements

module mux_assign (
	input a,
	input b,
	input sel,
	output out
);

	assign out = (sel) ? b : a;

endmodule

module top_module (
	input a,
	input b,
	input sel_b1,
	input sel_b2,
	output out_assign,
	output reg out_always
);

	mux_assign mux1 (
		.a(a),
		.b(b),
		.sel(sel_b1 & sel_b2),
		.out(out_assign)
	);

	always @* begin
		if(sel_b1 & sel_b2)
			out_always = b;
		else
			out_always = a;
	end

endmodule
