module top_module(
  input clk,
  input load,
  input [511:0] data,
  output reg [511:0] q);

  // Declare internal wires and registers
  wire [510:0] left_neighbor;
  wire [510:0] right_neighbor;
  reg [511:0] next_state;

  // Assign values to the internal wires and registers
  assign left_neighbor = {q[510:0], 1'b0};
  assign right_neighbor = {1'b0, q[511:1]};
  
  always @(posedge clk) begin
    if (load) begin
      q <= data;
    end else begin
      next_state[510:0] = left_neighbor ^ right_neighbor;
      next_state[511] = 1'b0;
      q <= next_state;
    end
  end
endmodule
