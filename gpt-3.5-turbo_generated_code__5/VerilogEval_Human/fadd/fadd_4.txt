// Full adder module
module full_adder (
    input a,
    input b,
    input cin,
    output cout,
    output sum
);

    assign {cout, sum} = a + b + cin;

endmodule

// Top module connecting full adder
module top_module (
    input a,
    input b,
    input cin,
    output cout,
    output sum
);

    full_adder FA (
        .a(a),
        .b(b),
        .cin(cin),
        .cout(cout),
        .sum(sum)
    );

endmodule
