module full_adder (
  input a,
  input b,
  input cin,
  output cout,
  output sum
);

  assign sum = a ^ b ^ cin;
  assign cout = (a & b) | ((a ^ b) & cin);

endmodule

module top_module (
  input a,
  input b,
  input cin,
  output cout,
  output sum
);

  full_adder fa (
    .a(a),
    .b(b),
    .cin(cin),
    .cout(cout),
    .sum(sum)
  );

endmodule
