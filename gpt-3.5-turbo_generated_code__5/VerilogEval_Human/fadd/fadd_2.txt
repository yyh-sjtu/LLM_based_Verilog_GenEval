module top_module (
	input a,
	input b,
	input cin,
	output cout,
	output sum
);
	
	// Internal wires declaration
	wire w1;
	wire w2;
	wire w3;
	
	// Full adder logic implementation
	assign w1 = a ^ b;
	assign sum = w1 ^ cin;
	assign w2 = a & b;
	assign w3 = cin & w1;
	assign cout = w2 | w3;
	
endmodule
