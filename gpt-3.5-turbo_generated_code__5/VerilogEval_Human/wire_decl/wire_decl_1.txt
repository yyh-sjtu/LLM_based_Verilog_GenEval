module top_module (
    input a,
    input b,
    input c,
    input d,
    output out,
    output out_n );

    wire intermediate_wire_1;
    wire intermediate_wire_2;
    
    and_gate and1(a, b, intermediate_wire_1);
    and_gate and2(c, d, intermediate_wire_2);
    or_gate or1(intermediate_wire_1, intermediate_wire_2, out);
    not_gate not1(out, out_n);
    
endmodule

module and_gate (
    input input_a,
    input input_b,
    output output);

    assign output = input_a & input_b;

endmodule

module or_gate (
    input input_a,
    input input_b,
    output output);

    assign output = input_a | input_b;

endmodule

module not_gate (
    input input,
    output output);

    assign output = ~input;

endmodule
