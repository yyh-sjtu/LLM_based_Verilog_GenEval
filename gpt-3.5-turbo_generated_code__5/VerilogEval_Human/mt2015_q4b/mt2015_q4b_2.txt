// Declare the inputs and outputs
	reg x_reg, y_reg, z_reg;

	// Sequential logic to update the registers
	always @(posedge clk) begin
		x_reg <= x;
		y_reg <= y;
		
		if (x_reg & y_reg)
			z_reg <= 1'b1;
		else if (!x_reg & !y_reg)
			z_reg <= 1'b0;
	end

	// Assign the output
	assign z = z_reg;
endmodule
