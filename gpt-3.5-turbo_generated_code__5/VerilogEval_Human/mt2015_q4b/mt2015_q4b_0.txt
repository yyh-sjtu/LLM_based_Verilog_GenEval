```verilog
module top_module(
   input x,
   input y,
   output z);
   
   reg z_reg;
   
   always @(*)
   begin
      case({x,y})
         2'b00   : z_reg = 1'b1;

         2'b11   : z_reg = 1'b1;

      endcase
   end
   
   assign z = z_reg;
   
endmodule
```

Inside the `always` block, a case statement is used to determine the appropriate value for `z_reg` based on the values of `x` and `y`. The final value of `z_reg` is then assigned to the output `z` using the `assign` statement.
