module top_module (
    input clk,
    input areset,
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output walk_left,
    output walk_right,
    output aaah,
    output digging
);

    reg [4:0] state;
    reg walk_left, walk_right, aaah_output, digging_output;
    reg [4:0] counter;

    parameter IDLE = 0;
    parameter WALK_LEFT = 1;
    parameter WALK_RIGHT = 2;
    parameter FALL = 3;
    parameter DIGGING = 4;

    always @(posedge clk or posedge areset) begin
        if (areset) begin
            state <= IDLE;
            walk_left <= 0;
            walk_right <= 0;
            aaah_output <= 0;
            digging_output <= 0;
            counter <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (ground) begin
                        if (bump_left && bump_right) begin
                            state <= IDLE;
                        end else if (bump_left) begin
                            state <= WALK_RIGHT;
                        end else if (bump_right) begin
                            state <= WALK_LEFT;
                        end else if (dig) begin
                            state <= DIGGING;
                        end else begin
                            state <= IDLE;
                        end
                    end else begin
                        state <= FALL;
                    end
                end
                WALK_LEFT: begin
                    if (ground) begin
                        walk_left <= 1;
                        walk_right <= 0;
                        aaah_output <= 0;
                        digging_output <= 0;
                        if (bump_right) begin
                            state <= WALK_RIGHT;
                        end else if (dig) begin
                            state <= DIGGING;
                        end else begin
                            state <= IDLE;
                        end
                    end else begin
                        state <= FALL;
                    end
                end
                WALK_RIGHT: begin
                    if (ground) begin
                        walk_left <= 0;
                        walk_right <= 1;
                        aaah_output <= 0;
                        digging_output <= 0;
                        if (bump_left) begin
                            state <= WALK_LEFT;
                        end else if (dig) begin
                            state <= DIGGING;
                        end else begin
                            state <= IDLE;
                        end
                    end else begin
                        state <= FALL;
                    end
                end
                FALL: begin
                    aaah_output <= 1;
                    walk_left <= 0;
                    walk_right <= 0;
                    digging_output <= 0;
                    if (ground) begin
                        state <= IDLE;
                    end else if (counter == 20) begin
                        state <= IDLE;
                    end else begin
                        state <= FALL;
                    end
                end
                DIGGING: begin
                    if (ground) begin
                        walk_left <= 0;
                        walk_right <= 0;
                        aaah_output <= 0;
                        digging_output <= 1;
                        if (bump_left && bump_right) begin
                            state <= IDLE;
                        end else if (bump_left) begin
                            state <= WALK_RIGHT;
                        end else if (bump_right) begin
                            state <= WALK_LEFT;
                        end else begin
                            state <= DIGGING;
                        end
                    end else begin
                        state <= FALL;
                    end
                end
            endcase
        end
    end

    always @(posedge clk or posedge areset) begin
        if (areset) begin
            counter <= 0;
        end else begin
            if (state == FALL && !ground) begin
                counter <= counter + 1;
            end else begin
                counter <= 0;
            end
        end
    end

    assign aaah = aaah_output;
    assign digging = digging_output;

endmodule
