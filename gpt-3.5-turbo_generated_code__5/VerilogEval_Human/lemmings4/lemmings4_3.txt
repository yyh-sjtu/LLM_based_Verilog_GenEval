module lemming_fsm (
    input clk,
    input areset,
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output reg walk_left,
    output reg walk_right,
    output reg aaah,
    output reg digging
);

    reg [4:0] state;
    reg [4:0] next_state;

    parameter WALK_LEFT = 3'b001;
    parameter WALK_RIGHT = 3'b010;
    parameter FALL = 3'b011;
    parameter DIG = 3'b100;
    parameter SPLATTER = 3'b101;

    always @(posedge clk or posedge areset) begin
        if (areset) begin
            state <= WALK_LEFT;
        end else begin
            state <= next_state;
        end
    end

    always @* begin
        next_state = state;
        walk_left = 0;
        walk_right = 0;
        aaah = 0;
        digging = 0;

        case (state)
            WALK_LEFT: begin
                if (bump_right || (bump_left && !bump_right)) begin
                    next_state = WALK_RIGHT;
                end else if (!ground) begin
                    next_state = FALL;
                end else if (dig) begin
                    next_state = DIG;
                end
                walk_left = 1;
            end

            WALK_RIGHT: begin
                if (bump_left || (bump_right && !bump_left)) begin
                    next_state = WALK_LEFT;
                end else if (!ground) begin
                    next_state = FALL;
                end else if (dig) begin
                    next_state = DIG;
                end
                walk_right = 1;
            end

            FALL: begin
                if (ground) begin
                    next_state = state;
                end else begin
                    if (dig) begin
                        next_state = DIG;
                    end else begin
                        next_state = FALL;
                    end
                end
                aaah = 1;
            end

            DIG: begin
                if (ground) begin
                    next_state = state;
                end else begin
                    next_state = FALL;
                end
                digging = 1;
            end

            SPLATTER: begin
                // No transitions, all outputs are 0
            end
        endcase
    end

    always @(posedge clk or posedge areset) begin
        if (areset) begin
            walk_left <= 0;
            walk_right <= 0;
            aaah <= 0;
            digging <= 0;
        end else begin
            case (state)
                SPLATTER: begin
                    walk_left <= 0;
                    walk_right <= 0;
                    aaah <= 0;
                    digging <= 0;
                end
                default: begin
                    walk_left <= walk_left;
                    walk_right <= walk_right;
                    aaah <= aaah;
                    digging <= digging;
                end
            endcase
        end
    end
endmodule

module top_module (
    input clk,
    input areset,
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output walk_left,
    output walk_right,
    output aaah,
    output digging
);

    wire walk_left_wire, walk_right_wire, aaah_wire, digging_wire;

    lemming_fsm fsm (
        .clk(clk),
        .areset(areset),
        .bump_left(bump_left),
        .bump_right(bump_right),
        .ground(ground),
        .dig(dig),
        .walk_left(walk_left_wire),
        .walk_right(walk_right_wire),
        .aaah(aaah_wire),
        .digging(digging_wire)
    );

    assign walk_left = walk_left_wire;
    assign walk_right = walk_right_wire;
    assign aaah = aaah_wire;
    assign digging = digging_wire;
endmodule
