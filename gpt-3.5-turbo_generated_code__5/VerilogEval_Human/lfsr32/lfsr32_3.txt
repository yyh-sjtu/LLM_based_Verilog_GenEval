module galois_lfsr(
  input clk,
  input reset,
  output reg [31:0] q
);

  reg [31:0] q_next;

  always @(posedge clk or posedge reset) begin
    if (reset) begin
      q <= 32'h1;
      q_next <= 32'h1;
    end else begin
      q <= q_next;
      q_next[31] <= q[1] ^ q[2] ^ q[21] ^ q[31];
      q_next[30:0] <= q[31:1];
    end
  end

endmodule

module top_module(
  input clk,
  input reset,
  output reg [31:0] q
);

  galois_lfsr lfsr(
    .clk(clk),
    .reset(reset),
    .q(q)
  );

endmodule
