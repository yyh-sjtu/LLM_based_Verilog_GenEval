
   reg [98:0] out_both;
   reg [99:1] out_any;
   reg [99:0] out_different;

   integer i;

   // Output for out_both
   always @* begin
      for (i = 0; i < 99; i = i + 1) begin
         out_both[i] = (in[i] & in[i+1]);
      end
   end

   // Output for out_any
   always @* begin
      for (i = 1; i < 100; i = i + 1) begin
         out_any[i] = (in[i] | in[i-1]);
      end
   end
   out_any[0] = in[0]; // For the case where there is no neighbour to the right

   // Output for out_different
   always @* begin
      for (i = 0; i < 100; i = i + 1) begin
         out_different[i] = (in[i] != in[(i+99)%100]);
      end
   end

endmodule
