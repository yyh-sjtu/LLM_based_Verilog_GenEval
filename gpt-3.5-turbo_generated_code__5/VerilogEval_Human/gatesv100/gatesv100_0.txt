    output [98:0] out_both,
    output [99:1] out_any,
    output [99:0] out_different

assign out_both[0] = in[0] & in[1];

        assign out_both[i] = in[i] & in[i+1];
    end
endgenerate

assign out_any[99] = in[99] | in[0]; // Wrap around for in[99]
generate
    genvar j;

        assign out_any[j] = in[j] | in[j+1];
    end

// Logic for out_different
assign out_different[0] = in[0] ^ in[99]; // Wrap around for in[99]

    genvar k;

    end
