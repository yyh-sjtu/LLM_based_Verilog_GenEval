```verilog
module top_module (
	input clk,
	input areset,
	input x,
	output reg z
);

reg [2:0] state;
parameter S0 = 3'b000, S1 = 3'b001, S2 = 3'b010, S3 = 3'b011;

always @(posedge clk or negedge areset) begin
	if (!areset) begin
		state <= S0;
		z <= 0;
	end else begin
		case (state)
			S0: begin
				if (x == 1'b1) begin
					state <= S1;
					z <= 1;
				end else begin
					state <= S0;
					z <= 0;
				end
			end
			S1: begin
				if (x == 1'b1) begin
					state <= S1;
					z <= 0;
				end else begin
					state <= S2;
					z <= 1;
				end
			end
			S2: begin
				if (x == 1'b1) begin
					state <= S2;
					z <= 1;
				end else begin
					state <= S2;
					z <= 0;
				end
			end
			S3: begin
				state <= S3;
				z <= 1;
			end
		endcase
	end
end

endmodule
```
