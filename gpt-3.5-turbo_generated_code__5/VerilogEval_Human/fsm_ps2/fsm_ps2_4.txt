```verilog
module top_module (
	input clk,
	input [7:0] in,
	input reset,
	output done
);

typedef enum {IDLE, BYTE1, BYTE2, BYTE3} states;
reg [1:0] state, next_state;

always @ (posedge clk or posedge reset) begin
	if (reset) begin
		state <= IDLE;
	end else begin
		state <= next_state;
	end
end

always @* begin
	case (state)
		IDLE: begin
			if (in[3] == 1) begin
				next_state = BYTE1;
			end else begin
				next_state = IDLE;
			end
		end
		BYTE1: begin
			next_state = BYTE2;
		end
		BYTE2: begin
			next_state = BYTE3;
		end
		BYTE3: begin
			next_state = IDLE;
		end
		default: begin
			next_state = IDLE;
		end
	endcase
end

reg done_reg;
always @ (posedge clk or posedge reset) begin
	if (reset) begin
		done_reg <= 0;
	end else if (state == BYTE3) begin
		done_reg <= 1;
	end else begin
		done_reg <= 0;
	end
end

assign done = done_reg;

endmodule
```
