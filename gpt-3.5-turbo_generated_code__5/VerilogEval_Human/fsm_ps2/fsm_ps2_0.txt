// Define states for the finite state machine
parameter IDLE = 0;
parameter BYTE1 = 1;
parameter BYTE2 = 2;
parameter BYTE3 = 3;

reg [1:0] state;
reg done;

always @ (posedge clk or posedge reset) begin
	if (reset) begin
		state <= IDLE;
		done <= 0;
	end else begin
		case (state)
			IDLE: begin
				if (in[3] == 1) begin
					state <= BYTE1;
				end
			end

			BYTE1: begin
				state <= BYTE2;
			end

			BYTE2: begin
				state <= BYTE3;
			end

			BYTE3: begin
				state <= IDLE;
				done <= 1;
			end
		endcase
	end
end

assign done = done;

endmodule
