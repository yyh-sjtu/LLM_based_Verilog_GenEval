// Define states using enum
typedef enum logic [1:0] {B, A} state_t;
state_t current_state, next_state;

// Define internal signals
reg out_reg;

// Define combinational logic for next state and output generation
always_comb begin
	if (reset)
		next_state = B;
	else begin
		case (current_state)
			B: begin
				if (in)
					next_state = B;
				else
					next_state = A;
			end
			A: begin
				if (in)
					next_state = A;
				else
					next_state = B;
			end
		endcase
	end
end

// Define sequential logic for state update and output assignment
always_ff @(posedge clk) begin
	if (reset)
		current_state <= B;
	else
		current_state <= next_state;

	case (current_state)
		B: out_reg <= 1;
		A: out_reg <= 0;
	endcase
end

// Assign output
assign out = out_reg;

endmodule
