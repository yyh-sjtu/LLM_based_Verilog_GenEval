module top_module (
	input clk,
	input in,
	input reset,
	output out
);

reg [1:0] state;
reg out;

always @(posedge clk)
begin
    if(reset)
        state <= 2'b10;
    else
    begin
        case(state)
            2'b10: // B
                begin
                    if(in)
                        state <= 2'b10;
		            else
                        state <= 2'b01;
                end
            2'b01: // A
                begin
                    if(in)
                        state <= 2'b01;
		            else
                        state <= 2'b10;
                end
        endcase
    end
end

always @(state)
begin
    case(state)
        2'b10: // B
            out = 1'b1;
        2'b01: // A
            out = 1'b0;
    endcase
end

endmodule
