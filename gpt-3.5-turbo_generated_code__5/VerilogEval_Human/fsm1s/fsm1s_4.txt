
  // Define states
  parameter B = 2'b00;
  parameter A = 2'b01;
  
  // Define state register
  reg [1:0] state, next_state;
  
  // Define output register
  reg out_reg;
  
  // Moore state machine logic
  always @(posedge clk or posedge reset) begin
    if (reset) begin
      state <= B;
      out_reg <= 1;
    end
    else begin
      state <= next_state;
      out_reg <= state == B ? 1 : 0;
    end
  end
  
  // State transition and output logic
  always @* begin
    case (state)
      B: begin
        next_state = in ? B : A;
      end
      A: begin
        next_state = in ? A : B;
      end
    endcase
  end
  
  // Assign output
  assign out = out_reg;
  
endmodule
