// Define state machine states
typedef enum logic [1:0] {
    IDLE = 0,
    DETECT = 1,
    SHIFT = 2,
    COUNT = 3,
    WAIT_ACK = 4
} state_type;

// Define constants
localparam logic [3:0] COUNT_CYCLES = 1000;

// Define module
reg [3:0] delay;
reg [3:0] count_reg;
reg [1:0] state;
reg start_detected;
reg shift_done;

// Declare wires
wire data_detected;
wire [3:0] remaining_time;

// Counter to keep track of clock cycles
reg [11:0] counter;

// Assign wires
assign data_detected = (data && !start_detected) && (data << 3'b101);

// State machine logic
always @(posedge clk or posedge reset) begin
    if (reset) begin
        state <= IDLE;
        count_reg <= 4'b0;
        counting <= 0;
        done <= 0;
        counter <= 0;
        shift_done <= 0;
        start_detected <= 0;
        delay <= 4'b0;
    end else begin
        case (state)
            IDLE: begin
                if (data_detected) begin
                    start_detected <= 1;
                    state <= DETECT;
                end else begin
                    start_detected <= 0;
                    state <= IDLE;
                end
                counting <= 0;
                done <= 0;
                counter <= 0;
            end
            DETECT: begin
                if (counter >= 4) begin
                    start_detected <= 0;
                    state <= SHIFT;
                end else begin
                    start_detected <= 1;
                    state <= DETECT;
                end
                counting <= 0;
                done <= 0;
                counter <= counter + 1;
            end
            SHIFT: begin
                if (counter >= 8) begin
                    shift_done <= 1;
                    delay <= count_reg;
                    state <= COUNT;
                end else begin
                    shift_done <= 0;
                    state <= SHIFT;
                end
                counting <= 0;
                done <= 0;
                counter <= counter + 1;
            end
            COUNT: begin
                if (counter >= (COUNT_CYCLES * (delay + 1))) begin
                    counting <= 0;
                    done <= 1;
                    state <= WAIT_ACK;
                end else begin
                    counting <= 1;
                    done <= 0;
                    counter <= counter + 1;
                    state <= COUNT;
                end
            end
            WAIT_ACK: begin
                if (ack) begin
                    state <= IDLE;
                end else begin
                    state <= WAIT_ACK;
                end
                counting <= 0;
                done <= 0;
                counter <= 0;
                start_detected <= 0;
                delay <= 4'b0;
            end
            default: state <= IDLE;
        endcase
    end
end

// Assign remaining_time based on delay and counter
always @(*) begin
    case (state)
        IDLE, DETECT, SHIFT, WAIT_ACK: remaining_time = delay;
        COUNT: remaining_time = delay - (counter / COUNT_CYCLES);
        default: remaining_time = 4'b0;
    endcase
end

// Assign outputs
assign count = count_reg;
assign remaining_time = remaining_time;

endmodule
