// Declare internal signals
reg [3:0] delay;
reg [3:0] count_temp;
reg [2:0] state;
reg [11:0] timer;

// Define states
parameter IDLE = 3'd0;
parameter START = 3'd1;
parameter COUNTING = 3'd2;
parameter DONE = 3'd3;

always @(posedge clk) begin
    if (reset) begin
        delay <= 4'b0000;
        counting <= 0;
        done <= 0;
        state <= IDLE;
        count <= 4'bxxxx;
        timer <= 12'd0;
    end else begin
        case (state)
            IDLE: begin
                if (data == 4'b1101) begin
                    state <= START;
                end
            end
            START: begin
                delay <= {data, data, data, data}; // Shift in next 4 bits
                state <= COUNTING;
                counting <= 1;
            end
            COUNTING: begin
                if (timer < {(delay + 1) * 1000}) begin
                    timer <= timer + 1;
                    if (timer % 1000 == 0) begin
                        count_temp <= count_temp - 1;
                    end
                end else begin
                    state <= DONE;
                    counting <= 0;
                    done <= 1;
                end
            end
            DONE: begin
                if (ack) begin
                    state <= IDLE;
                    done <= 0;
                end
            end
        endcase
    end
end

assign count = (state == COUNTING) ? count_temp : 4'bxxxx;
assign count = (state == COUNTING) ? count_temp : 4'bxxxx;

endmodule
