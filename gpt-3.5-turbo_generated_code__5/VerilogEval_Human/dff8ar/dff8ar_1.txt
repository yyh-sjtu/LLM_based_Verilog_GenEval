module dff_with_reset(
    input clk,
    input d,
    input areset,
    output reg q
);

always @(posedge clk or posedge areset) begin
    if (areset) begin
        q <= 8'b0;
    end else begin
        q <= d;
    end
end

endmodule

module top_module(
    input clk,
    input [7:0] d,
    input areset,
    output reg [7:0] q
);

reg [7:0] q_temp;
reg [7:0] d_temp;

// Instantiate 8 D flip-flops with reset
dff_with_reset dff0(clk, d_temp[0], areset, q_temp[0]);
dff_with_reset dff1(clk, d_temp[1], areset, q_temp[1]);
dff_with_reset dff2(clk, d_temp[2], areset, q_temp[2]);
dff_with_reset dff3(clk, d_temp[3], areset, q_temp[3]);
dff_with_reset dff4(clk, d_temp[4], areset, q_temp[4]);
dff_with_reset dff5(clk, d_temp[5], areset, q_temp[5]);
dff_with_reset dff6(clk, d_temp[6], areset, q_temp[6]);
dff_with_reset dff7(clk, d_temp[7], areset, q_temp[7]);

always @(posedge clk) begin
    q <= q_temp;
    d_temp <= d;
end

endmodule
