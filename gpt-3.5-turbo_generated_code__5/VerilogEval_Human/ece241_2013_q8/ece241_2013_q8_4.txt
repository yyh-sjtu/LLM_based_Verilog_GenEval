module fsm_101 (
	input clk,
	input aresetn,
	input x,
	output reg z
);

// Define states
parameter S0 = 2'b00;
parameter S1 = 2'b01;
parameter S2 = 2'b10;

// Define next state logic
always @(posedge clk or negedge aresetn) begin
	if (~aresetn) begin
		state <= S0;
	elseif (state == S0) begin
		if (x) begin
			state <= S1;
		else
			state <= S0;
	end
	elseif (state == S1) begin
		if (!x) begin
			state <= S0;
		elseif (x) begin
			state <= S2;
	end
	elseif (state == S2) begin
		if (x) begin
			state <= S1;
		else
			state <= S0;
	end
	else begin
		state <= S0;
	end
end

// Define output logic
always @(state, x) begin
	if (state == S2 && !x) begin
		z <= 1;
	end
	else begin
		z <= 0;
	end
end

endmodule
