module shift_register(
input clk,
input areset,
input load,
input ena,
input [3:0] data,
output reg [3:0] q
);

    always @(posedge clk or posedge areset) begin
        if (areset) begin
            q <= 4'b0;
        end else if (load) begin
            q <= data;
        end else if (ena) begin
            q <= {q[2:0], 1'b0};
        end
    end

endmodule

module top_module(
input clk,
input areset,
input load,
input ena,
input [3:0] data,
output reg [3:0] q
);

    shift_register sr (
        .clk(clk),
        .areset(areset),
        .load(load),
        .ena(ena),
        .data(data),
        .q(q)
    );

endmodule
