module decade_counter(
  input clk,
  input slowena,
  input reset,
  output reg [3:0] q
);

  reg [3:0] count;

  always @(posedge clk) begin
    if (reset) begin
      count <= 4'b0;
    end else if (slowena) begin
      count <= count + 1;
    end
  end

  always @(posedge clk) begin
    if (reset) begin
      q <= 4'b0;
    end else if (slowena) begin
      q <= count;
    end
  end

endmodule

module top_module(
  input clk,
  input slowena,
  input reset,
  output reg [3:0] q
);

  decade_counter counter(
    .clk(clk),
    .slowena(slowena),
    .reset(reset),
    .q(q)
  );

endmodule
