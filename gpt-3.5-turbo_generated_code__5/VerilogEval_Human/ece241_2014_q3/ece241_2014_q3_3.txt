wire ab_00, ab_01, ab_11, ab_10;

// generate value of ab for each input combination of c and d
assign ab_00 = ~c & ~d;
assign ab_01 = ~c & d;
assign ab_11 = c & d;
assign ab_10 = c & ~d;

// connect ab values to mux_in
assign mux_in[0] = ab_00;
assign mux_in[1] = ab_01;
assign mux_in[2] = ab_11;
assign mux_in[3] = ab_10;

endmodule
