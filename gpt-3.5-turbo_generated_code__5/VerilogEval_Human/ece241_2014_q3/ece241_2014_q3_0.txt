module top_module (
	input c,
	input d,
	output [3:0] mux_in
);

wire m0, m1, m2, m3;

// Implementing the Karnaugh Map using one 4-to-1 multiplexer and as few 2-to-1 multiplexers as possible
// Output connections based on the Karnaugh Map
assign m0 = (~c & d);
assign m1 = (c & d);
assign m2 = (c | ~d);
assign m3 = (c & ~d);

// Connecting the outputs of the Karnaugh Map to the 4-to-1 multiplexer inputs
assign mux_in[0] = (m0 & ~c & ~d) | (m1 & ~c & d) | (m2 & c & d) | (m3 & c & ~d);
assign mux_in[1] = (m0 & ~c & ~d) | (m1 & ~c & d) | (m2 & c & d) | (m3 & c & ~d);
assign mux_in[2] = (m0 & ~c & ~d) | (m1 & ~c & d) | (m2 & c & d) | (m3 & c & ~d);
assign mux_in[3] = (m0 & ~c & ~d) | (m1 & ~c & d) | (m2 & c & d) | (m3 & c & ~d);

endmodule
