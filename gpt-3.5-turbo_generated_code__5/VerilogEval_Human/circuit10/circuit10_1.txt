// Defining the state and q signals
reg state, q;

// Defining the flip-flop and the combinational logic
always @(posedge clk) begin
	if (a == 0 && b == 1) begin
		state <= 1;
		q <= 1;
	end
	else if (a == 1 && b == 0) begin
		state <= 0;
		q <= 0;
	end
	else if (a == 1 && b == 1) begin
		state <= state;
		q <= state;
	end
	else begin
		state <= 0;
		q <= 0;
	end
end

// Assigning the outputs
assign state = state;
assign q = q;
endmodule
