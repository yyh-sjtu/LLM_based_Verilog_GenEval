module top_module (
  input clk,
  input a,
  input b,
  output q,
  output state
);

  reg q_reg;
  
  always @(posedge clk) begin
    if (a && !b && !q_reg)
      q_reg <= 1;
    else if (q_reg && (!a || b))
      q_reg <= 0;
    else
      q_reg <= q_reg;
  end

  assign q = q_reg;
  assign state = (a && !b && !q_reg) ? 1 : 0;

endmodule
