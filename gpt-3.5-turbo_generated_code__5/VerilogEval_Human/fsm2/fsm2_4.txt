module top_module (
    input clk,
    input j,
    input k,
    input areset,
    output reg out
);
    
    // Define states
    localparam OFF = 1'b0;
    localparam ON = 1'b1;
    
    // Define current state variable
    reg currentState;
    
    always @(posedge clk or posedge areset) begin
        if (areset)
            currentState <= OFF; // Reset to OFF state
        else begin
            case (currentState)
                OFF: begin
                    if (j)
                        currentState <= ON; // Transition to ON state if j=1
                    else
                        currentState <= OFF; // Stay in OFF state if j=0
                end
                
                ON: begin
                    if (k)
                        currentState <= OFF; // Transition to OFF state if k=1
                    else
                        currentState <= ON; // Stay in ON state if k=0
                end
            endcase
        end
    end
    
    always @* begin
        case (currentState)
            OFF: out = 1'b0; // Output is 0 in OFF state
            ON: out = 1'b1; // Output is 1 in ON state
        endcase
    end
endmodule
