module top_module (
	input clk,
	input areset,
	input x,
	output z
);

enum logic [1:0] {A, B} state;

always_ff @(posedge clk or posedge areset) begin
	if (areset)
		state <= A;
	else begin
		case(state)
			A: if (x) state <= B; else state <= A;
			B: if (x) state <= A; else state <= B;
		endcase
	end
end

assign z = (state == B) ? 1'b0 : 1'b1;

endmodule
