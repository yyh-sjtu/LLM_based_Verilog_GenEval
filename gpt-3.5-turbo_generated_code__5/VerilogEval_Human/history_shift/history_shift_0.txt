module top_module
(
    input clk,
    input areset,
    input predict_valid,
    input predict_taken,
    output logic [31:0] predict_history,
    
    input train_mispredicted,
    input train_taken,
    input [31:0] train_history
);

    logic [31:0] global_history_shift_reg; // Global history shift register
    
    always @(posedge clk or posedge areset) begin
        if (areset) begin
            global_history_shift_reg <= 0; // Reset the shift register to zero
        end 
        else begin
            if (train_mispredicted) begin
                global_history_shift_reg <= {train_history, train_taken}; // Load the shift register with train_history concatenated with train_taken
            end
            else if (predict_valid) begin
                global_history_shift_reg <= {global_history_shift_reg[30:0], predict_taken}; // Shift in predict_taken from the LSB side
            end
        end
    end
    
    assign predict_history = global_history_shift_reg;
    
endmodule
