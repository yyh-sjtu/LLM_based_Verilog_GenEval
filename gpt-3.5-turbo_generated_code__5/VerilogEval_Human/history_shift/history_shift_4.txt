    // define 32-bit shift register for branch history
    logic [31:0] history_register;

    // update history register in response to prediction and misprediction
    always @(posedge clk or posedge areset) begin
        if (areset) begin
            history_register <= 32'b0;
        end else begin
            if (train_mispredicted) begin
                history_register <= {train_history[30:0], train_taken};
            end else if (predict_valid) begin
                history_register <= {history_register[30:0], predict_taken};
            end
        end
    end

    // output the current value of the history register
    assign predict_history = history_register;
endmodule
