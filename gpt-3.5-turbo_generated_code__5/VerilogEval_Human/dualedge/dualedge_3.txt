// Implementing a dual-edge triggered flip-flop using two D flip-flops

// First D flip-flop triggered on the positive edge of the clock
reg q1;
always @(posedge clk)
begin
	q1 <= d;
end

// Second D flip-flop triggered on the negative edge of the clock
reg q2;
always @(negedge clk)
begin
	q2 <= q1;
end

// Output q is taken from the second D flip-flop
assign q = q2;

endmodule
