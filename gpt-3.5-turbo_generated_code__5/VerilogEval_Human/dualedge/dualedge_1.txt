module top_module(
	input clk,
	input d,
	output reg q);

	// Flip flop registers
	reg q1;
	reg q2;

	// Edge detection signals
	wire pos_edge;
	wire neg_edge;

	// Detect positive and negative edges
	assign pos_edge = clk & ~q1;
	assign neg_edge = ~clk & q1;

	// Hold the value of d at positive edge
	always @(posedge clk)
	begin
		if (pos_edge)
			q1 <= d;
	end

	// Hold the value of q1 at negative edge
	always @(negedge clk)
	begin
		if (neg_edge)
			q2 <= q1;
	end

	// Assign the value of q2 to the output
	always @*
	begin
		q = q2;
	end

endmodule
