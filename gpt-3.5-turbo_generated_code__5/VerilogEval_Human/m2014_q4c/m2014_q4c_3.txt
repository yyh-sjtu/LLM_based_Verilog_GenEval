module d_ff_reset(
    input clk,
    input d,
    input r,
    output logic q
    );

    always_ff@(posedge clk) begin
        if (r)
            q <= 0;
        else
            q <= d;
    end
    
endmodule

module top_module(
    input clk,
    input d,
    input r,
    output logic q
    );

    d_ff_reset d_ff(
        .clk(clk),
        .d(d),
        .r(r),
        .q(q)
    );
    
endmodule
