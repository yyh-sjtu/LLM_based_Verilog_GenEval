module top_module (
	input clk,
	input areset,
	input bump_left,
	input bump_right,
	input ground,
	input dig,
	output reg walk_left,
	output reg walk_right,
	output reg aaah,
	output reg digging
);
	parameter WALK_LEFT_STATE = 2'b00;
	parameter WALK_RIGHT_STATE = 2'b01;
	parameter FALL_STATE = 2'b10;
	
	reg [1:0] state;
	
	always @(posedge clk or posedge areset) begin
		if (areset) begin
			state <= WALK_LEFT_STATE;
			walk_left <= 1'b1;
			walk_right <= 1'b0;
			aaah <= 1'b0;
			digging <= 1'b0;
		end else begin
			case (state)
				WALK_LEFT_STATE: begin
					if (bump_right) begin
						state <= WALK_RIGHT_STATE;
						walk_left <= 1'b0;
						walk_right <= 1'b1;
						aaah <= 1'b0;
						digging <= 1'b0;
					end else if (dig && ground) begin
						state <= FALL_STATE;
						walk_left <= 1'b0;
						walk_right <= 1'b0;
						aaah <= 1'b1;
						digging <= 1'b1;
					end else if (!ground) begin
						state <= FALL_STATE;
						walk_left <= 1'b0;
						walk_right <= 1'b0;
						aaah <= 1'b1;
						digging <= 1'b0;
					end 
				end
				WALK_RIGHT_STATE: begin
					if (bump_left) begin
						state <= WALK_LEFT_STATE;
						walk_left <= 1'b1;
						walk_right <= 1'b0;
						aaah <= 1'b0;
						digging <= 1'b0;
					end else if (dig && ground) begin
						state <= FALL_STATE;
						walk_left <= 1'b0;
						walk_right <= 1'b0;
						aaah <= 1'b1;
						digging <= 1'b1;
					end else if (!ground) begin
						state <= FALL_STATE;
						walk_left <= 1'b0;
						walk_right <= 1'b0;
						aaah <= 1'b1;
						digging <= 1'b0;
					end 
				end
				FALL_STATE: begin
					if (ground) begin
						state <= (walk_left) ? WALK_LEFT_STATE : WALK_RIGHT_STATE;
						walk_left <= (walk_left) ? 1'b1 : 1'b0;
						walk_right <= (walk_left) ? 1'b0 : 1'b1;
						aaah <= 1'b0;
						digging <= 1'b0;
					end 
				end
			endcase
		end
	end
endmodule
