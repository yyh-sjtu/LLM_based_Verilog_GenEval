module Lemming(
    input clk,
    input areset,
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output reg walk_left,
    output reg walk_right,
    output reg aaah,
    output reg digging
);
    
    reg [2:0] state;
    reg [2:0] next_state;
    
    parameter WALKING = 3'b000;
    parameter FALLING = 3'b001;
    parameter DIGGING = 3'b010;
    parameter SWITCH_LEFT = 3'b011;
    parameter SWITCH_RIGHT = 3'b100;
    
    always @(posedge clk or posedge areset) begin
        if (areset) begin
            state <= WALKING;
        end else begin
            state <= next_state;
        end
    end
    
    always @(*) begin
        case(state)
            WALKING: begin
                walk_left = 1;
                walk_right = 0;
                aaah = 0;
                digging = 0;
                
                if(bump_left) begin
                    next_state = SWITCH_RIGHT;
                end else if(bump_right) begin
                    next_state = SWITCH_LEFT;
                end else if(ground) begin
                    next_state = WALKING;
                end else if(dig) begin
                    next_state = DIGGING;
                end else begin
                    next_state = FALLING;
                end
            end
            
            FALLING: begin
                walk_left = 0;
                walk_right = 0;
                aaah = 1;
                digging = 0;
                
                if(ground) begin
                    next_state = WALKING;
                end else if(dig) begin
                    next_state = DIGGING;
                end else begin
                    next_state = FALLING;
                end
            end
            
            DIGGING: begin
                walk_left = 0;
                walk_right = 0;
                aaah = 0;
                digging = 1;
                
                if(ground) begin
                    next_state = DIGGING;
                end else begin
                    next_state = FALLING;
                end
            end
            
            SWITCH_LEFT: begin
                walk_left = 1;
                walk_right = 0;
                aaah = 0;
                digging = 0;
                
                if(ground) begin
                    next_state = WALKING;
                end else if(dig) begin
                    next_state = DIGGING;
                end else begin
                    next_state = FALLING;
                end
            end
            
            SWITCH_RIGHT: begin
                walk_left = 0;
                walk_right = 1;
                aaah = 0;
                digging = 0;
                
                if(ground) begin
                    next_state = WALKING;
                end else if(dig) begin
                    next_state = DIGGING;
                end else begin
                    next_state = FALLING;
                end
            end
        endcase
    end
endmodule

module top_module(
    input clk,
    input areset,
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output reg walk_left,
    output reg walk_right,
    output reg aaah,
    output reg digging
);
    
    Lemming lemming(
        .clk(clk),
        .areset(areset),
        .bump_left(bump_left),
        .bump_right(bump_right),
        .ground(ground),
        .dig(dig),
        .walk_left(walk_left),
        .walk_right(walk_right),
        .aaah(aaah),
        .digging(digging)
    );
    
endmodule
