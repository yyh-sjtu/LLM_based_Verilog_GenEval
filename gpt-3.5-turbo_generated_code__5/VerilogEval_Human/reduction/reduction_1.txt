module parity_checker (
  input [7:0] in,
  output parity
);

  wire [7:0] xor_out;
  wire xor_all;

  // XOR all the input bits
  assign xor_out[7] = in[0] ^ in[1] ^ in[2] ^ in[3] ^ in[4] ^ in[5] ^ in[6] ^ in[7];

  // XOR all the XOR output bits to get the final parity bit
  assign xor_all = xor_out[0] ^ xor_out[1] ^ xor_out[2] ^ xor_out[3] ^ xor_out[4] ^ xor_out[5] ^ xor_out[6] ^ xor_out[7];

  // The output of the parity bit
  assign parity = xor_all;

endmodule

module top_module (
  input [7:0] in,
  output parity
);

  parity_checker checker(
    .in(in),
    .parity(parity)
  );

endmodule
