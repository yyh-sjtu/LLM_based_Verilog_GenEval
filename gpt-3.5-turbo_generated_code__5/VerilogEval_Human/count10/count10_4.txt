module decade_counter(
	input clk,
	input reset,
	output [3:0] q);

	reg [3:0] count;

	always @(posedge clk) begin
		if (reset)
			count <= 0;
		else if (count == 9)
			count <= 0;
		else
			count <= count + 1;
	end

	assign q = count;

endmodule

module top_module(
	input clk,
	input reset,
	output reg [3:0] q);

	decade_counter counter(
		.clk(clk),
		.reset(reset),
		.q(q)
	);

endmodule
