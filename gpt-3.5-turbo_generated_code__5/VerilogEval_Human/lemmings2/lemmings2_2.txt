// Define states for the finite state machine
parameter IDLE = 2'b00;
parameter WALK_LEFT = 2'b01;
parameter WALK_RIGHT = 2'b10;

// Define outputs for the finite state machine
output reg walk_left;
output reg walk_right;
output reg aaah;

// Define internal signals for state transitions
reg [1:0] state;
reg [1:0] next_state;

always @(posedge clk or posedge areset) begin
	if (areset) begin
		state <= IDLE;
	end else begin
		state <= next_state;
	end
end

always @(state or bump_left or bump_right or ground) begin
	case (state)
		IDLE: begin
			if (areset) begin
				next_state <= IDLE;
			end else if (ground == 0) begin
				aaah <= 1;
				next_state <= IDLE;
			end else if (bump_left == 1 && bump_right == 0) begin
				walk_left <= 1;
				walk_right <= 0;
				next_state <= WALK_LEFT;
			end else if (bump_left == 0 && bump_right == 1) begin
				walk_left <= 0;
				walk_right <= 1;
				next_state <= WALK_RIGHT;
			end else if (bump_left == 1 && bump_right == 1) begin
				walk_left <= 0;
				walk_right <= 1;
				next_state <= WALK_RIGHT;
			end
		end
		WALK_LEFT: begin
			if (ground == 0) begin
				aaah <= 1;
				next_state <= IDLE;
			end else if (bump_left == 1 && bump_right == 0) begin
				walk_left <= 1;
				walk_right <= 0;
				next_state <= WALK_LEFT;
			end else if (bump_left == 0 && bump_right == 1) begin
				walk_left <= 0;
				walk_right <= 1;
				next_state <= WALK_RIGHT;
			end else if (bump_left == 1 && bump_right == 1) begin
				walk_left <= 0;
				walk_right <= 1;
				next_state <= WALK_RIGHT;
			end
		end
		WALK_RIGHT: begin
			if (ground == 0) begin
				aaah <= 1;
				next_state <= IDLE;
			end else if (bump_left == 1 && bump_right == 0) begin
				walk_left <= 1;
				walk_right <= 0;
				next_state <= WALK_LEFT;
			end else if (bump_left == 0 && bump_right == 1) begin
				walk_left <= 0;
				walk_right <= 1;
				next_state <= WALK_RIGHT;
			end else if (bump_left == 1 && bump_right == 1) begin
				walk_left <= 0;
				walk_right <= 1;
				next_state <= WALK_RIGHT;
			end
		end
	endcase
end
endmodule
