module top_module(
    input mode,
    input too_cold,
    input too_hot,
    input fan_on,
    output heater,
    output aircon,
    output fan
);

// Define internal signals
reg fan_on_internal;

// Assign internal signal based on input fan_on
always @(fan_on)
begin
    fan_on_internal = fan_on;
end

// Turn on heater in heating mode when too_cold is 1, otherwise turn off
assign heater = (mode == 1 && too_cold) ? 1'b1 : 1'b0;

// Turn on air conditioner in cooling mode when too_hot is 1, otherwise turn off
assign aircon = (mode == 0 && too_hot) ? 1'b1 : 1'b0;

// Turn on fan when heater, air conditioner, or fan_on_internal is 1, otherwise turn off
assign fan = (heater || aircon || fan_on_internal) ? 1'b1 : 1'b0;

endmodule
