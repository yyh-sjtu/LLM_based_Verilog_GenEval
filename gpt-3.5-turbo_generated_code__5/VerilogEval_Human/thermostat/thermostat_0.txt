```
module top_module(
	input mode,
	input too_cold, 
	input too_hot,
	input fan_on,
	output heater,
	output aircon,
	output fan
);

// Define internal signals
reg fan_control;

// Logic for fan control
always @(*)
begin
    if (mode == 1 && too_cold) // Heating mode
        fan_control = 1;
    else if (mode == 0 && too_hot) // Cooling mode
        fan_control = 1;
    else if (fan_on) // User requested fan
        fan_control = 1;
    else
        fan_control = 0;
end

// Output logic for heater, aircon, and fan
assign heater = (mode == 1 && too_cold) ? 1 : 0;
assign aircon = (mode == 0 && too_hot) ? 1 : 0;
assign fan = fan_control;

endmodule
```
