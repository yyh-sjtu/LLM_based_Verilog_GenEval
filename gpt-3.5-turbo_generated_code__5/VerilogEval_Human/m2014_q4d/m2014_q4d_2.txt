module d_ff (
  input logic clk,
  input logic d,
  output logic q
);

  always_ff @(posedge clk) begin
    q <= d;
  end
endmodule

module xor (
  input logic a,
  input logic b,
  output logic y
);

  assign y = a ^ b;
endmodule
