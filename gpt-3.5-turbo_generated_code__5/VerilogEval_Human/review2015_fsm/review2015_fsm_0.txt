
reg [3:0] state;

 always @(posedge clk or posedge reset) begin
    if (reset) begin
        state <= 4'b0000; // initial state
        shift_ena <= 0;
        counting <= 0;
        done <= 0;
    end else begin
        case (state)
            4'b0000: begin // Idle state
                if (data == 4'b1101) begin
                    state <= 4'b0001; // Transition to shift_ena state
                    shift_ena <= 1;
                end else begin
                    shift_ena <= 0;
                end
            end
            4'b0001: begin // Shift_ena state
                shift_ena <= 1;
                state <= 4'b0010; // Transition to counting state
            end
            4'b0010: begin // Counting state
                counting <= 1;
                if (done_counting) begin
                    state <= 4'b0011; // Transition to done state
                end
            end
            4'b0011: begin // Done state
                done <= 1;
                if (ack) begin
                    state <= 4'b0000; // Transition back to idle state
                end
            end
            default: begin
                state <= 4'b0000; // Default idle state
            end
        endcase
    end
end

endmodule
