    case ({y, x})

            Y0 <= 1'b0;

        3'b0001, 3'b0100: begin
            Y0 <= 1'b1;

        end

            Y0 <= 1'b0;
            z <= 1'b0;
        end

            Y0 <= 1'b0;

    endcase
end
