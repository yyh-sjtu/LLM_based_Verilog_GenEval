reg [2:0] next_state;

always @(posedge clk) begin
	case({y, x})
		3'b0000: next_state = 3'b000;
		3'b0001: next_state = 3'b001;
		3'b0010: next_state = 3'b010;
		3'b0011: next_state = 3'b001;
		3'b0100: next_state = 3'b011;
	endcase
	
	Y0 <= next_state[0];
	z <= (next_state == 3'b011) || (next_state == 3'b100);
end
