// Define states for the state machine
parameter IDLE = 2'b00;
parameter ABOVE_S3 = 2'b01;
parameter BETWEEN_S3_S2 = 2'b10;
parameter BETWEEN_S2_S1 = 2'b11;

// Define nominal flow rates
parameter FR1 = 1'b0;
parameter FR2 = 1'b1;
parameter FR3 = 1'b1;

// Define initial state and outputs
reg [1:0] state, next_state;
reg [1:0] prev_state;
reg [2:0] flow_rate;

always @(posedge clk) begin
	if(reset) begin
		state <= IDLE;
		flow_rate <= 3'b111;
		dfr <= 0;
	end
	else begin
		state <= next_state;
		flow_rate <= flow_rate;
		dfr <= 0;
	end
end

always @* begin
	prev_state = state;

	case(state)
		IDLE: begin
			if(s[1] == 1'b0) begin
				next_state = ABOVE_S3;
				flow_rate = 3'b000;
			end
			else begin
				next_state = state;
				dfr = 1;
			end
		end
		ABOVE_S3: begin
			if(s[3] == 1'b0) next_state = BETWEEN_S3_S2;
			else next_state = ABOVE_S3;
		end
		BETWEEN_S3_S2: begin
			if(s[2] == 1'b0) next_state = BETWEEN_S2_S1;
			else next_state = BETWEEN_S3_S2;
			flow_rate = 3'b001;
			if(prev_state < state) dfr = 1;
		end
		BETWEEN_S2_S1: begin
			if(s[1] == 1'b0) next_state = IDLE;
			else next_state = BETWEEN_S2_S1;
			flow_rate = 3'b011;
			if(prev_state < state) dfr = 1;
		end
	endcase
end

// Assign flow rate outputs based on state and flow_rate
assign fr3 = flow_rate[0];
assign fr2 = flow_rate[1];
assign fr1 = flow_rate[2];

endmodule
