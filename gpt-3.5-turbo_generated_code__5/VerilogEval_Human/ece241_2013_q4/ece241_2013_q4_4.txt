module top_module (
    input clk,
    input reset,
    input [3:1] s,
    output reg fr3,
    output reg fr2,
    output reg fr1,
    output reg dfr
);

// Define states for the state machine
parameter IDLE = 2'b00;
parameter ABOVE_S3 = 2'b01;
parameter BETWEEN_S3_S2 = 2'b10;
parameter BETWEEN_S2_S1 = 2'b11;

// Define constants for nominal flow rates
parameter FR1 = 1'b1;
parameter FR2 = 1'b1;
parameter FR3 = 1'b1;

// Initialize state and output registers
reg [1:0] state_reg, state_next;
reg reset_reg, reset_next;

always @ (posedge clk or posedge reset) begin
    if (reset) begin
        // Synchronous reset
        state_reg <= IDLE;
        fr3 <= 1'b1;
        fr2 <= 1'b1;
        fr1 <= 1'b1;
        dfr <= 1'b1;
        reset_reg <= 1'b1;
    end else begin
        reset_reg <= reset_next;
        state_reg <= state_next;
        fr3 <= fr3_next;
        fr2 <= fr2_next;
        fr1 <= fr1_next;
        dfr <= dfr_next;
    end
end

always @ (*) begin
    // Default next state and output values
    state_next = state_reg;
    fr3_next = 1'b1;
    fr2_next = 1'b1;
    fr1_next = 1'b1;
    dfr_next = 1'b1;
    
    case (state_reg)
        IDLE: begin
            if (s[1]) begin
                state_next = BETWEEN_S2_S1;
                fr1_next = FR1;
                fr2_next = FR2;
            end else if (s[2]) begin
                state_next = BETWEEN_S3_S2;
                fr1_next = FR1;
            end else if (s[3]) begin
                state_next = ABOVE_S3;
            end else begin
                state_next = IDLE;
                fr1_next = FR1;
                fr2_next = FR2;
                fr3_next = FR3;
                dfr_next = 1'b0;
            end
        end
        
        ABOVE_S3: begin
            if (!s[3]) begin
                state_next = IDLE;
                fr1_next = FR1;
                fr2_next = FR2;
                fr3_next = FR3;
                dfr_next = 1'b1;
            end
        end
        
        BETWEEN_S3_S2: begin
            if (!s[2]) begin
                state_next = IDLE;
                fr1_next = FR1;
                fr2_next = FR2;
                fr3_next = FR3;
                dfr_next = 1'b1;
            end
        end
        
        BETWEEN_S2_S1: begin
            if (!s[1]) begin
                state_next = IDLE;
                fr1_next = FR1;
                fr2_next = FR2;
                fr3_next = FR3;
                dfr_next = 1'b1;
            end
        end
    end
    
    // Reset logic
    if (reset_reg && !reset) begin
        reset_next = 1'b0;
    end else begin
        reset_next = reset;
    end
end

endmodule
