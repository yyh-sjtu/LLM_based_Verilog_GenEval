module top_module (
    input clk,
    input reset,
    input [3:1] s,
    output reg fr3,
    output reg fr2,
    output reg fr1,
    output reg dfr
);

reg [2:0] state;

always @(posedge clk) begin
    if (reset) begin
        state <= 0;
        fr3 <= 1;
        fr2 <= 1;
        fr1 <= 1;
        dfr <= 0;
    end else begin
        case(state)
            0: begin
                fr3 <= 1;
                fr2 <= 1;
                fr1 <= 1;
                dfr <= 0;
                if (s == 3'b001) // Below s[1]
                    state <= 1;
            end
            1: begin
                fr3 <= 0;
                fr2 <= 1;
                fr1 <= 1;
                dfr <= 0;
                if (s == 3'b011) // Between s[2] and s[1]
                    state <= 2;
                else if (s == 3'b000) // Above s[3]
                    state <= 0;
            end
            2: begin
                fr3 <= 0;
                fr2 <= 1;
                fr1 <= 0;
                dfr <= 1;
                if (s == 3'b010) // Between s[3] and s[2]
                    state <= 1;
                else if (s == 3'b000) // Above s[3]
                    state <= 0;
            end
        endcase
    end
end

endmodule
