// Define states for the state machine
parameter IDLE = 2'b00;
parameter ABOVE_S3 = 2'b01;
parameter BETWEEN_S3_S2 = 2'b10;
parameter BETWEEN_S2_S1 = 2'b11;
parameter BELOW_S1 = 2'b100;

reg [1:0] state;
reg prev_level;

always @(posedge clk or posedge reset) begin
	if (reset) begin
		state <= IDLE;
		prev_level <= 0;
		fr3 <= 1'b1;
		fr2 <= 1'b1;
		fr1 <= 1'b1;
		dfr <= 1'b1;
	end else begin
		case(state)
			IDLE: begin
				if(s[1] & !(s[2] | s[3]) begin
					state <= BELOW_S1;
					prev_level <= 3;
				end else if(s[1] & s[2] & !s[3]) begin
					state <= BETWEEN_S2_S1;
					prev_level <= 2;
				end else if(s[1] & s[2] & s[3]) begin
					state <= BETWEEN_S3_S2;
					prev_level <= 1;
				end else if(!(s[1] | s[2] | s[3]) begin
					state <= ABOVE_S3;
					prev_level <= 0;
				end
			end
			ABOVE_S3: begin
				if(s[1] & s[2] & !s[3]) begin
					state <= BETWEEN_S3_S2;
					prev_level <= 3;
				end
			end
			BETWEEN_S3_S2: begin
				if(s[1] & !s[2]) begin
					state <= BETWEEN_S2_S1;
					prev_level <= 2;
				end
			end
			BETWEEN_S2_S1: begin
				if(!s[1]) begin
					state <= BELOW_S1;
					prev_level <= 1;
				end
			end
			BELOW_S1: begin
				// No state transitions from this state
			end
		endcase
	end
end

// Output assignments based on current state and previous level
always @(state or prev_level) begin
	case(state)
		IDLE: begin
			fr3 <= 1'b1;
			fr2 <= 1'b1;
			fr1 <= 1'b1;
			dfr <= 1'b1;
		end
		ABOVE_S3: begin
			fr3 <= 1'b0;
			fr2 <= 1'b1;
			fr1 <= 1'b1;
			dfr <= 1'b0;
		end
		BETWEEN_S3_S2: begin
			fr3 <= 1'b0;
			fr2 <= 1'b0;
			fr1 <= 1'b1;
			if(prev_level < 2) begin
				dfr <= 1'b1;
			end
		end
		BETWEEN_S2_S1: begin
			fr3 <= 1'b0;
			fr2 <= 1'b0;
			fr1 <= 1'b0;
			if(prev_level < 1) begin
				dfr <= 1'b1;
			end
		end
		BELOW_S1: begin
			fr3 <= 1'b0;
			fr2 <= 1'b0;
			fr1 <= 1'b0;
			dfr <= 1'b1;
		end
	endcase
