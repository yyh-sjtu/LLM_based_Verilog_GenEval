// Define the JK flip-flop module
module JK_flip_flop (
    input j,
    input k,
    input clk,
    output reg Q
);

    always @(posedge clk) begin
        if (j==1 && k==1)
            Q <= ~Q;
        else if (j==1)
            Q <= 1;
        else if (k==1)
            Q <= 0;
    end
    
endmodule

// Instantiate the JK flip-flop module in the top module
module top_module (
    input clk,
    input j,
    input k,
    output reg Q
);

    JK_flip_flop FlipFlop (
        .j(j),
        .k(k),
        .clk(clk),
        .Q(Q)
    );
    
endmodule
