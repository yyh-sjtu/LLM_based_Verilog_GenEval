module top_module (
    input clk,
    input reset,
    input s,
    input w,
    output reg z
);

reg [2:0] state;

always @(posedge clk) begin
    if (reset) begin
        state <= 3'b000; // Reset state A
        z <= 0;
    end else begin
        case (state)
            3'b000: begin // State A
                if (s) begin
                    state <= 3'b001; // Move to state B
                end
            end
            
            3'b001: begin // State B
                if (w) begin
                    state <= 3'b010; // Transition to State C
                end else begin
                    state <= 3'b000; // Return to State A
                end
            end
            
            3'b010: begin // State C
                if (w) begin
                    state <= 3'b011; // Transition to State D
                end else begin
                    state <= 3'b000; // Return to State A
                end
            end
            
            3'b011: begin // State D
                if (w) begin
                    z <= 1; // Set output z to 1
                end else begin
                    z <= 0; // Set output z to 0
                end
                state <= 3'b100; // Transition to State E
            end
            
            3'b100: begin // State E
                if (w) begin
                    state <= 3'b101; // Transition to State F
                end else begin
                    state <= 3'b000; // Return to State A
                end
            end
            
            3'b101: begin // State F
                if (w) begin
                    state <= 3'b110; // Transition to State G
                end else begin
                    z <= 0; // Set output z to 0
                    state <= 3'b000; // Return to State A
                end
            end
            
            3'b110: begin // State G
                if (w) begin
                    z <= 0; // Set output z to 0
                end else begin
                    z <= 1; // Set output z to 1
                end
                state <= 3'b000; // Return to State A
            end
        endcase
    end
end

endmodule
