module top_module (
	input clk,
	input reset,
	input x,
	output reg z
);
	// Define states
	parameter S0 = 3'b000; // Present state 000
	parameter S1 = 3'b001; // Present state 001
	parameter S2 = 3'b010; // Present state 010
	parameter S3 = 3'b011; // Present state 011
	parameter S4 = 3'b100; // Present state 100
	
	reg [2:0] present_state; // Present state register
	reg [2:0] next_state; // Next state register
	
	always @(posedge clk or posedge reset) begin
		if (reset) begin
			present_state <= S0; // Reset to state 000
		end else begin
			present_state <= next_state; // Update present state with next state
		end
	end
	
	always @(present_state or x) begin
		case(present_state)
			S0: begin
				if (x) begin
					next_state <= S1; // Next state 001 if x=1
				end else begin
					next_state <= S0; // Next state 000 if x=0
				end
				z <= 0; // Output is 0
			end
			
			S1: begin
				if (x) begin
					next_state <= S4; // Next state 100 if x=1
				end else begin
					next_state <= S1; // Next state 001 if x=0
				end
				z <= 0; // Output is 0
			end
			
			S2: begin
				if (x) begin
					next_state <= S1; // Next state 001 if x=1
				end else begin
					next_state <= S2; // Next state 010 if x=0
				end
				z <= 0; // Output is 0
			end
			
			S3: begin
				if (x) begin
					next_state <= S2; // Next state 010 if x=1
				end else begin
					next_state <= S1; // Next state 001 if x=0
				end
				z <= 1; // Output is 1
			end
			
			S4: begin
				if (x) begin
					next_state <= S3; // Next state 011 if x=1
				end else begin
					next_state <= S4; // Next state 100 if x=0
				end
				z <= 1; // Output is 1
			end
		endcase
	end
endmodule
