// Define states
parameter S0 = 3'b000;
parameter S1 = 3'b001;
parameter S2 = 3'b010;
parameter S3 = 3'b011;
parameter S4 = 3'b100;

// Define next state logic
always @ (posedge clk or posedge reset)
begin
	if (reset) 
		y <= S0;
	else begin
		case(y)
			S0: y <= (x) ? S1 : S1;
			S1: y <= (x) ? S4 : S2;
			S2: y <= (x) ? S1 : S1;
			S3: y <= (x) ? S2 : S2;
			S4: y <= (x) ? S3 : S4;
		endcase
	end
end

// Define output logic
always @ (y)
begin
	case(y)
		S0: z = 0;
		S1: z = 0;
		S2: z = 0;
		S3: z = 1;
		S4: z = 1;
	endcase
end

// Define state register
reg [2:0] y;

endmodule
