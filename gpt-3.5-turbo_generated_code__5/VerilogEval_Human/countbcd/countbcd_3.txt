```verilog
module top_module(
    input clk,
    input reset,
    output [3:1] ena,
    output reg [15:0] q);

    reg [3:0] ones_digit, tens_digit, hundreds_digit, thousands_digit;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            ones_digit <= 4'b0000;
            tens_digit <= 4'b0000;
            hundreds_digit <= 4'b0000;
            thousands_digit <= 4'b0000;
        end else begin
            if (ones_digit == 4'b1001) begin
                ones_digit <= 4'b0000;
                if (tens_digit == 4'b1001) begin
                    tens_digit <= 4'b0000;
                    if (hundreds_digit == 4'b1001) begin
                        hundreds_digit <= 4'b0000;
                        if (thousands_digit == 4'b1001) begin
                            thousands_digit <= 4'b0000;
                        end else begin
                            thousands_digit <= thousands_digit + 1;
                        end
                    end else begin
                        hundreds_digit <= hundreds_digit + 1;
                    end
                end else begin
                    tens_digit <= tens_digit + 1;
                end
            end else begin
                ones_digit <= ones_digit + 1;
            end
        end
    end

    assign q = {thousands_digit, hundreds_digit, tens_digit, ones_digit};
    assign ena = {thousands_digit != 4'b1001, hundreds_digit != 4'b1001, tens_digit != 4'b1001};

endmodule
```
