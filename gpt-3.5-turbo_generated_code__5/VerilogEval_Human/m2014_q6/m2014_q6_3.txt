// Define the states using 3-bit binary codes
parameter [2:0] A = 3'b000;
parameter [2:0] B = 3'b001;
parameter [2:0] C = 3'b010;
parameter [2:0] D = 3'b011;
parameter [2:0] E = 3'b100;
parameter [2:0] F = 3'b101;

// Define the initial state
reg [2:0] current_state = A;

// Define the next state logic
always @(posedge clk or posedge reset) begin
	if (reset) begin
		current_state <= A;
	end else begin
		case (current_state)
			A: begin
				if (w) begin
					current_state <= A;
				else begin
					current_state <= B;
				end
			end
			B: begin
				if (w) begin
					current_state <= D;
				else begin
					current_state <= C;
				end
			end
			C: begin
				if (w) begin
					current_state <= D;
				else begin
					current_state <= E;
				end
			end
			D: begin
				if (w) begin
					current_state <= A;
				else begin
					current_state <= F;
				end
			end
			E: begin
				if (w) begin
					current_state <= D;
				else begin
					current_state <= E;
				end
			end
			F: begin
				if (w) begin
					current_state <= D;
				else begin
					current_state <= C;
				end
			end
		endcase
	end
end

// Define the output logic based on the current state
assign z = (current_state == C);

endmodule
