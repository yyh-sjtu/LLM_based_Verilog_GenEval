reg [2:0] state;
always @(posedge clk or posedge reset) begin
	if (reset) begin
		state <= 0;
	end else begin
		case(state)
			0: begin
				if (w) state <= 1;
				else state <= 2;
			end
			1: begin
				if (w) state <= 0;
				else state <= 3;
			end
			2: begin
				if (w) state <= 4;
				else state <= 3;
			end
			3: begin
				if (w) state <= 0;
				else state <= 5;
			end
			4: begin
				if (w) state <= 4;
				else state <= 3;
			end
			5: begin
				if (w) state <= 2;
				else state <= 3;
			end
			default: state <= 0;
		endcase
	end
end

assign z = (state == 6) ? 1'b1 : 1'b0;
endmodule
