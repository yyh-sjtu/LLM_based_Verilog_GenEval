reg [2:0] state;

always @(posedge clk or posedge reset) begin
	if (reset)
		state <= 3'd0;
	else if (w) begin
		case (state)
			3'd0: state <= 3'd1;
			3'd1: state <= 3'd0;
			3'd2: state <= 3'd3;
			3'd3: state <= 3'd4;
			3'd4: state <= 3'd6;
			3'd5: state <= 3'd4;
			3'd6: state <= 3'd2;
			default: state <= 3'd0;
		endcase
	end else begin
		case (state)
			3'd0: state <= 3'd2;
			3'd1: state <= 3'd0;
			3'd2: state <= 3'd5;
			3'd3: state <= 3'd3;
			3'd4: state <= 3'd5;
			3'd5: state <= 3'd3;
			3'd6: state <= 3'd7;
			default: state <= 3'd0;
		endcase
	end
end

assign z = (state == 3'd7);

endmodule
