reg [2:0] state;
parameter [2:0] A = 3'b000;
parameter [2:0] B = 3'b001;
parameter [2:0] C = 3'b010;
parameter [2:0] D = 3'b011;
parameter [2:0] E = 3'b100;
parameter [2:0] F = 3'b101;

always @(posedge clk or posedge reset) begin
	if (reset)
		state <= A;
	else begin
		case(state)
			A: if (w) state <= A; else state <= B;
			B: if (w) state <= D; else state <= C;
			C: if (w) state <= D; else state <= E;
			D: if (w) state <= A; else state <= F;
			E: if (w) state <= D; else state <= E;
			F: if (w) state <= D; else state <= C;
		endcase
	end
end

assign z = (state == E || state == F) ? 1'b1 : 1'b0;

endmodule
