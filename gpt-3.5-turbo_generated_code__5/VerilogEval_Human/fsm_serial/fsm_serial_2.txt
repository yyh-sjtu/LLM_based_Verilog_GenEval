
	parameter IDLE = 2'b00;
	parameter START = 2'b01;
	parameter DATA = 2'b10;
	parameter STOP = 2'b11;
	
	reg [1:0] state, next_state;
	reg [2:0] count;
	reg start_bit, stop_bit;
	reg [7:0] data_byte;
	
	always @(posedge clk) begin
		if (reset) begin
			state <= IDLE;
			next_state <= IDLE;
			count <= 0;
			start_bit <= 0;
			stop_bit <= 1;
			done <= 0;
		end else begin
			state <= next_state;
			case (state)
				IDLE: begin
					if (in == 0) begin
						next_state <= START;
					end
				end
				START: begin
					if (count < 8) begin
						data_byte[count] <= in;
						count <= count + 1;
						next_state <= DATA;
					end else begin
						next_state <= STOP;
					end
				end
				DATA: begin
					if (count < 8) begin
						data_byte[count] <= in;
						count <= count + 1;
						next_state <= DATA;
					end else begin
						next_state <= STOP;
					end
				end
				STOP: begin
					if (in == 1) begin
						stop_bit <= 1;
						done <= 1;
						next_state <= IDLE;
					end else begin
						stop_bit <= 0;
						next_state <= STOP;
					end
				end
			endcase
		end
	end
	
endmodule
