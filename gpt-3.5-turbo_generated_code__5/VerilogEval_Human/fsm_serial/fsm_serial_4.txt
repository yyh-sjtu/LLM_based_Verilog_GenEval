module top_module (
    input clk,
    input in,
    input reset,
    output done
);


reg [3:0] state;
reg start_bit_detected;
reg [7:0] data_bits;
reg stop_bit_detected;
reg [2:0] count;

always @(posedge clk or posedge reset) begin
    if (reset) begin
        state <= 0;
        start_bit_detected <= 0;
        data_bits <= 0;
        stop_bit_detected <= 0;
        count <= 0;
    end else begin
        case(state)
            0: begin // Idle state
                if (in == 0) begin // Start bit detected
                    state <= 1;
                    start_bit_detected <= 1;
                    data_bits <= 0;
                    stop_bit_detected <= 0;
                    count <= 0;
                end
            end
            1: begin // Receiving data bits
                if (count < 8) begin // Still receiving data bits
                    data_bits <= {data_bits[6:0], in};
                    count <= count + 1;
                end else begin // All data bits received
                    state <= 2;
                end
            end
            2: begin // Verifying stop bit
                if (in == 1) begin // Stop bit detected
                    stop_bit_detected <= 1;
                    state <= 3;
                end else begin // Stop bit not detected yet
                    state <= 2;
                end
            end
            3: begin // Byte received successfully
                state <= 0;
                start_bit_detected <= 0;
                stop_bit_detected <= 0;
                count <= 0;
            end
        endcase
    end
end

assign done = (state == 3);

endmodule
