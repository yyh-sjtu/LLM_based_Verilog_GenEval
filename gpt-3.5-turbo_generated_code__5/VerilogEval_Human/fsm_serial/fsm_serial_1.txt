module top_module (
	input clk,
	input in,
	input reset,
	output done
);

	// Define states
	typedef enum logic [2:0] {
		IDLE,
		START_BIT,
		DATA_BITS,
		STOP_BIT
	} state_t;

	// Define outputs
	logic done;
   
	// Define internal signals
	logic start;
	logic [7:0] data_bits;
	logic stop;

	// Define current state
	state_t current_state;

	// Define next state
	state_t next_state;

	// Finite State Machine (FSM)
	always_ff @(posedge clk, posedge reset)
	begin
		if (reset)
			current_state <= IDLE;
		else
			current_state <= next_state;
	end

	always_comb
	begin
		done = (current_state == STOP_BIT);

		// Default next state is current state
		next_state = current_state;

		// Update next state based on current state and input
		case (current_state)
			IDLE:
				if (in == 1'b0)
					next_state = START_BIT;
			START_BIT:
				if (in == 1'b0)
				begin
					start = 1'b1;
					next_state = DATA_BITS;
				end
				else
				begin
					start = 1'b0;
					next_state = IDLE;
				end
			DATA_BITS:
				begin
					if (in == 1'b0)
						data_bits = {data_bits[6:0], in};
					
					if (data_bits[7] == 1'b1)
						next_state = STOP_BIT;
				end
			STOP_BIT:
				if (in == 1'b1)
				begin
					stop = 1'b1;
					next_state = IDLE;
				end
				else
				begin
					stop = 1'b0;
					next_state = STOP_BIT;
				end
			default:
				next_state = IDLE;
		endcase
	end
endmodule
