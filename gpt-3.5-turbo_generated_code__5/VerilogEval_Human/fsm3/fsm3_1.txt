reg [1:0] state, next_state;
reg out;

always @(posedge clk or posedge areset) begin
	if (areset)
		state <= 2'b00; // reset to state A
	else
		state <= next_state;
end

always @* begin
	case (state)
		2'b00: begin
			if (in)
				next_state = 2'b10; // next state is B
			else
				next_state = 2'b01; // next state is A
			out = 1'b0;
		end
		2'b01: begin
			if (in)
				next_state = 2'b01; // next state is A
			else
				next_state = 2'b10; // next state is B
			out = 1'b0;
		end
		2'b10: begin
			if (in)
				next_state = 2'b11; // next state is D
			else
				next_state = 2'b01; // next state is A
			out = 1'b0;
		end
		2'b11: begin
			if (in)
				next_state = 2'b10; // next state is B
			else
				next_state = 2'b11; // next state is D
			out = 1'b1;
		end
	endcase
end

assign out = out;

endmodule
