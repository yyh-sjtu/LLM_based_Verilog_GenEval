// Define states
parameter A = 2'b00;
parameter B = 2'b01;
parameter C = 2'b10;
parameter D = 2'b11;

// Define outputs
reg [1:0] state, next_state;
reg out;

always @ (posedge clk or areset) begin
	if (areset) begin
		state <= A;
	end
	else begin
		state <= next_state;
	end
end

always @ (*) begin
	case (state)
		A: begin
			if (in) begin
				next_state = B;
			end
			else begin
				next_state = A;
			end
			out = 1'b0;
		end
		B: begin
			if (in) begin
				next_state = B;
			end
			else begin
				next_state = C;
			end
			out = 1'b0;
		end
		C: begin
			if (in) begin
				next_state = D;
			end
			else begin
				next_state = A;
			end
			out = 1'b0;
		end
		D: begin
			if (in) begin
				next_state = B;
			end
			else begin
				next_state = C;
			end
			out = 1'b1;
		end
	endcase
end

endmodule
