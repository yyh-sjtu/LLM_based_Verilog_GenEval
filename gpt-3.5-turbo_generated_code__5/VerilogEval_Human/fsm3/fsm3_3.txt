```verilog
module moore_fsm (
	input clk,
	input reset,
	input in,
	output reg out,
	output reg [1:0] state
);

parameter A = 2'b00;
parameter B = 2'b01;
parameter C = 2'b10;
parameter D = 2'b11;

always @(posedge clk or posedge reset) begin
	if (reset) begin
		state <= A;
		out <= 1'b0;
	end else begin
		case(state)
			A: begin
				if (in) begin
					state <= B;
				end else begin
					state <= A;
				end
				out <= 1'b0;
			end
			B: begin
				if (in) begin
					state <= B;
				end else begin
					state <= C;
				end
				out <= 1'b0;
			end
			C: begin
				if (in) begin
					state <= D;
				end else begin
					state <= A;
				end
				out <= 1'b0;
			end
			D: begin
				if (in) begin
					state <= B;
				end else begin
					state <= C;
				end
				out <= 1'b1;
			end
		endcase
	end
end

endmodule
```
