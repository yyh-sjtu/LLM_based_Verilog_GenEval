// Define states
parameter A = 2'b00;
parameter B = 2'b01;
parameter C = 2'b10;
parameter D = 2'b11;

// Define state register and next state logic
reg [1:0] state, next_state;
always @(posedge clk or areset) begin
	if (areset) begin
		state <= A;
	end
	else begin
		state <= next_state;
	end
end

// Define output logic
assign out = (state == D) ? 1'b1 : 1'b0;

// Define next state and output logic
always @* begin
	case (state)
		A: begin
			next_state = in ? B : A;
		end
		B: begin
			next_state = in ? B : C;
		end
		C: begin
			next_state = in ? D : A;
		end
		D: begin
			next_state = in ? B : C;
		end
	endcase
end

endmodule
