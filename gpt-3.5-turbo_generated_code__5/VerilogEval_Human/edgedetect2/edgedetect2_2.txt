module top_module(
	input clk,
	input [7:0] in,
	output reg [7:0] anyedge);

// Register to store the previous value of the input vector
reg [7:0] prev_in;

always @(posedge clk) begin
	for (int i = 0; i < 8; i = i + 1) begin
		if (in[i] & ~prev_in[i]) begin
			anyedge[i] <= 1;
		end else begin
			anyedge[i] <= 0;
		end
	end
	prev_in <= in;
end

endmodule
