module top_module(
	input clk,
	input [7:0] in,
	output reg [7:0] pedge);

	reg [7:0] prev_in; // previous input value

	always @(posedge clk) begin
		for (integer i = 0; i < 8; i = i + 1) begin
			if (in[i] == 1 && prev_in[i] == 0) begin
				pedge[i] <= 1; // set output bit on positive transition
			end else begin
				pedge[i] <= 0; // reset output bit if no transition
			end
		end
		
		prev_in <= in; // store current input for next cycle
	end
endmodule
