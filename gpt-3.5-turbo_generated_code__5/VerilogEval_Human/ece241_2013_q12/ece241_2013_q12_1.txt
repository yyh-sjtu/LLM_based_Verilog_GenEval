reg [7:0] Q;
reg [2:0] select;

// Shift Register
always @(posedge clk) begin
	if (enable) begin
		Q[0] <= S;
		Q[1] <= Q[0];
		Q[2] <= Q[1];
		Q[3] <= Q[2];
		Q[4] <= Q[3];
		Q[5] <= Q[4];
		Q[6] <= Q[5];
		Q[7] <= Q[6];
	end
end

// Multiplexers
always @* begin
	case ({A, B, C})
		3'b000 : select = 3'b000;
		3'b001 : select = 3'b001;
		3'b010 : select = 3'b010;
		3'b011 : select = 3'b011;
		3'b100 : select = 3'b100;
		3'b101 : select = 3'b101;
		3'b110 : select = 3'b110;
		3'b111 : select = 3'b111;
	endcase
end

always @(posedge clk) begin
	Z <= Q[select];
end
