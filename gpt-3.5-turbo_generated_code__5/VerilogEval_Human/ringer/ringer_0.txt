module top_module(
	input ring, 
	input vibrate_mode,
	output ringer,
	output motor
);

	// Define internal signal
	wire ringer_enable;
	
	// Turn on the ringer if the phone is not in vibrate mode
	assign ringer_enable = ~vibrate_mode;
	
	// Assign the values to the outputs
	assign ringer = ring & ringer_enable;
	assign motor = ring & ~ringer_enable;

endmodule
