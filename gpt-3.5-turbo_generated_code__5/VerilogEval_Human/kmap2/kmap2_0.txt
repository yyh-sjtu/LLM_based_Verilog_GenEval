// Declare internal wire variables
wire w1, w2, w3, w4;

// Implement the logic gates based on the Karnaugh map
and #(2) and1 (w1, a, b); // AND gate for (a && b)
and #(3) and2 (w2, c, a, d); // AND gate for (c && a && d)
and #(3) and3 (w3, c, b, d); // AND gate for (c && b && d)
or #(4) or1 (w4, w1, w2, w3); // OR gate for (w1 || w2 || w3)

// Assign the output
assign out = w4;

endmodule
