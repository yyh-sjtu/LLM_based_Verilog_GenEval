reg [1:0] state, next_state;
reg out;

// Define the states
parameter A = 2'b00;
parameter B = 2'b01;
parameter C = 2'b10;
parameter D = 2'b11;

// Determine the next state and output based on current state and input
always @ (posedge clk) begin
	if (reset) begin
		state <= A;
		out <= 1'b0;
	end
	else begin
		state <= next_state;
		out <= (state == D) ? 1'b1 : 1'b0;
	end
end

// Determine the next state based on current state and input
always @ (*) begin
	case (state)
		A: begin
			if (in) next_state = B;
			else next_state = A;
		end
		B: begin
			if (in) next_state = B;
			else next_state = C;
		end
		C: begin
			if (in) next_state = D;
			else next_state = A;
		end
		D: begin
			if (in) next_state = B;
			else next_state = C;
		end
	endcase
end

endmodule
