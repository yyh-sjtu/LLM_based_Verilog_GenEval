module shift_register(
    input clk,
    input load,
    input ena,
    input [1:0] amount,
    input [63:0] data,
    output reg [63:0] q);

    reg [63:0] q_temp; // temporary register for shifting

    always @(posedge clk) begin
        if (load) begin
            q <= data; // load data when load signal is active
        end else if (ena) begin
            case(amount)
                2'b00: q_temp = {q[62:0], q[63]}; // shift left by 1 bit
                2'b01: q_temp = {q[55:0], q[63:56]}; // shift left by 8 bits
                2'b10: q_temp = {q[0], q[63:1]}; // shift right by 1 bit (arithmetic right shift)
                2'b11: q_temp = {q[7], q[63:8]}; // shift right by 8 bits (arithmetic right shift)
            endcase
            q <= q_temp; // update contents of the shifter
        end
    end
endmodule
