module shifter(
	input clk,
 	input load,
 	input ena,
 	input [1:0] amount,
 	input [63:0] data,
 	output reg [63:0] q
);
	reg [63:0] temp;
  	
  	always @(posedge clk) begin
		if(load) begin
			temp <= data;
		end
		else if(ena) begin
			case(amount)
				2'b00: temp <= temp << 1; // shift left by 1 bit
				2'b01: temp <= temp << 8; // shift left by 8 bits
				2'b10: temp <= $signed(temp) >> 1; // arithmetic shift right by 1 bit
				2'b11: temp <= $signed(temp) >> 8; // arithmetic shift right by 8 bits
				default: temp <= temp; // no shift
			endcase
		end
		else begin
			temp <= temp; // no shift
		end
  	end

  	assign q = temp;
endmodule

module top_module(
	input clk,
	input load,
	input ena,
	input [1:0] amount,
	input [63:0] data,
	output reg [63:0] q
);
  
	shifter shifter_inst (
		.clk(clk),
		.load(load),
		.ena(ena),
		.amount(amount),
		.data(data),
		.q(q)
	);

endmodule
