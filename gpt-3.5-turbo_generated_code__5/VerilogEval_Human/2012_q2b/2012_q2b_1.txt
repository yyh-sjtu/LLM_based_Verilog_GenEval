reg Y1;
reg Y3;

always @(y) begin
    case(y)
        6'b000001: begin
            if (w)
                Y1 = 1'b0;
            else
                Y1 = 1'b0;
                
            Y3 = 1'b0;
        end
        
        6'b000010: begin
            if (w)
                Y1 = 1'b1;
            else
                Y1 = 1'b0;
                
            Y3 = 1'b0;
        end
        
        6'b000100: begin
            if (w)
                Y1 = 1'b0;
            else
                Y1 = 1'b1;
            
            Y3 = 1'b0;
        end
        
        6'b001000: begin
            if (w)
                Y1 = 1'b1;
            else
                Y1 = 1'b0;
            
            Y3 = 1'b1;
        end
        
        6'b010000: begin
            if (w)
                Y1 = 1'b0;
            else
                Y1 = 1'b0;
            
            Y3 = 1'b0;
        end
        
        6'b100000: begin
            if (w)
                Y1 = 1'b0;
            else
                Y1 = 1'b1;
            
            Y3 = 1'b0;
        end
        
        default: begin
            Y1 = 1'b0;
            Y3 = 1'b0;
        end
    endcase
end

endmodule
