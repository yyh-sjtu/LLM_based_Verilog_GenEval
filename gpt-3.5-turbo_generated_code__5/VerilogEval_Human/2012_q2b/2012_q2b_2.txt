reg Y1, Y3;

always @(y or w) begin
	case(y)
		6'b000001: begin // A
			Y1 = w;
			Y3 = 0;
		end
		6'b000010: begin // B
			Y1 = w;
			Y3 = 0;
		end
		6'b000100: begin // C
			Y1 = w;
			Y3 = 1;
		end
		6'b001000: begin // D
			Y1 = w;
			Y3 = 0;
		end
		6'b010000: begin // E
			Y1 = w;
			Y3 = 0;
		end
		6'b100000: begin // F
			Y1 = w;
			Y3 = 1;
		end
	endcase
end

endmodule
