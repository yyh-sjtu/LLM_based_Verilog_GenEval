```verilog
module top_module(
	input clk,
	input reset,
	input ena,
	output reg pm,
	output reg [7:0] hh,
	output reg [7:0] mm,
	output reg [7:0] ss);

	reg [3:0] hours_tens, hours_ones;
	reg [3:0] minutes_tens, minutes_ones;
	reg [3:0] seconds_tens, seconds_ones;
	reg [3:0] am_pm;

	always @(posedge clk or posedge reset) begin
		if (reset) begin
			hh <= 8'b0001_0010; // Set to 12 AM
			mm <= 8'b0000_0000;
			ss <= 8'b0000_0000;
			pm <= 1'b0;
		end else if (ena) begin
			ss <= ss + 1;
			if (ss == 8'b0110_0000) begin // Check for 60 seconds
				ss <= 8'b0000_0000;
				mm <= mm + 1;
				if (mm == 8'b0101_1001) begin // Check for 60 minutes
					mm <= 8'b0000_0000;
					hh <= hh + 1;
					if (hh == 8'b0001_0011) begin // Check for 12 hours
						hh <= 8'b0000_0001;
						am_pm <= ~am_pm;
					end
					if (hh == 8'b0000_0000) begin
						hh <= 8'b0001_0010;
					end
				end
			end
			if (hh == 8'b0001_0010) begin
				if (am_pm) begin
					pm <= 1'b1;
				end else begin
					pm <= 1'b0;
				end
			end
		end
	end
endmodule
```
