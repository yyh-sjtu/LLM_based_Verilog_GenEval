module xnor (
    input x,
    input y,
    output z
);
    assign z = ~(x^y);
endmodule

module xor (
    input x,
    input y,
    output z
);
    assign z = x^y;
endmodule

module top_module (
    input in1,
    input in2,
    input in3,
    output logic out
);
    wire xnor_out;
    xnor xnor_inst (.x(in1), .y(in2), .z(xnor_out));
    xor xor_inst (.x(xnor_out), .y(in3), .z(out));
endmodule
