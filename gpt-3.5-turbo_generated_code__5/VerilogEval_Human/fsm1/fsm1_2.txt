
	// State encoding
	parameter A = 1'b0;
	parameter B = 1'b1;

	// Define state register
	reg [1:0] state, next_state;

	// Define state transition logic
	always @ (posedge clk or posedge areset) begin
		if (areset) begin
			state <= B;
		end else begin
			state <= next_state;
		end
	end

	// Define next state logic
	always @* begin
		case(state)
			A: begin
				if (in) begin
					next_state = A;
				end else begin
					next_state = B;
				end
			end
			B: begin
				if (in) begin
					next_state = B;
				end else begin
					next_state = A;
				end
			end
		endcase
	end

	// Define output logic
	assign out = state;

endmodule
