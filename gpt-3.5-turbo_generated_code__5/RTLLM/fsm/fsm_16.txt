module fsm (
  input   wire        IN,
  input   wire        CLK,
  input   wire        RST,
  output  reg         MATCH
);

  // Create FSM states
  typedef enum logic [3:0] {
    S0,   // Initial state
    S1,   
    S2,
    S3,
    S4,   // Final state
    S5    // Loop detection state
  } state_t;

  // Define state register
  reg   state_t state, next_state;

  // Sequential logic
  always @(posedge CLK or posedge RST) begin
    if (RST) begin
      state <= S0;
      MATCH <= 0;
    end else begin
      state <= next_state;
      MATCH <= (state == S4);  // Output is 1 only in the final state
    end
  end

  // Next-state logic based on input
  always @(state or IN) begin
    next_state = state;
    case (state)
      S0: if (IN == 1) next_state = S1;
      S1: if (IN == 0) next_state = S2;
      S2: if (IN == 0) next_state = S3;
      S3: if (IN == 1) next_state = S4;
      S4: next_state = S5;
      S5: if (IN == 1) next_state = S1;
      default: next_state = S0;
    endcase
  end

endmodule
