module fsm (
  input wire IN,
  input wire CLK,
  input wire RST,
  output reg MATCH
);

  reg [3:0] state;

  always @(posedge CLK or posedge RST) begin
    if (RST) begin
      state <= 4'b0000;
      MATCH <= 0;
    end else begin
      case (state)
        4'b0000: begin
          if (IN) begin
            state <= 4'b0001;
            MATCH <= 0;
          end else begin
            state <= 4'b0000;
            MATCH <= 0;
          end
        end
        
        4'b0001: begin
          if (!IN) begin
            state <= 4'b0010;
            MATCH <= 0;
          end else begin
            state <= 4'b0000;
            MATCH <= 0;
          end
        end
        
        4'b0010: begin
          if (!IN) begin
            state <= 4'b0011;
            MATCH <= 0;
          end else begin
            state <= 4'b0000;
            MATCH <= 0;
          end
        end
        
        4'b0011: begin
          if (IN) begin
            state <= 4'b0100;
            MATCH <= 0;
          end else begin
            state <= 4'b0000;
            MATCH <= 0;
          end
        end
        
        4'b0100: begin
          if (IN) begin
            state <= 4'b0101;
            MATCH <= 1;
          end else begin
            state <= 4'b0000;
            MATCH <= 0;
          end
        end
        
        4'b0101: begin
          if (!IN) begin
            state <= 4'b0010;
            MATCH <= 0;
          end else begin
            state <= 4'b0000;
            MATCH <= 0;
          end
        end
      endcase
    end
  end
endmodule
