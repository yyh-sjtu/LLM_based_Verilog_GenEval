module fsm (
  input wire IN,
  input wire CLK,
  input wire RST,
  output reg MATCH
);

// Define the states of the FSM
typedef enum [2:0] {
  S0, // Initial state
  S1, // Input is 1
  S2, // Input is 0 after 1
  S3, // Input is 0 after 10
  S4  // Input is 1 after 100
} state_type;

// Define the state register and next state logic
reg [2:0] state, next_state;

// Define the Mealy output logic
always @(*) begin
  case(state)
    S0: MATCH = 0;
    S1: MATCH = 0;
    S2: MATCH = 0;
    S3: MATCH = 0;
    S4: MATCH = 1;
  endcase
end

// Define the next state logic based on the input
always @(*) begin
  case(state)
    S0: begin
      if (IN)
        next_state = S1;
      else
        next_state = S0;
    end
    S1: begin
      if (IN)
        next_state = S1;
      else
        next_state = S2;
    end
    S2: begin
      if (IN)
        next_state = S1;
      else
        next_state = S3;
    end
    S3: begin
      if (IN)
        next_state = S1;
      else
        next_state = S4;
    end
    S4: begin
      if (IN)
        next_state = S1;
      else
        next_state = S0;
    end
  endcase
end

// State register update
always @(posedge CLK) begin
  if (RST)
    state <= S0;
  else
    state <= next_state;
end

endmodule
