module pulse_detect(
    input clk,
    input rst_n,
    input data_in,
    output reg data_out
);

    typedef enum logic [1:0] {
        IDLE,
        FIRST_HIGH,
        SECOND_HIGH,
        THIRD_HIGH
    } state_type;

    state_type state, next_state;
    
    always @(posedge clk or negedge rst_n) begin
        if (~rst_n) begin
            state <= IDLE;
            data_out <= 0;
        end
        else begin
            state <= next_state;
            case (state)
                IDLE: begin
                    if (data_in) begin
                        next_state = FIRST_HIGH;
                    end 
                    else begin
                        next_state = IDLE;
                    end
                end
                FIRST_HIGH: begin
                    if (data_in) begin
                        next_state = SECOND_HIGH;
                    end 
                    else begin
                        next_state = IDLE;
                    end
                end
                SECOND_HIGH: begin
                    if (data_in) begin
                        next_state = THIRD_HIGH;
                    end 
                    else begin
                        next_state = IDLE;
                    end
                end
                THIRD_HIGH: begin
                    if (data_in) begin
                        next_state = THIRD_HIGH;
                    end 
                    else begin
                        next_state = IDLE;
                    end
                end
            endcase
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (~rst_n) begin
            data_out <= 0;
        end
        else begin
            if (state == THIRD_HIGH) begin
                data_out <= 1;
            end
            else begin
                data_out <= 0;
            end
        end
    end

endmodule
