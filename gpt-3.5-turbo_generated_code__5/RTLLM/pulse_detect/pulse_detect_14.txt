module pulse_detect (
  input clk,
  input rst_n,
  input data_in,
  output reg data_out
);

  // Declare state register
  reg [2:0] state;

  // State machine for pulse detection
  always @ (posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      state <= 3'b000; // initial state
      data_out <= 1'b0; // no pulse
    end
    else begin
      case (state)
        3'b000: begin // state 0
          if (data_in) begin
            state <= 3'b001;
          end
        end
        3'b001: begin // state 1
          if (!data_in) begin
            state <= 3'b010;
          end
        end          
        3'b010: begin // state 2
          if (data_in) begin
            state <= 3'b011;
          end
        end
        3'b011: begin // state 3 (end of pulse)
          if (!data_in) begin
            state <= 3'b000;
            data_out <= 1'b1; // indicate end of pulse
          end
        end
      endcase
    end
  end

endmodule
