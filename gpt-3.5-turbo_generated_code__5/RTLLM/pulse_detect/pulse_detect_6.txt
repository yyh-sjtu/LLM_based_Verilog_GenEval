module pulse_detect(
  input clk,
  input rst_n,
  input data_in,
  output reg data_out
);

  // Define the states for pulse detection
  localparam IDLE = 2'b00;
  localparam ACTIVE = 2'b01;
  localparam DETECTED = 2'b10;

  // Register to keep track of the current state
  reg [1:0] state;

  // Always block to detect pulses and generate output
  always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      state <= IDLE;    // Reset state to idle
      data_out <= 1'b0; // Output indicates no pulse
    end else begin
      case (state)
        IDLE: begin
          if (data_in == 1'b0)
            state <= ACTIVE; // Move to active state if input is 0
        end
        ACTIVE: begin
          if (data_in == 1'b1)
            state <= DETECTED; // Move to detected state if input is 1
        end
        DETECTED: begin
          if (data_in == 1'b0)
            state <= IDLE;   // Move back to idle state if input is 0
        end
      endcase
      
      // Generate output based on the state
      if (state == DETECTED)
        data_out <= 1'b1; // Output indicates end of pulse
      else
        data_out <= 1'b0; // Output indicates no pulse
    end
  end
  
endmodule
