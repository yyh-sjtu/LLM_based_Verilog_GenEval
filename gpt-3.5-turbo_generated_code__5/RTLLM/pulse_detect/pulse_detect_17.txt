```verilog
module pulse_detect (
  input wire clk,
  input wire rst_n,
  input wire data_in,
  output reg data_out
);

  reg [1:0] state;

  parameter IDLE = 2'b00, STATE1 = 2'b01, STATE2 = 2'b10;

  always @(posedge clk or negedge rst_n) begin
    if (~rst_n)
      begin
        state <= IDLE;
        data_out <= 1'b0;
      end
    else
      case (state)
        IDLE: begin
          if (data_in == 1'b0)
            state <= STATE1;
          else
            state <= IDLE;
        end

        STATE1: begin
          if (data_in == 1'b1)
            state <= STATE1;
          else
            state <= STATE2;
        end

        STATE2: begin
          if (data_in == 1'b0)
            state <= IDLE;
          else
            state <= STATE2;
        end
      endcase

    if (state == STATE2)
      data_out <= 1'b1;
    else
      data_out <= 1'b0;
  end

endmodule
