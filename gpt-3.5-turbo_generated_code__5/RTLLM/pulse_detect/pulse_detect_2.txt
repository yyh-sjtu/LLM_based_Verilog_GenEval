module pulse_detect (
  input clk,
  input rst_n,
  input data_in,
  output reg data_out
);

  // Declare state register
  reg [2:0] state;

  // State definitions
  parameter IDLE = 3'b000;
  parameter START = 3'b001;
  parameter MID = 3'b010;
  parameter END = 3'b011;

  // Always block for pulse detection and output generation
  always @(posedge clk, negedge rst_n) begin
    if (~rst_n) begin
      state <= IDLE;
      data_out <= 1'b0;
    end
    else begin
      case (state)
        IDLE: begin
          if (data_in == 1'b1)
            state <= START;
          else
            state <= IDLE;
        end
        START: begin
          if (data_in == 1'b0)
            state <= MID;
          else
            state <= START;
        end
        MID: begin
          if (data_in == 1'b1)
            state <= END;
          else
            state <= MID;
        end
        END: begin
          state <= IDLE;
          data_out <= 1'b1;
        end
      endcase
      if (state != END)
        data_out <= 1'b0;
    end
  end

endmodule
