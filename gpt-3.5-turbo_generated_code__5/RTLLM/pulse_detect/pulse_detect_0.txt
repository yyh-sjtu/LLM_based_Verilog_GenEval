module pulse_detect (
    input clk,
    input rst_n,
    input data_in,
    output reg data_out
);

    reg [2:0] state;

    always @(posedge clk or negedge rst_n) begin
        if (~rst_n) begin
            state <= 3'b000;
            data_out <= 0;
        end else begin
            case (state)
                3'b000: if (data_in) state <= 3'b001;
                3'b001: if (data_in) state <= 3'b010;
                3'b010: if (~data_in) state <= 3'b011;
                3'b011: begin
                            state <= 3'b100;
                            data_out <= 1;
                        end
                3'b100: if (~data_in) state <= 3'b000;
            endcase
        end
    end

endmodule
