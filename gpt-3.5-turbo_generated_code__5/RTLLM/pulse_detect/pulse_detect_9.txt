module pulse_detect(
    input clk,
    input rst_n,
    input data_in,
    output reg data_out
);

// Declare state register
reg [1:0] state;

// Define states
parameter IDLE = 2'b00;
parameter FIRST_HIGH = 2'b01;
parameter SECOND_HIGH = 2'b10;

// State machine logic
always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        state <= IDLE;  // Reset state machine
        data_out <= 0;  // Reset data_out
    end
    else begin
        case(state)
            IDLE: begin
                if (data_in) begin
                    state <= FIRST_HIGH;  // Transition to FIRST_HIGH state
                end
            end
            FIRST_HIGH: begin
                if (~data_in) begin
                    state <= SECOND_HIGH;  // Transition to SECOND_HIGH state
                end
            end
            SECOND_HIGH: begin
                if (~data_in) begin
                    state <= IDLE;  // Transition back to IDLE state
                end
                else begin
                    data_out <= 1;  // Indicates end of pulse
                end
            end
            default: state <= IDLE;  // Default state
        endcase
    end
end

endmodule
