module multi_pipe_4bit (
    input wire clk,
    input wire rst_n,
    input wire [3:0] mul_a,
    input wire [3:0] mul_b,
    output reg [7:0] mul_out
);

parameter size = 4;

always @ (posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        mul_out <= 8'b0;
    end else begin
        reg [7:0] partial_product [3:0];
        reg [7:0] sum_reg1, sum_reg2;

        // Extension of input signals
        reg [7:0] ext_mul_a, ext_mul_b;
        ext_mul_a = {4'd0, mul_a};
        ext_mul_b = {4'd0, mul_b};

        // Multiplication operation
        generate
            genvar i;
            for (i = 0; i < size; i = i + 1) begin
                always @* begin
                    if (ext_mul_b[i] == 1'b1) begin
                        partial_product[i] = ext_mul_a << i;
                    end else begin
                        partial_product[i] = 8'd0;
                    end
                end
            end
        endgenerate

        // Add of partial products
        always @ (posedge clk or negedge rst_n) begin
            if (rst_n == 1'b0) begin
                sum_reg1 <= 8'b0;
                sum_reg2 <= 8'b0;
            end else begin
                sum_reg1 <= partial_product[0] + partial_product[1];
                sum_reg2 <= partial_product[2] + partial_product[3];
            end
        end

        // Final product calculation
        always @ (posedge clk or negedge rst_n) begin
            if (rst_n == 1'b0) begin
                mul_out <= 8'b0;
            end else begin
                mul_out <= sum_reg1 + sum_reg2;
            end
        end
    end
end

endmodule
