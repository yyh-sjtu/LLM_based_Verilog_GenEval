module multi_pipe_4bit (
  input wire clk,
  input wire rst_n,
  input wire [3:0] mul_a,
  input wire [3:0] mul_b,
  output reg [7:0] mul_out
);

  parameter size = 4;
  reg [7:0] reg1, reg2;

  always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      reg1 <= 8'd0;
      reg2 <= 8'd0;
      mul_out <= 8'd0;
    end else begin
      reg1 <= {mul_a, 4'b0000};
      reg2 <= mul_b;
    end
  end

  always @(*) begin
    for (i=0; i<size; i=i+1) begin
      if (reg2[i] == 1) begin
        reg1 <= reg1 << i;
        mul_out <= mul_out + reg1;
        reg1 <= {mul_a, 4'b0000};
      end
    end
  end

endmodule
