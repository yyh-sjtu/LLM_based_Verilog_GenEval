module multi_pipe_4bit (
    input           clk,
    input           rst_n,
    input  [3:0]    mul_a,
    input  [3:0]    mul_b,
    output  [7:0]   mul_out
);

parameter size = 4;

reg [size*2-1:0]    mul_partial[size-1:0];
reg [size*2-1:0]    mul_sum;

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        mul_sum <= 8'b0;
        for (i = 0; i < size; i = i + 1) begin
            mul_partial[i] <= 8'b0;
        end
    end
    else begin
        for (j = 0; j < size; j = j + 1) begin
            if (mul_b[j] == 1'b1) begin
                mul_partial[j] <= {mul_a, (size-1-j){4'b0}} << j;
            end
            else begin
                mul_partial[j] <= 8'b0;
            end
        end
        mul_sum <= mul_partial[0];
        for (k = 1; k < size; k = k + 1) begin
            mul_sum <= mul_sum + mul_partial[k];
        end
    end
end

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        mul_out <= 8'b0;
    end
    else begin
        mul_out <= mul_sum;
    end
end

endmodule
