module multi_pipe_4bit (
    input clk,
    input rst_n,
    input [3:0] mul_a,
    input [3:0] mul_b,
    output reg [7:0] mul_out
);

parameter size = 4;

reg [7:0] partial_products [0:size-1];
reg [7:0] intermediate_sums [0:size-1];

// Extend input signals
assign extended_mul_a = {size{mul_a}};
assign extended_mul_b = {size{mul_b}};

// Multiplication operation
genvar i;
generate
    for (i=0; i<size; i=i+1) begin : partial_product_gen
        always @(*) begin
            if (extended_mul_b[i]) begin
                partial_products[i] = extended_mul_a << i;
            end
            else begin
                partial_products[i] = 8'b0;
            end
        end
    end
endgenerate

// Add of partial products
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        intermediate_sums <= 8'b0;
    end
    else begin
        for (i=0; i<size; i=i+1) begin
            intermediate_sums[i] <= intermediate_sums[i] + partial_products[i];
        end
    end
end

// Final product calculation
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        mul_out <= 8'b0;
    end
    else begin
        mul_out <= intermediate_sums[0];
    end
end

endmodule
