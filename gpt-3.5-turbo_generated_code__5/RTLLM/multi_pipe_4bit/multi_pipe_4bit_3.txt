module multi_pipe_4bit (
    input clk,
    input rst_n,
    input [3:0] mul_a,
    input [3:0] mul_b,
    output reg [7:0] mul_out
);

parameter size = 4;

reg [7:0] partial_products [0:size-1];
reg [7:0] sum_reg1, sum_reg2;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        sum_reg1 <= 8'b0;
        sum_reg2 <= 8'b0;
        for (int i = 0; i < size; i = i + 1) begin
            partial_products[i] <= 8'b0;
        end
    end
    else begin
        sum_reg1 <= partial_products[0];
        sum_reg2 <= partial_products[1];
        for (int i = 2; i < size; i = i + 1) begin
            sum_reg1 <= sum_reg1 + partial_products[i];
            sum_reg2 <= sum_reg2 + partial_products[i];
        end
    end
end

always @(*) begin
    for (int i = 0; i < size; i = i + 1) begin
        if (mul_b[i] == 1) begin
            partial_products[i] <= {mul_a, (size-1)-i{1'b0}};
        end
        else begin
            partial_products[i] <= 8'b0;
        end
    end
end

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        mul_out <= 8'b0;
    end
    else begin
        mul_out <= sum_reg1 + (sum_reg2 << 4);
    end
end

endmodule
