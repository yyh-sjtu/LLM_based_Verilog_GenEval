module radix2_div (
    input wire clk,
    input wire rst,
    input wire sign,
    input wire [7:0] dividend,
    input wire [7:0] divisor,
    input wire opn_valid,
    output wire res_valid,
    output wire [15:0] result
);

reg [15:0] result_reg;
reg [7:0] SR;
reg [7:0] NEG_DIVISOR;
reg [7:0] cnt;
reg start_cnt;

always @ (posedge clk or posedge rst) begin
    if (rst) begin
        SR <= 8'b0;
        NEG_DIVISOR <= 8'b0;
        cnt <= 8'b0;
        start_cnt <= 1'b0;
        res_valid <= 1'b0;
    end else begin
        if (opn_valid && !res_valid) begin
            SR <= {1'b0,dividend} << 1;
            NEG_DIVISOR <= -divisor;
            cnt <= 8'b1;
            start_cnt <= 1'b1;
        end
        
        if (start_cnt) begin
            if(cnt == 8'b11111111) begin
                cnt <= 8'b0;
                start_cnt <= 1'b0;
                res_valid <= 1'b1;
                result_reg <= {SR, cnt};
            end else begin
                cnt <= cnt + 1;
                if (sign) begin
                    if (SR[7]) begin
                        SR <= (SR + NEG_DIVISOR) << 1 | 1'b1;
                    end else begin
                        SR <= (SR << 1);
                    end
                end else begin
                    SR <= (SR + NEG_DIVISOR) << 1 | SR[8];
                end
            end
        end
    end
end

assign result = result_reg;

endmodule
