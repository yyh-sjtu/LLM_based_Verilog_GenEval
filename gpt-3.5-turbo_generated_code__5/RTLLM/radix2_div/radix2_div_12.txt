    input wire rst,
    input wire sign,

    input wire [7:0] divisor,
    input wire opn_valid,
    output wire res_valid,
    output wire [15:0] result
);

    reg [15:0] shift_reg;
    reg signed [15:0] NEG_DIVISOR;

    reg [3:0] start_cnt;
    reg [15:0] final_result;

            cnt <= 0;
            start_cnt <= 0;

            res_valid_reg <= 0;

                res_valid_reg <= 1;

                cnt <= cnt + 1;
                shift_reg <= {shift_reg[14:0], shift_reg[15]};

                    if (shift_reg[15] == 1'b0 && shift_reg[14:8] >= NEG_DIVISOR[14:8]) begin
                        shift_reg[8:1] <= shift_reg[8:1] - NEG_DIVISOR[14:8];

                        shift_reg[0] <= 0;
                    end else begin
                        shift_reg[0] <= 0;
                    end
                end else begin
                    if (shift_reg[15] == 1'b0 && shift_reg[14:8] >= divisor[7:1]) begin
                        shift_reg[8:1] <= shift_reg[8:1] - divisor[7:1];
                        shift_reg[0] <= 1;

                        shift_reg[8:1] <= shift_reg[8:1] + divisor[7:1];

                    end else begin
                        shift_reg[0] <= 0;
                    end
                end
            end
        end else if (res_valid_reg && ~result_reg[15]) begin
            result_reg <= {result_reg[14:0], 1'b1};
        end else if (res_valid_reg && result_reg[15]) begin
            res_valid_reg <= 0;
            result_reg <= 0;
        end
    end
    
    assign res_valid = res_valid_reg;
    assign result = result_reg;
    
endmodule
