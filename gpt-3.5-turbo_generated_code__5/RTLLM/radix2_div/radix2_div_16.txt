Here is the complete code for the simplified radix-2 divider:

```verilog
module radix2_div (
  input clk,
  input rst,
  input sign,
  input [7:0] dividend,
  input [7:0] divisor,
  input opn_valid,
  output reg res_valid,
  output reg [15:0] result
);

reg [7:0] SR;
reg [7:0] NEG_DIVISOR;
reg [2:0] cnt;
reg start_cnt;

always @(posedge clk) begin
  if (rst == 1'b1) begin
    SR <= 8'b0;
    NEG_DIVISOR <= 8'b0;
    cnt <= 3'b0;
    start_cnt <= 1'b0;
    res_valid <= 1'b0;
    result <= 16'b0;
  end else if (opn_valid == 1'b1 && res_valid == 1'b0) begin
    SR <= {1'b0, dividend};
    if (sign == 1'b1)
      NEG_DIVISOR <= -divisor;
    else
      NEG_DIVISOR <= 8'b0 - divisor;
    cnt <= 3'b1;
    start_cnt <= 1'b1;
    res_valid <= 1'b0;
    result <= 16'b0;
  end else if (start_cnt == 1'b1) begin
    if (cnt == 3'b100) begin
      SR <= {SR[6:0], carry_out};
      cnt <= 3'b0;
      start_cnt <= 1'b0;
      res_valid <= 1'b1;
      result <= {SR[7], SR[6:0], SR[7]};
    end else begin
      cnt <= cnt + 1'b1;
      if (SR >= NEG_DIVISOR)
        carry_out <= 1'b0;
      else
        carry_out <= 1'b1;
      SR <= {carry_out, SR[6:0]};
    end
  end else begin
    res_valid <= 1'b0;
    result <= 16'b0;
  end
end

endmodule
```

Please note that this code assumes the divisor and dividend are already in 2's complement format for signed division. If the inputs are unsigned, you will need to convert them to 2's complement format before passing them to the module.
Also, make sure to replace the `clk` signal in the module with your clock signal, and connect the `rst` signal as necessary in your design.
