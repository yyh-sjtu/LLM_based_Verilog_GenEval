module radix2_div (
    input wire clk,
    input wire rst,
    input wire sign,
    input wire [7:0] dividend,
    input wire [7:0] divisor,
    input wire opn_valid,
    output reg res_valid,
    output reg [15:0] result
);

reg [7:0] SR;
reg [7:0] NEG_DIVISOR;
reg [3:0] cnt;
reg start_cnt;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        res_valid <= 0;
        SR <= 0;
        NEG_DIVISOR <= 0;
        cnt <= 0;
        start_cnt <= 0;
        result <= 0;
    end
    else begin
        if (opn_valid && !res_valid) begin
            SR <= {1'b0, abs(dividend)};
            NEG_DIVISOR <= -abs(divisor);
            cnt <= 4'd1;
            start_cnt <= 1;
        end
        else begin
            if (start_cnt) begin
                if (cnt == 8) begin
                    result <= {SR, SR};
                    cnt <= 0;
                    start_cnt <= 0;
                    res_valid <= 1;
                end
                else begin
                    cnt <= cnt + 1;
                    {SR, _} <= SR - NEG_DIVISOR;
                    SR <= {SR[6:0], _};
                end
            end
        end
    end
end

endmodule
