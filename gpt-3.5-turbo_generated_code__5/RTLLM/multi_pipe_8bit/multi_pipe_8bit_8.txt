module multi_pipe_8bit (
  input wire clk,
  input wire rst_n,
  input wire mul_en_in,
  input wire [7:0] mul_a,
  input wire [7:0] mul_b,
  output reg mul_en_out,
  output reg [15:0] mul_out
);

  reg [7:0] mul_a_reg;
  reg [7:0] mul_b_reg;
  reg [15:0] mul_out_reg;
  reg [7:0] temp [7:0];
  reg [15:0] sum [7:0];
  reg [1:0] stage;
  reg [15:0] tmp;
  
  always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      mul_en_out <= 0;
      mul_a_reg <= 0;
      mul_b_reg <= 0;
      mul_out_reg <= 0;
      stage <= 0;
    end else begin
      mul_a_reg <= mul_en_in ? mul_a : mul_a_reg;
      mul_b_reg <= mul_en_in ? mul_b : mul_b_reg;
      temp[0] <= mul_b_reg[0] ? mul_a_reg : 0;
      sum[0] <= temp[0];
      tmp <= {8'b0, sum[0]};
      for (i = 1; i <= 6; i = i + 1) begin
        temp[i] <= mul_b_reg[i] ? mul_a_reg : 0;
        sum[i] <= temp[i] ^ temp[i-1];
        tmp <= {tmp[14:0], sum[i]};
      end
      mul_out_reg <= tmp ^ (mul_b_reg[7] ? {8'd0, mul_a_reg} : 0);
      mul_en_out <= mul_en_in;
    end
  end

  assign mul_out = mul_en_out ? mul_out_reg : 16'd0;

endmodule
