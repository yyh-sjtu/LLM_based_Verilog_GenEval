module multi_pipe_8bit (
  input wire clk,
  input wire rst_n,
  input wire mul_en_in,
  input wire [7:0] mul_a,
  input wire [7:0] mul_b,
  output wire mul_en_out,
  output wire [15:0] mul_out
);

  reg mul_en_out_reg;
  reg [7:0] mul_a_reg;
  reg [7:0] mul_b_reg;
  wire [7:0] temp [7:0];
  reg [15:0] sum [7:0];
  reg [15:0] mul_out_reg;

  always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      mul_en_out_reg <= 0;
      mul_a_reg <= 0;
      mul_b_reg <= 0;
      mul_out_reg <= 0;
      for (integer i = 0; i < 8; i = i + 1) begin
        temp[i] <= 0;
        sum[i] <= 0;
      end
    end
    else begin
      mul_en_out_reg <= mul_en_in;
      if (mul_en_in) begin
        mul_a_reg <= mul_a;
        mul_b_reg <= mul_b;
        for (integer i = 0; i < 8; i = i + 1) begin
          temp[i] <= mul_a_reg & (mul_b_reg[i] ? 8'hFF : 8'h00);
        end
      end
    end
  end
  
  always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      mul_out_reg <= 0;
    end
    else begin
      if (mul_en_out_reg) begin
        for (integer i = 0; i < 8; i = i + 1) begin
          sum[i] <= mul_en_out_reg ? ((i == 0) ? temp[i] : (temp[i] + sum[i-1])) : sum[i];
        end
        mul_out_reg <= sum[7];
      end
    end
  end
  
  assign mul_en_out = mul_en_out_reg;
  assign mul_out = (mul_en_out_reg) ? mul_out_reg[15:0] : 16'b0;

endmodule
