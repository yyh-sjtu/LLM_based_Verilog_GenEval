module multi_pipe_8bit(
    input clk,
    input rst_n,
    input mul_en_in,
    input [7:0] mul_a,
    input [7:0] mul_b,
    output reg mul_en_out,
    output reg [15:0] mul_out
);

reg [7:0] mul_a_reg;
reg [7:0] mul_b_reg;
wire [15:0] temp;
reg [15:0] sum;

always @(posedge clk or negedge rst_n) begin
    if(~rst_n) begin
        mul_a_reg <= 8'b0;
        mul_b_reg <= 8'b0;
    end
    else if(mul_en_in) begin
        mul_a_reg <= mul_a;
        mul_b_reg <= mul_b;
    end
end

always @(posedge clk) begin
    if(mul_en_in) begin
        mul_en_out <= mul_en_in;
    end
    else begin
        mul_en_out <= 1'b0;
    end
end

assign temp[0] = mul_en_in ? mul_a_reg[0] & mul_b_reg : 0;
assign temp[1] = mul_en_in ? mul_a_reg[1] & mul_b_reg : 0;
assign temp[2] = mul_en_in ? mul_a_reg[2] & mul_b_reg : 0;
assign temp[3] = mul_en_in ? mul_a_reg[3] & mul_b_reg : 0;
assign temp[4] = mul_en_in ? mul_a_reg[4] & mul_b_reg : 0;
assign temp[5] = mul_en_in ? mul_a_reg[5] & mul_b_reg : 0;
assign temp[6] = mul_en_in ? mul_a_reg[6] & mul_b_reg : 0;
assign temp[7] = mul_en_in ? mul_a_reg[7] & mul_b_reg : 0;
    
always @(posedge clk) begin
    if(mul_en_in) begin
        sum <= temp[0] + {8'b0, temp[1:7]};
    end
end

always @(posedge clk) begin
    if(mul_en_in) begin
        mul_out <= sum;
    end
    else begin
        mul_out <= 16'b0;
    end
end

endmodule
