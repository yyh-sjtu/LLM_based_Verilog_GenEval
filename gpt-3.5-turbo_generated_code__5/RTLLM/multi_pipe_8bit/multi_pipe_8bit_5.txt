        mul_a_reg <= mul_a;

          temp[i] <= {mul_a_reg[7:0], 1'b0} >> i;

        sum[i] <= sum[i] + temp[i];
      mul_out_reg <= {sum[6], sum[5:0]};

    end
  end

endmodule
