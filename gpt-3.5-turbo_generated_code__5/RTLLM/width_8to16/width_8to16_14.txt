module width_8to16 (
    input wire clk,
    input wire rst_n,
    input wire valid_in,
    input wire [7:0] data_in,
    output reg valid_out,
    output reg [15:0] data_out
);

reg [7:0] data_register;
reg [1:0] state;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        data_register <= 8'b0;
        valid_out <= 1'b0;
        data_out <= 16'b0;
        state <= 2'b00;
    end else begin
        case (state)
            2'b00: begin // Waiting for first valid input
                if (valid_in) begin
                    data_register <= data_in;
                    state <= 2'b01; // Move to the next state
                end
            end
            2'b01: begin // Waiting for second valid input
                if (valid_in) begin
                    data_out <= {data_register, data_in}; // Concatenate first and second data
                    valid_out <= 1'b1;
                    state <= 2'b00; // Reset to initial state
                end
            end
        endcase
    end
end

endmodule
