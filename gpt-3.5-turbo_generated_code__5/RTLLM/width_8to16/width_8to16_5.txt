```verilog
module width_8to16 (
    input clk,
    input rst_n,
    input valid_in,
    input [7:0] data_in,
    output valid_out,
    output [15:0] data_out
);

    reg valid_out;
    reg [15:0] data_out;

    reg [7:0] data_lock;
    reg [7:0] data_in_reg;
    reg flag;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out <= 16'h0;
            valid_out <= 1'b0;
            flag <= 1'b0;
        end else begin
            if (valid_in & !flag) begin
                // Store first valid data
                data_lock <= data_in;

                // Set flag to activate concatenation
                flag <= 1'b1;
            end else if (valid_in & flag) begin
                // Concatenate first and second valid data
                data_out <= {data_lock, data_in};
                valid_out <= 1'b1;

                // Reset the flag and store new first valid data
                flag <= 1'b0;
                data_lock <= data_in;
            end
        end
    end

endmodule
```

This verilog code implements a data width conversion circuit that converts 8-bit data input to 16-bit data output. The circuit waits for the arrival of two valid data inputs before generating the output signals valid_out and data_out as per the provided requirements. The output data_out is formed by concatenating the higher 8 bits of the first valid data input and the lower 8 bits of the second valid data input. The flag signal is used to indicate whether the first valid data input has been received or not. On reset, the circuit clears the data_out register to 0 and sets valid_out to 0.
