// Verilog module for 8-to-16 bit data width conversion circuit

module width_8to16 (
    input clk,
    input rst_n,
    input valid_in,
    input [7:0] data_in,
    output reg valid_out,
    output reg [15:0] data_out
);

reg [7:0] data_reg;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        data_reg <= 8'b0;
        valid_out <= 1'b0;
        data_out <= 16'b0;
    end else if (valid_in) begin
        data_reg <= data_in;
        if (valid_out) begin
            valid_out <= 1'b1;
            data_out <= {data_reg, data_in};
        end else begin
            valid_out <= 1'b0;
            data_out <= 16'b0;
        end
    end
end

endmodule
