module counter_12 (
  input wire rst_n,
  input wire clk,
  input wire valid_count,
  output reg [3:0] out
);

always @(posedge clk or negedge rst_n) begin
  if (!rst_n) begin
    out <= 4'b0000; // reset to 0
  end else begin
    if (valid_count) begin
      // increment counter
      if (out == 4'd11) begin
        out <= 4'b0000; // wrap around to 0
      end else begin
        out <= out + 1; // increment by 1
      end
    end
    // else: stop the counter, do nothing
  end
end

endmodule
