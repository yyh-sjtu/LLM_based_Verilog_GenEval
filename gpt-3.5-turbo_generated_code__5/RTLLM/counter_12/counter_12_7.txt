// Verilog code for the 4-bit counter design

module counter_12 (
    input wire rst_n, // Reset signal (active low)
    input wire clk,   // Clock signal
    input wire valid_count, // Signal to enable counting
    output reg [3:0] out // 4-bit output representing the current count value
);

// Internal signal for incrementing the counter
reg [3:0] next_count;

// Always block for counting and control
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        out <= 4'b0000; // Reset to 0
    end else begin
        if (valid_count) begin
            if (out == 4'd11) begin
                out <= 4'b0000; // Wrap around to 0
            end else begin
                out <= out + 4'd1; // Increment the count
            end
        end
        // If valid_count = 0, the counter is paused and remains at the current count value
    end
end

endmodule

// End of Verilog code for the 4-bit counter design
