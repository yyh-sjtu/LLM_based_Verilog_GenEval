module adder_8bit (
    input [7:0] a,
    input [7:0] b,
    input cin,
    output [7:0] sum,
    output cout
);

wire [7:0] c;
wire [6:0] cin_chain;
wire [7:0] sum_internal;

// First bit-level (full adder) addition
full_adder fa0(.a(a[0]), .b(b[0]), .cin(cin), .sum(sum_internal[0]), .cout(c[0]));

// Remaining bit-level additions
genvar i;
generate
    for (i = 1; i < 8; i = i + 1) begin: adder_chain
        full_adder fa(
            .a(a[i]),
            .b(b[i]),
            .cin(cin_chain[i-1]),
            .sum(sum_internal[i]),
            .cout(c[i])
        );
    end
endgenerate

assign cin_chain[6:0] = c[0:6];
assign cout = c[7];
assign sum = sum_internal;

endmodule
