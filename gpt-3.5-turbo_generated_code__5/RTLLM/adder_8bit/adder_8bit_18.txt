module adder_8bit(
    input [7:0] a,
    input [7:0] b,
    input cin,
    output [7:0] sum,
    output cout
);

wire [7:0] c;
wire [6:0] cin_internal;

assign cin_internal[0] = cin;
assign cin_internal[1] = c[0];
assign cin_internal[2] = c[1];
assign cin_internal[3] = c[2];
assign cin_internal[4] = c[3];
assign cin_internal[5] = c[4];
assign cin_internal[6] = c[5];

full_adder fa0(a[0], b[0], cin_internal[0], sum[0], c[0]);
full_adder fa1(a[1], b[1], cin_internal[1], sum[1], c[1]);
full_adder fa2(a[2], b[2], cin_internal[2], sum[2], c[2]);
full_adder fa3(a[3], b[3], cin_internal[3], sum[3], c[3]);
full_adder fa4(a[4], b[4], cin_internal[4], sum[4], c[4]);
full_adder fa5(a[5], b[5], cin_internal[5], sum[5], c[5]);
full_adder fa6(a[6], b[6], cin_internal[6], sum[6], c[6]);
full_adder fa7(a[7], b[7], c[7], sum[7], cout);

endmodule

module full_adder(
    input a,
    input b,
    input cin,
    output sum,
    output cout
);

assign sum = a ^ b ^ cin;
assign cout = (a & b) | (a & cin) | (b & cin);

endmodule
