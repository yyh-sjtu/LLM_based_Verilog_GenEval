module adder_16bit (
    input [15:0] a,
    input [15:0] b,
    input Cin,
    output [15:0] y,
    output Co
);

wire [7:0] c;
wire [7:0] s;

// Instantiating eight 8-bit adders
genvar i;
generate
    for (i = 0; i < 8; i = i + 1) begin : adder_instance
        full_adder_8bit adder_inst (
            .a(a[i*2 +: 8]),
            .b(b[i*2 +: 8]),
            .Cin(i == 0 ? Cin : c[i-1]),
            .y(y[i*2 +: 8]),
            .Co(c[i])
        );
    end
endgenerate

assign Co = c[7];

endmodule

module full_adder_8bit (
    input [7:0] a,
    input [7:0] b,
    input Cin,
    output [7:0] y,
    output Co
);

wire [6:0] c;
wire [6:0] s;

// Instantiating 8 1-bit full adders
genvar j;
generate
    for (j = 0; j < 8; j = j + 1) begin : full_adder_instance
        full_adder  adder_inst (
            .a(a[j]),
            .b(b[j]),
            .Cin(j == 0 ? Cin : c[j-1]),
            .y(y[j]),
            .Co(c[j])
        );
    end
endgenerate

assign Co = c[7];

endmodule

module full_adder (
    input a,
    input b,
    input Cin,
    output y,
    output Co
);

assign {Co, y} = a + b + Cin;

endmodule
