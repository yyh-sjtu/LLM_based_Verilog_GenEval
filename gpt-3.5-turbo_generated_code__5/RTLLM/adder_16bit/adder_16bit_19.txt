module adder_16bit(
  input [15:0] a,
  input [15:0] b,
  input Cin,
  output [15:0] y,
  output Co
);

  wire [7:0] c;
  wire [7:0] s;

  // 8-bit adders
  genvar i;
  generate
    for (i = 0; i < 2; i = i + 1) begin : gen_adders
      full_adder_8bit fa(
        .a(a[i*8 +: 8]),
        .b(b[i*8 +: 8]),
        .Cin(Cin),
        .y(y[i*8 +: 8]),
        .Co(c[i])
      );
    end
  endgenerate

  // Carry-out
  assign Co = c[1];

endmodule

module full_adder_8bit(
  input [7:0] a,
  input [7:0] b,
  input Cin,
  output [7:0] y,
  output Cout
);

  wire [6:0] c;
  wire [6:0] s;

  // 1st stage adder
  genvar i;
  generate
    for (i = 0; i < 7; i = i + 1) begin : gen_adders_half
      full_adder fa(
        .a(a[i]),
        .b(b[i]),
        .Cin(Cin),
        .y(s[i]),
        .Co(c[i])
      );
    end
  endgenerate
  full_adder fa_last(
    .a(a[7]),
    .b(b[7]),
    .Cin(Cin),
    .y(s[7]),
    .Co(c[7])
  );

  // 2nd stage adder
  full_adder fa2(
    .a(s[0]),
    .b(s[1]),
    .Cin(c[0]),
    .y(y[0]),
    .Co(c[1])
  );
  for (i = 1; i < 6; i = i + 1) begin
    full_adder fa(
      .a(s[i]),
      .b(y[i-1]),
      .Cin(c[i]),
      .y(y[i]),
      .Co(c[i+1])
    );
  end
  full_adder fa_last2(
    .a(s[7]),
    .b(y[6]),
    .Cin(c[7]),
    .y(y[7]),
    .Co(Cout)
  );

endmodule

module full_adder(
  input a,
  input b,
  input Cin,
  output y,
  output Co
);

  assign {Co, y} = a + b + Cin;

endmodule
