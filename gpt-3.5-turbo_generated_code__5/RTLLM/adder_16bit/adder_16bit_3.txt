module adder_16bit (
  input [15:0] a,
  input [15:0] b,
  input Cin,
  output [15:0] y,
  output Co
);

  wire [7:0] carry_outs;
  wire [7:0] sum_bits;

  // Instantiating 8-bit adders
  genvar i;
  generate
    for (i = 0; i < 8; i = i + 1) begin : adder_gen
      full_adder_8bit adder_inst (
        .a(a[i*2 +: 8]),
        .b(b[i*2 +: 8]),
        .Cin(i ? carry_outs[i-1] : Cin),
        .y(sum_bits[i*2 +: 8]),
        .Co(carry_outs[i])
      );
    end
  endgenerate

  assign y = {sum_bits, carry_outs[7]}; // Concatenating sum bits with the last carry_out
  assign Co = carry_outs[7]; // Setting the final carry_out

endmodule

module full_adder_8bit (
  input [7:0] a,
  input [7:0] b,
  input Cin,
  output [7:0] y,
  output Co
);

  wire [7:0] c_outs;
  wire [7:0] s_bits;

  genvar i;
  generate
    for (i = 0; i < 8; i = i + 1) begin : adder_gen
      full_adder adder_inst (
        .a(a[i]),
        .b(b[i]),
        .Cin(i ? c_outs[i-1] : Cin),
        .y(s_bits[i]),
        .Co(c_outs[i])
      );
    end
  endgenerate

  assign y = s_bits;
  assign Co = c_outs[7];

endmodule

module full_adder (
  input a,
  input b,
  input Cin,
  output y,
  output Co
);

  // Full adder logic
  assign y = a ^ b ^ Cin;
  assign Co = (a & b) | (a & Cin) | (b & Cin);

endmodule
