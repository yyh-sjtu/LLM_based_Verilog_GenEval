```
module parallel2serial(
  input clk,
  input rst_n,
  input [3:0] d,
  output valid_out,
  output dout
);

  reg [2:0] cnt;
  reg [3:0] data;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      cnt <= 0;
      data <= 0;
      valid_out <= 0;
      dout <= 0;
    end
    else begin
      if (cnt == 3) begin
        data <= d;
        cnt <= 0;
        valid_out <= 1;
        dout <= data[3];
      end
      else begin
        cnt <= cnt + 1;
        valid_out <= 0;
        data <= {data[2:0], d[3]};
        dout <= data[3];
      end
    end
  end

endmodule
```

In this case, we update the `data` register with the new parallel input `d`, reset the `cnt` to 0, and set the `valid_out` signal to 1. The most significant bit of `data` is then assigned to `dout` for serial output.

The `rst_n` signal is used for asynchronous reset. When it is low, all register values are reset to their initial values.
