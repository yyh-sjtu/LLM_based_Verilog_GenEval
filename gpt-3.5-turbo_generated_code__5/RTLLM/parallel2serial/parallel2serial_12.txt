module parallel2serial (
    input clk,
    input rst_n,
    input [3:0] d,
    output valid_out,
    output reg dout
);

reg [3:0] data;
reg [1:0] cnt;
reg valid;

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        cnt <= 2'b00;
    end
    else begin
        if (cnt == 2'b11) begin
            data <= d;
            cnt <= 2'b00;
            valid <= 1;
        end
        else begin
            cnt <= cnt + 1;
            valid <= 0;
            data <= {data[2:0], data[3]};
        end
    end
end

always @(posedge clk) begin
    if (valid) begin
        dout <= data[3];
    end
end

endmodule
