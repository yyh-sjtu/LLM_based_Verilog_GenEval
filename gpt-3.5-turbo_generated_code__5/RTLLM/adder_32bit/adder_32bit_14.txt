module cla_16bit (
    input [15:0] A,
    input [15:0] B,
    output [15:0] S,
    output C
);

// Generate and Propagate section
wire [3:0] G, P;
assign G = A & B;
assign P = A ^ B;

// Generate and Propagate for first stage
wire G0, G1, P0, P1;
assign G0 = G[0];
assign G1 = G[1];
assign P0 = P[0];
assign P1 = P[1];

// Generate and Propagate for second stage
wire G2, G3, P2, P3;
assign G2 = G[2] | (G0 & P[1]);
assign G3 = G[3] | (G1 & P1);
assign P2 = P[2] & P0;
assign P3 = P[3] & P1;

// Generate and Propagate for third stage
wire G4, G5, P4, P5;
assign G4 = G[4] | (G2 & P[3]);
assign G5 = G[5] | (G3 & P3);
assign P4 = P[4] & P2;
assign P5 = P[5] & P3;

// Carry Out Calculation
assign C = G[7] | (G4 & P[5]);
assign S = P ^ G[6:0];

endmodule

module adder_32bit (
    input [31:0] A,
    input [31:0] B,
    output [31:0] S,
    output C32
);

wire [15:0] P1, G1, P2, G2, P3, G3, P4, G4;
wire [15:0] C1, C2, C4;
wire [15:0] C;

// Instantiate 16-bit CLA blocks
cla_16bit CLA1(.A(A[15:0]), .B(B[15:0]), .S(S[15:0]), .C(C1));
cla_16bit CLA2(.A(A[31:16]), .B(B[31:16]), .S(S[31:16]), .C(C2));
cla_16bit CLA3(.A(S[15:0]), .B(C[15:0]), .S(P1), .C(G1));
cla_16bit CLA4(.A(S[31:16]), .B(C[31:16]), .S(P2), .C(G2));
cla_16bit CLA5(.A(P1), .B(G1), .S(P3), .C(G3));
cla_16bit CLA6(.A(P2), .B(G2), .S(S), .C(C4));
cla_16bit CLA7(.A(P3), .B(G3), .S(C[15:0]), .C(G4));

// Generate and Propagate for the final carry
assign C32 = G4 | (C4 & G4);

endmodule
