module adder_32bit (
  input [31:0] A,
  input [31:0] B,
  output reg [31:0] S,
  output reg C32
);

  wire [31:0] P, G, C;

  CLA_16bit cla1 (.A(A[15:0]), .B(B[15:0]), .P(P[15:0]), .G(G[15:0]), .C(C[15:0]));
  CLA_16bit cla2 (.A(A[31:16]), .B(B[31:16]), .P(P[31:16]), .G(G[31:16]), .C(C[31:16]));

  always @* begin
    S[0] = A[0] ^ B[0] ^ C[0];
    C[0] = (A[0] & B[0]) | (A[0] & C[0]) | (B[0] & C[0]);

    for (int i=1; i<32; i=i+1) begin
      S[i] = A[i] ^ B[i] ^ C[i];
      C[i] = (A[i] & B[i]) | (A[i] & C[i]) | (B[i] & C[i]) | (P[i-1] & G[i-1]);
    end

    C32 = C[31];
  end

endmodule

module CLA_16bit (
  input [15:0] A,
  input [15:0] B,
  output reg [15:0] P,
  output reg [15:0] G,
  output reg [15:0] C
);

  always @* begin
    P[0] = A[0] ^ B[0];
    G[0] = A[0] & B[0];
    C[0] = A[0] | B[0];

    for (int i=1; i<16; i=i+1) begin
      P[i] = A[i] ^ B[i] ^ C[i-1];
      G[i] = (A[i] & B[i]) | (A[i] & C[i-1]) | (B[i] & C[i-1]);
      C[i] = A[i] | B[i] | C[i-1];
    end
  end

endmodule
