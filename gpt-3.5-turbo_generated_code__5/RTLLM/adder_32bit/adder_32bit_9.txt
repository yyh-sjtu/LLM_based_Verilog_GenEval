module cla_16bit (
    input [15:0] A,
    input [15:0] B,
    input generate,
    input propagate,
    output [15:0] S,
    output carry
);

assign generate = A & B;
assign propagate = A | B;
assign S = A ^ B;
assign carry = generate | (propagate & A);

endmodule

module adder_32bit (
    input [31:0] A,
    input [31:0] B,
    output [31:0] S,
    output C32
);

wire [15:0] generate;
wire [15:0] propagate;
wire [15:0] S_intermediate;
wire [16:0] carry;

cla_16bit cla0 (.A(A[15:0]), .B(B[15:0]), .generate(generate[0]), .propagate(propagate[0]), .S(S_intermediate[15:0]), .carry(carry[0]));
cla_16bit cla1 (.A(A[31:16]), .B(B[31:16]), .generate(generate[1]), .propagate(propagate[1]), .S(S_intermediate[31:16]), .carry(carry[1]));

assign S = {S_intermediate[15:0], S_intermediate[31:16]};
assign carry[16] = generate[1] | (propagate[1] & generate[0]);
assign C32 = carry[16];

endmodule
