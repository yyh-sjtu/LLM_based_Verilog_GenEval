module edge_detect (
    input clk,
    input rst_n,
    input a,
    output reg rise,
    output reg down
);

reg a_prev;

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        rise <= 0;
        down <= 0;
        a_prev <= 0;
    end else begin
        if (a && !a_prev) begin // Rising edge detected
            rise <= 1;
            down <= 0;
        end else if (!a && a_prev) begin // Falling edge detected
            rise <= 0;
            down <= 1;
        end else begin
            rise <= 0;
            down <= 0;
        end
        a_prev <= a; // Save current signal state for comparison next time
    end
end

endmodule
