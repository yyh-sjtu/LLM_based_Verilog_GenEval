Here is the complete Verilog code for the implementation of the ALU module:

```verilog
module alu (
  input [31:0] a,
  input [31:0] b,
  input [5:0] aluc,
  output reg [31:0] r,
  output reg zero,
  output reg carry,
  output reg negative,
  output reg overflow,
  output reg flag
);

  localparam ADD = 6'b100000;
  localparam ADDU = 6'b100001;
  localparam SUB = 6'b100010;
  localparam SUBU = 6'b100011;
  localparam AND = 6'b100100;
  localparam OR = 6'b100101;
  localparam XOR = 6'b100110;
  localparam NOR = 6'b100111;
  localparam SLT = 6'b101010;
  localparam SLTU = 6'b101011;
  localparam SLL = 6'b000000;
  localparam SRL = 6'b000010;
  localparam SRA = 6'b000011;
  localparam SLLV = 6'b000100;
  localparam SRLV = 6'b000110;
  localparam SRAV = 6'b000111;
  localparam LUI = 6'b001111;

  wire [31:0] res;

  always @(*) begin
    case (aluc)
      ADD: res = $signed(a) + $signed(b);
      ADDU: res = a + b;
      SUB: res = $signed(a) - $signed(b);
      SUBU: res = a - b;
      AND: res = a & b;
      OR: res = a | b;
      XOR: res = a ^ b;
      NOR: res = ~(a | b);
      SLT: begin
        res = ($signed(a) < $signed(b)) ? 32'b1 : 32'b0;
        flag = res[0];
      end
      SLTU: begin
        res = (a < b) ? 32'b1 : 32'b0;
        flag = res[0];
      end
      SLL: res = a << (b & 5'b11111);
      SRL: res = a >> (b & 5'b11111);
      SRA: res = $signed(a) >>> (b & 5'b11111);
      SLLV: res = a << (b & 5'b11111);
      SRLV: res = a >> (b & 5'b11111);
      SRAV: res = $signed(a) >>> (b & 5'b11111);
      LUI: res = {a[15:0], 16'b0};
      default: res = 32'bz;
    endcase

    r = res[31:0];
    zero = (res == 32'b0) ? 1'b1 : 1'b0;
    carry = (res[31] & b[31] & ~a[31]) ? 1'b1 : 1'b0;
    negative = res[31];
    overflow = (a[31] == b[31] && res[31] != a[31]) ? 1'b1 : 1'b0;
  end

endmodule
```

Note: This code assumes that you have a proper testbench and a top-level module that connects the ALU module to the rest of the MIPS-ISA CPU design.
