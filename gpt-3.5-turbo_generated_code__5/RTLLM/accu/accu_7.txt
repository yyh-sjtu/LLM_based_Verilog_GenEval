reg [2:0] count;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        count <= 3'b0;
        data_accum <= 8'b0;

        data_out <= 10'b0;

            valid_out <= 1'b1;
            data_out <= {2'b0, data_accum};

            valid_out <= 1'b0;
        end
    end
end

endmodule
