module adder_pipe_64bit (
    input clk,
    input rst_n,
    input i_en,
    input [63:0] adda,
    input [63:0] addb,
    output reg [64:0] result,
    output reg o_en
);

reg [63:0] a_reg1, b_reg1;
reg [64:0] sum_reg1;
reg [1:0] carry_reg1;

reg [63:0] a_reg2, b_reg2;
reg [64:0] sum_reg2;
reg [1:0] carry_reg2;

reg [63:0] a_reg3, b_reg3;
reg [64:0] sum_reg3;
reg [1:0] carry_reg3;

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        a_reg1 <= 64'h0;
        b_reg1 <= 64'h0;
        sum_reg1 <= 65'h0;
        carry_reg1 <= 2'h0;

        a_reg2 <= 64'h0;
        b_reg2 <= 64'h0;
        sum_reg2 <= 65'h0;
        carry_reg2 <= 2'h0;

        a_reg3 <= 64'h0;
        b_reg3 <= 64'h0;
        sum_reg3 <= 65'h0;
        carry_reg3 <= 2'h0;
    end else begin
        if (i_en) begin
            a_reg1 <= adda;
            b_reg1 <= addb;
            {sum_reg1, carry_reg1} <= a_reg1 + b_reg1;
        end

        a_reg2 <= a_reg1;
        b_reg2 <= b_reg1;
        {sum_reg2, carry_reg2} <= {1'b0, sum_reg1} + {1'b0, carry_reg1};

        a_reg3 <= a_reg2;
        b_reg3 <= b_reg2;
        {sum_reg3, carry_reg3} <= {1'b0, sum_reg2} + {1'b0, carry_reg2};
    end
end

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        result <= 65'h0;
        o_en <= 1'b0;
    end else begin
        if (i_en) begin
            result <= sum_reg3;
            o_en <= 1'b1;
        end else begin
            result <= 65'h0;
            o_en <= 1'b0;
        end
    end
end

endmodule
