module adder_pipe_64bit (
    input              clk,
    input              rst_n,
    input              i_en,
    input      [63:0]  adda,
    input      [63:0]  addb,
    output     [64:0]  result,
    output             o_en
);

reg        [63:0]  stage1_out, stage2_out, stage3_out, stage4_out;
reg                 stage1_en, stage2_en, stage3_en, stage4_en;
wire       [64:0]  final_result;
wire        carry;

assign result = final_result[64:0];
assign o_en   = stage4_en;

// Stage 1
// Synchronous adder using carry signal
always @ (posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        stage1_out <= 64'b0;
        stage1_en  <= 1'b0;
    end
    else begin
        if (i_en) begin
            stage1_out <= adda + addb;
            stage1_en  <= 1'b1;
        end
    end
end

// Stage 2
always @ (posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        stage2_out <= 64'b0;
        stage2_en  <= 1'b0;
    end
    else begin
        if (stage1_en) begin
            stage2_out <= stage1_out;
            stage2_en  <= 1'b1;
        end
    end
end

// Stage 3
always @ (posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        stage3_out <= 64'b0;
        stage3_en  <= 1'b0;
    end
    else begin
        if (stage2_en) begin
            stage3_out <= stage2_out;
            stage3_en  <= 1'b1;
        end
    end
end

// Stage 4
always @ (posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        stage4_out <= 64'b0;
        stage4_en  <= 1'b0;
    end
    else begin
        if (stage3_en) begin
            stage4_out <= stage3_out;
            stage4_en  <= 1'b1;
        end
    end
end

// Final result calculation
assign final_result = {1'b0, stage4_out} + {1'b0, carry};

// Carry generation
assign carry = (stage4_out[63] & adda[63]) | (stage4_out[63] & addb[63]) | (adda[63] & addb[63]);

endmodule
