Here's the complete Verilog code for the 64-bit ripple carry adder with pipeline stages:

```verilog
module adder_pipe_64bit (
  input clk,
  input rst_n,
  input i_en,
  input [63:0] adda,
  input [63:0] addb,
  output reg [64:0] result,
  output reg o_en
);

reg [63:0] a_reg, b_reg;
reg [64:0] sum_reg;
reg [63:0] carry;

always @(posedge clk or negedge rst_n) begin
  if (~rst_n) begin
    a_reg <= 64'b0;
    b_reg <= 64'b0;
    sum_reg <= 65'b0;
    carry <= 64'b0;
    o_en <= 0;
  end else begin
    if (i_en) begin
      a_reg <= adda;
      b_reg <= addb;
      sum_reg <= {1'b0, a_reg} + {1'b0, b_reg} + {1'b0, carry};
      carry <= sum_reg[64];
      o_en <= 1;
    end
  end
end

assign result = sum_reg[64:0];

endmodule
```
