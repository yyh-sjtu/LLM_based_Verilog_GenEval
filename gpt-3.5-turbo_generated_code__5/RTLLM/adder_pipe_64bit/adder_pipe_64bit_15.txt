module adder_pipe_64bit (
    input clk,
    input rst_n,
    input i_en,
    input [63:0] adda,
    input [63:0] addb,
    output reg [64:0] result,
    output reg o_en
);

reg [63:0] a_reg, b_reg, sum_reg;
reg [64:0] result_reg;
reg [6:0] stage;

always @ (posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        a_reg <= 64'b0;
        b_reg <= 64'b0;
        sum_reg <= 64'b0;
        result_reg <= 65'b0;
        stage <= 7'b0;
        o_en <= 1'b0;
    end else begin
        if (i_en) begin
            case (stage)
                7'b0000000: begin
                    a_reg <= adda;
                    b_reg <= addb;
                    sum_reg <= a_reg + b_reg;
                    stage <= 7'b0000001;
                end
                7'b0000001: begin
                    a_reg <= sum_reg[63:0];
                    b_reg <= sum_reg[63:0];
                    sum_reg <= a_reg + b_reg;
                    stage <= 7'b0000010;
                end
                // continue adding pipeline stages as needed
                // ...
                7'b1000000: begin
                    result_reg <= {1'b0, sum_reg};
                    o_en <= 1'b1;
                    stage <= 7'b0000000;
                end
            endcase
        end
    end
end

assign result = result_reg;
assign o_en = o_en;

endmodule
