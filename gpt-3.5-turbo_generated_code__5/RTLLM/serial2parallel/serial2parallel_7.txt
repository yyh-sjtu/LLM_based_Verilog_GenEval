// Verilog code for series-parallel conversion circuit

module serial2parallel(
    input clk,         // Clock signal
    input rst_n,       // Reset signal (active low)
    input din_serial,  // Serial input data
    input din_valid,   // Validity signal for input data
    output reg [7:0] dout_parallel,  // Parallel output data (8 bits wide)
    output reg dout_valid              // Validity signal for the output data
);

// Internal signals
reg [3:0] cnt;  // Counter to keep track of the number of serial input data bits received

// Synchronous design using positive edge-triggered clock
always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        // Reset values
        cnt <= 4'b0000;
        dout_parallel <= 8'b00000000;
        dout_valid <= 1'b0;
    end
    else begin
        // Check for valid input data and increment counter
        if (din_valid) begin
            cnt <= cnt + 1;
            
            // Shift in serial input data to form parallel output
            dout_parallel <= {dout_parallel[6:0], din_serial};
        end
        
        // Output valid signal when 8 serial data bits are received
        if (cnt == 8) begin
            dout_valid <= 1'b1;
        end
        else begin
            dout_valid <= 1'b0;
        end
    end
end

endmodule
