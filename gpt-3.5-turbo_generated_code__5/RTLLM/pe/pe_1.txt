module pe (
  input wire clk,
  input wire rst,
  input wire [31:0] a,
  input wire [31:0] b,
  output reg [31:0] c
);

  always @(posedge clk or posedge rst) begin
    if (rst) begin
      c <= 0; // Reset the accumulator register
    end else begin
      c <= c + (a * b); // Accumulate the result of a multiplied by b
    end
  end

endmodule
