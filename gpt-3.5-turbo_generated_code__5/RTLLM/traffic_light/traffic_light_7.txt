```verilog
module traffic_light (
    input rst_n, // Reset signal (active low)
    input clk,   // Clock signal
    input pass_request, // Request signal for allowing vehicles to pass
    
    output reg [7:0] clock, // 8-bit output representing the count value of the internal counter
    output reg red,        // Output signal for the red light
    output reg yellow,     // Output signal for the yellow light
    output reg green       // Output signal for the green light
);

    // Enumeration values for different states
    parameter idle = 2'b00;
    parameter s1_red = 2'b01;
    parameter s2_yellow = 2'b10;
    parameter s3_green = 2'b11;
    
    // Internal registers for state and next values of the lights
    reg [1:0] state, next_state;
    reg p_red, p_yellow, p_green;
    
    // Internal counter for timing
    reg [7:0] cnt;
    
    // State transition and control logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= idle;
        end else begin
            state <= next_state;
        end
    end
    
    always @* begin
        case(state)
            idle: begin
                next_state = s1_red;
            end
            s1_red: begin
                next_state = (cnt == 3) ? s3_green : s1_red;
            end
            s2_yellow: begin
                next_state = (cnt == 3) ? s1_red : s2_yellow;
            end
            s3_green: begin
                next_state = (cnt == 3) ? s2_yellow : s3_green;
            end
            default: begin
                next_state = idle;
            end
        endcase
    end
    
    // Counting logic for the internal counter
    always @(posedge clk, negedge rst_n) begin
        if (!rst_n) begin
            cnt <= 10;
        end else if (pass_request && green) begin
            cnt <= 10;
        end else if (!green && p_green) begin
            cnt <= 60;
        end else if (!yellow && p_yellow) begin
            cnt <= 5;
        end else if (!red && p_red) begin
            cnt <= 10;
        end else begin
            cnt <= cnt - 1;
        end
    end
    
    // Assignment of internal counter value to output clock
    assign clock = cnt;
    
    // Output control logic for the lights
    always @(posedge clk, negedge rst_n) begin
        if (!rst_n) begin
            red <= 0;
            yellow <= 0;
            green <= 0;
        end else begin
            p_red <= red;
            p_yellow <= yellow;
            p_green <= green;
            
            case(state)
                s1_red: begin
                    red <= 1;
                    yellow <= 0;
                    green <= 0;
                end
                s2_yellow: begin
                    red <= 0;
                    yellow <= 1;
                    green <= 0;
                end
                s3_green: begin
                    red <= 0;
                    yellow <= 0;
                    green <= 1;
                end
                default: begin
                    red <= 0;
                    yellow <= 0;
                    green <= 0;
                end
            endcase
        end
    end

endmodule
```
