```verilog
module traffic_light(
    input rst_n,
    input clk,
    input pass_request,
    output reg [7:0] clock,
    output reg red,
    output reg yellow,
    output reg green
);

    typedef enum logic [1:0] {idle, s1_red, s2_yellow, s3_green} states;
    reg [1:0] state;
    
    reg [7:0] cnt;
    reg p_red, p_yellow, p_green;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= idle;
            cnt <= 10;
            p_red <= 0;
            p_yellow <= 0;
            p_green <= 0;
        end else begin
            case (state)
                idle: begin
                    state <= s1_red;
                    cnt <= 0;
                end
                s1_red: begin
                    if (cnt >= 3) begin
                        state <= s3_green;
                        cnt <= 0;
                    end
                    else begin
                        state <= s1_red;
                        cnt <= cnt + 1;
                    end
                end
                s2_yellow: begin
                    if (cnt >= 3) begin
                        state <= s1_red;
                        cnt <= 0;
                    end
                    else begin
                        state <= s2_yellow;
                        cnt <= cnt + 1;
                    end
                end
                s3_green: begin
                    if (cnt >= 3) begin
                        state <= s2_yellow;
                        cnt <= 0;
                    end
                    else begin
                        state <= s3_green;
                        cnt <= cnt + 1;
                    end
                end
            endcase
        end
    end
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cnt <= 10;
        end else begin
            if ((pass_request && p_green) || (!cnt && p_green)) begin
                cnt <= 10;
            end else if (!cnt && p_green) begin
                cnt <= 60;
            end else if (!cnt && p_yellow) begin
                cnt <= 5;
            end else if (!cnt && p_red) begin
                cnt <= 10;
            end else begin
                cnt <= cnt - 1;
            end
        end
    end
    
    assign clock = cnt;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            red <= 0;
            yellow <= 0;
            green <= 0;
        end else begin
            case (state)
                idle: begin
                    red <= 0;
                    yellow <= 0;
                    green <= 0;
                end
                s1_red: begin
                    red <= 1;
                    yellow <= 0;
                    green <= 0;
                end
                s2_yellow: begin
                    red <= 0;
                    yellow <= 1;
                    green <= 0;
                end
                s3_green: begin
                    red <= 0;
                    yellow <= 0;
                    green <= 1;
                end
            endcase
            
            p_red <= red;
            p_yellow <= yellow;
            p_green <= green;
        end
    end

endmodule
```

Please note that this Verilog code assumes that the input signals `rst_n`, `clk`, and `pass_request` are provided from an external source. You would need to instantiate this `traffic_light` module in a top-level module and connect the input and output signals accordingly.
