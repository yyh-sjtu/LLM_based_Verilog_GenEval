module dual_port_RAM
    #(parameter DEPTH = 16,
      parameter WIDTH = 8)
    (input wclk, input wenc,
     input [$clog2(DEPTH)-1:0] waddr,
     input [WIDTH-1:0] wdata,
     input rclk, input renc,
     input [$clog2(DEPTH)-1:0] raddr,
     output reg [WIDTH-1:0] rdata);
  
  reg [WIDTH-1:0] RAM_MEM [0:DEPTH-1];

  always @(posedge wclk)
    if(wenc) RAM_MEM[waddr] <= wdata;
  
  always @(posedge rclk)
    if(renc) rdata <= RAM_MEM[raddr];
endmodule

module asyn_fifo
    #(parameter WIDTH = 8,
      parameter DEPTH = 16)
    (input wclk, input rclk,
     input wrstn, input rrstn,
     input winc, input rinc,
     input [WIDTH-1:0] wdata,
     output wfull, output rempty,
     output reg [WIDTH-1:0] rdata);
  
  reg [$clog2(DEPTH)-1:0] waddr = 0, raddr = 0;
  reg [WIDTH-1:0] wptr = 0, rptr = 0;
  reg [WIDTH-1:0] wptr_buff = 0, rptr_buff = 0;

  assign wfull = (wptr == {~rptr[($clog2(DEPTH)-1):($clog2(DEPTH)-1)], rptr[($clog2(DEPTH)-2):0]});
  assign rempty = (rptr == wptr);

  dual_port_RAM #(DEPTH, WIDTH) ram_inst(.wclk(wclk), .wenc(winc), .waddr(waddr), .wdata(wdata),
                                         .rclk(rclk), .renc(rinc), .raddr(raddr), .rdata(rdata));

  always @(posedge wclk)
    if(wrstn == 1'b0)
      waddr <= 0;
    else if(winc)
      waddr <= waddr + 1;
  
  always @(posedge rclk)
    if(rrstn == 1'b0)
      raddr <= 0;
    else if(rinc)
      raddr <= raddr + 1;

  always @(posedge wclk)
    if(wrstn == 1'b0)
      wptr <= 0;
    else
      wptr <= wptr ^ (wptr >> 1);
  
  always @(posedge rclk)
    if(rrstn == 1'b0)
      rptr <= 0;
    else
      rptr <= rptr ^ (rptr >> 1);

  always @(posedge wclk)
    if(wrstn == 1'b0)
      wptr_buff <= 0;
    else
      wptr_buff <= wptr;
  
  always @(posedge rclk)
    if(rrstn == 1'b0)
      rptr_buff <= 0;
    else
      rptr_buff <= rptr;

endmodule
