module dual_port_RAM (
    input wclk,
    input wenc,
    input [$clog2(DEPTH)-1:0] waddr,
    input [WIDTH-1:0] wdata,
    input rclk,
    input renc,
    input [$clog2(DEPTH)-1:0] raddr,
    output reg [WIDTH-1:0] rdata
);

reg [DEPTH-1:0][WIDTH-1:0] RAM_MEM;

always @(posedge wclk)
begin
    if(wenc)
        RAM_MEM[waddr] <= wdata;
end

always @(posedge rclk)
begin
    if(renc)
        rdata <= RAM_MEM[raddr];
end

endmodule

module asyn_fifo (
    input wclk,
    input rclk,
    input wrstn,
    input rrstn,
    input winc,
    input rinc,
    input [WIDTH-1:0] wdata,
    output reg wfull,
    output reg rempty,
    output reg [WIDTH-1:0] rdata
);

parameter WIDTH = 8;
parameter DEPTH = 16;

reg [WIDTH-1:0] wptr, wptr_buff;
reg [WIDTH-1:0] rptr, rptr_buff;
reg [WIDTH-1:0] raddr, waddr;
reg wenc, renc;

// Dual-port RAM instantiation
dual_port_RAM RAM_inst (
    .wclk(wclk),
    .wenc(wenc),
    .waddr(waddr),
    .wdata(wdata),
    .rclk(rclk),
    .renc(renc),
    .raddr(raddr),
    .rdata(rdata)
);

always @(posedge wclk or negedge wrstn)
begin
    if(~wrstn)
    begin
        wptr <= 0;
        wptr_buff <= 0;
        waddr <= 0;
        wenc <= 1;
    end
    else if(winc)
    begin
        wptr <= wptr + 1;
        waddr <= wptr[DEPTH-1:$clog2(DEPTH)];
    end
end

always @(posedge rclk or negedge rrstn)
begin
    if(~rrstn)
    begin
        rptr <= 0;
        rptr_buff <= 0;
        raddr <= 0;
        renc <= 1;
    end
    else if (rinc)
    begin
        rptr <= rptr + 1;
        raddr <= rptr[DEPTH-1:$clog2(DEPTH)];
    end
end

always @(posedge wclk)
begin
    wptr_buff <= wptr;
end

always @(posedge rclk)
begin
    rptr_buff <= rptr;
end

always @(*)
begin
    if ((wptr == rptr) && (wptr_buff == rptr_buff))
        rempty = 1;
    else
        rempty = 0;

    if ((wptr + 1 == rptr) || ((wptr[WIDTH-1] ^ rptr[WIDTH-1]) & 
        ((wptr[WIDTH-2:0] == rptr[WIDTH-2:0]))))
        wfull = 1;
    else
        wfull = 0;
end

endmodule
