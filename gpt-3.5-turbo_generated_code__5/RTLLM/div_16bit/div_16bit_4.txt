module div_16bit (
    input [15:0] A,
    input [7:0] B,
    output reg [15:0] result,
    output reg [15:0] odd
);

reg [15:0] a_reg;
reg [7:0] b_reg;

always @(*) begin
    a_reg = A;
    b_reg = B;
end

always @(*) begin
    reg [15:0] rem = a_reg;
    reg [15:0] quotient;
    
    for (int i = 0; i < 8; i = i + 1) begin
        if (rem >= (b_reg << (15 - i))) begin
            quotient[i] = 1;
            rem = rem - (b_reg << (15 - i));
        end else begin
            quotient[i] = 0;
        end
    end

    result = quotient;
    odd = rem;
end

endmodule
