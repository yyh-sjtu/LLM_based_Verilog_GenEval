module div_16bit (
    input [15:0] A,
    input [7:0] B,
    output reg [15:0] result,
    output reg [15:0] odd
);

reg [15:0] a_reg;
reg [7:0] b_reg;

always @(*)
begin
    a_reg = A;
    b_reg = B;
end

always @(*)
begin
    reg [15:0] temp_a;
    reg [15:0] quotient;
    reg [15:0] remainder;
    reg [3:0] shift_count;
    reg [7:0] divisor;

    temp_a = a_reg;
    divisor = b_reg;
    quotient = 0;
    shift_count = 0;

    while(shift_count < 16)
    begin
        remainder = temp_a[15] ? 1 : 0;
        temp_a = temp_a << 1;
        remainder = remainder | (temp_a[15] ? 1 : 0);
        if (remainder >= divisor)
        begin
            remainder = remainder - divisor;
            quotient = quotient | (1 << (15 - shift_count));
        end
        shift_count = shift_count + 1;
    end

    result = quotient;
    odd = remainder;
end

endmodule
