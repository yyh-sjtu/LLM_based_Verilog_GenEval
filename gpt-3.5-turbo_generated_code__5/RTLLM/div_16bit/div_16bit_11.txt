module div_16bit(
    input [15:0] A,
    input [7:0] B,
    output reg [15:0] result,
    output reg [15:0] odd
);

reg [15:0] a_reg;
reg [7:0] b_reg;
reg [3:0] count;

always @(*) begin
    a_reg = A;
    b_reg = B;
end

always @(*) begin
    result = 16'b0;
    odd = a_reg;

    for (count = 0; count < 16; count = count + 1) begin
        if (odd[15] == 1'b1) begin
            if (odd[15:8] >= b_reg) begin
                result[count] = 1'b1;
                odd[15:8] = odd[15:8] - b_reg;
            end
            odd = {odd[14:0], 1'b0};
        end
        else begin
            result[count] = 1'b0;
            odd = {odd[14:0], 0};
        end
    end

end

endmodule
