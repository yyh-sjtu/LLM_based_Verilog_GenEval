            end else if (left == 1 && center == 0 && right == 1) begin
                next_q[i] = 1;
            end else if (left == 1 && center == 0 && right == 0) begin
                next_q[i] = 0;
            end else if (left == 0 && center == 1 && right == 1) begin
                next_q[i] = 1;
            end else if (left == 0 && center == 1 && right == 0) begin
