module TopModule(
    input [5:0] y,

        case(y)

                Y1 <= w;
                Y3 <= 0;
            end

                Y1 <= w;
                Y3 <= 0;
            end

                Y1 <= 0;
                Y3 <= w;
            end
            6'b100000: begin // State F
                Y1 <= 0;
                Y3 <= 0;
            end
        endcase
    end

endmodule
