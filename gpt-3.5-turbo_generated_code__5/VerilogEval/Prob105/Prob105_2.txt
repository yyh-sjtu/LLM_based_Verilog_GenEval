module TopModule (

  input [99:0] data,

      q <= data;
    end else if (ena == 2'b01) begin
      q <= {q[0], q[99:1]};
    end else if (ena == 2'b10) begin
      q <= {q[98:0], q[99]};
    end
  end

endmodule
