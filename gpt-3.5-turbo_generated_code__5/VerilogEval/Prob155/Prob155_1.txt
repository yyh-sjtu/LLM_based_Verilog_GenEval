```verilog
module TopModule (
  input  clk,
  input  areset,
  input  bump_left,
  input  bump_right,
  input  ground,
  input  dig,
  output reg walk_left,
  output reg walk_right,
  output reg aaah,
  output reg digging
);

  // Define the states of the Lemming FSM
  typedef enum logic [1:0] {
    WALK_LEFT,
    WALK_RIGHT,
    FALL,
    DIG,
    SPLATTER
  } StateType;

  // Define the current state and next state signals
  reg [1:0] curr_state, next_state;

  // Define a counter to track the falling duration
  reg [4:0] fall_counter;

  // Define the initial state and current state assignment
  always @(posedge clk or posedge areset) begin
    if (areset) begin
      curr_state <= WALK_LEFT;
    end else begin
      curr_state <= next_state;
    end
  end

  // Define the next state logic
  always @(curr_state, bump_left, bump_right, ground, dig) begin
    case (curr_state)
      WALK_LEFT:
        if (bump_right) begin
          next_state <= WALK_RIGHT;
        end else if (dig && ground) begin
          next_state <= DIG;
        end else if (!ground) begin
          next_state <= FALL;
        end else begin
          next_state <= WALK_LEFT;
        end
      WALK_RIGHT:
        if (bump_left) begin
          next_state <= WALK_LEFT;
        end else if (dig && ground) begin
          next_state <= DIG;
        end else if (!ground) begin
          next_state <= FALL;
        end else begin
          next_state <= WALK_RIGHT;
        end
      FALL:
        if (fall_counter >= 20) begin
          next_state <= SPLATTER;
        end else if (ground) begin
          next_state <= curr_state;
        end else begin
          next_state <= FALL;
        end
      DIG:
        if (!ground) begin
          next_state <= FALL;
        end else if (bump_left || bump_right) begin
          next_state <= curr_state;
        end else begin
          next_state <= DIG;
        end
      SPLATTER:
        next_state <= SPLATTER;
    endcase
  end

  // Define the output logic
  always @(curr_state, fall_counter, ground) begin
    case (curr_state)
      WALK_LEFT:
        walk_left  <= 1;
        walk_right <= 0;
        aaah       <= 0;
        digging    <= 0;
      WALK_RIGHT:
        walk_left  <= 0;
        walk_right <= 1;
        aaah       <= 0;
        digging    <= 0;
      FALL:
        walk_left  <= 0;
        walk_right <= 0;
        aaah       <= 1;
        digging    <= 0;
      DIG:
        walk_left  <= 0;
        walk_right <= 0;
        aaah       <= 0;
        digging    <= 1;
      SPLATTER:
        walk_left  <= 0;
        walk_right <= 0;
        aaah       <= 0;
        digging    <= 0;
    endcase
  end

  // Define the falling counter
  always @(posedge clk or posedge areset) begin
    if (areset) begin
      fall_counter <= 0;
    end else if (curr_state == FALL) begin
      fall_counter <= fall_counter + 1;
    end else if (ground) begin
      fall_counter <= 0;
    end
  end

endmodule
```

This implementation uses a Moore state machine to model the behavior of the Lemming. The states are defined using an enum and the current state and next state are updated on the positive edge of the clock based on the input signals (bump_left, bump_right, ground, dig). The outputs (walk_left, walk_right, aaah, digging) are assigned based on the current state.

The falling duration is tracked using a counter, which increments when the current state is FALL and resets to 0 when the ground reappears.

The asynchronous reset (areset) is used to initialize the Lemming to the WALK_LEFT state.
