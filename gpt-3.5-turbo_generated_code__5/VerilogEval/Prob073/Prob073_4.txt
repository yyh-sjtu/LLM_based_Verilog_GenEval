                q_reg <= (others => '0');
            else
                q_reg <= d_reg;
            end if;

    process (clk, resetn)
    begin
        if resetn = '0' then
            clear_q <= '1';
        elsif rising_edge(clk) then
            if clear_q = '1' then
                clear_q <= '0';
            else
                clear_q <= '1';
            end if;
        end if;
