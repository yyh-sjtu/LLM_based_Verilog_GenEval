module flipflop (
  input clk,
  input D,
  input reset,
  output reg Q
);

  always @(posedge clk or posedge reset) begin
    if (reset) begin
      Q <= 0;
    end else begin
      Q <= D;
    end
  end

endmodule


module mux_2to1 (
  input S,
  input D0,
  input D1,
  output Y
);

  assign Y = (S == 0) ? D0 : D1;

endmodule


module TopModule (

  );

    .Q(q[1])

    .D(q[1]),
    .reset(L),
    .Q(q[2])

    .L(L),
    .clk(clk),
    .q(q)
  );

endmodule
