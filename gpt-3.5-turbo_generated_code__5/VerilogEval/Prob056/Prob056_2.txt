```vhdl
library ieee;
use ieee.std_logic_1164.all;

entity TopModule is
    port (
        clk : in std_logic;
        j   : in std_logic;
        k   : in std_logic;
        Q   : out std_logic
    );
end entity;

architecture rtl of TopModule is
    signal Qold : std_logic;
begin
    process (clk)
    begin
        if rising_edge(clk) then
            if (j = '0' and k = '0') then
                Q <= Qold;
            elsif (j = '0' and k = '1') then
                Q <= '0';
            elsif (j = '1' and k = '0') then
                Q <= '1';
            elsif (j = '1' and k = '1') then
                Q <= not Qold;
            end if;
        end if;
    end process;
end architecture;
```

Inside the architecture, a process is used to describe the behavior of the flip-flop based on the positive clock edge. The process checks the values of J and K inputs and assigns the appropriate value to the output Q based on the truth table provided.
