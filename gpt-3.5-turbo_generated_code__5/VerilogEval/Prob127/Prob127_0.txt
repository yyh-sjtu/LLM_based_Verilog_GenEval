      current_state <= WALK_LEFT;
    end else begin
      // Transition to next state based on inputs
      current_state <= next_state;

        end else begin
          next_state <= WALK_LEFT;
        end
      WALK_RIGHT:
        if (bump_left) begin
          next_state <= WALK_LEFT;
        end else begin
          next_state <= WALK_RIGHT;
        end
    endcase
  end

  // Output assignment

          walk_right <= 1'b0;
        end

          walk_left <= 1'b0;

        end
    endcase
  end
