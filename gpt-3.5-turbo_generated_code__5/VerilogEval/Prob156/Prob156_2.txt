```verilog
module TopModule (
    input clk,
    input reset,
    input data,
    output reg [3:0] count,
    output reg counting,
    output reg done,
    input ack
);

enum [2:0] states = {IDLE, DETECT_START, READ_DELAY, COUNTING, WAIT_FOR_ACK};
reg [2:0] state, next_state;
reg [3:0] delay_counter;
reg [3:0] remaining_time;

always @(posedge clk) begin
    if (reset) begin
        state <= IDLE;
        counting <= 0;
        done <= 0;
        delay_counter <= 0;
        remaining_time <= 0;
    end else begin
        state <= next_state;
    end
end

always @* begin
    next_state = state;
    counting = 0;
    done = 0;
    count = 4'b0;
    
    case (state)
        IDLE:
            if (data == 1 && $past(data) == 1 && $past($past(data)) == 0) begin
                next_state = DETECT_START;
            end
        DETECT_START:
            if (data == 0 && $past(data) == 1) begin
                next_state = READ_DELAY;
            end
        READ_DELAY:
            if (delay_counter < 4) begin
                delay_counter <= delay_counter + 1;
            end else begin
                next_state = COUNTING;
                remaining_time <= delay_counter;
                count <= remaining_time;
            end
        COUNTING:
            counting = 1;
            remaining_time = remaining_time - 1;
            if (remaining_time == 0) begin
                next_state = WAIT_FOR_ACK;
            end
        WAIT_FOR_ACK:
            if (ack == 1) begin
                next_state = IDLE;
                done = 1;
            end
    endcase
end

endmodule
```
