module TopModule (
  input d,
  input done_counting,
  input ack,
  input [9:0] state,
  output reg B3_next,
  output reg S_next,
  output reg S1_next,
  output reg Count_next,
  output reg Wait_next,
  output reg done,
  output reg counting,
  output reg shift_ena
);

  always @(state, d) begin
    case(state)
      10'b0000000010: begin
        if(d == 1'b0)
          S_next = 1'b1;
        else if(d == 1'b1)
          S1_next = 1'b1;
      end
      
      10'b0000000100: begin
        if(d == 1'b0)
          S_next = 1'b1;
        else if(d == 1'b1)
          S11_next = 1'b1;
      end
      
      10'b0000001100: begin
        if(d == 1'b0)
          S11_next = 1'b1;
        else if(d == 1'b1)
          S11_next = 1'b1;
      end
      
      10'b0000011000: begin
        if(d == 1'b0)
          S_next = 1'b1;
        else if(d == 1'b1)
          B0_next = 1'b1;
      end
      
      10'b1000000000: B3_next = 1'b1;
      
      default: begin
        S_next = 1'b0;
        S1_next = 1'b0;
        S11_next = 1'b0;
        B0_next = 1'b0;
        B1_next = 1'b0;
        B2_next = 1'b0;
        B3_next = 1'b0;
        Count_next = 1'b0;
        Wait_next = 1'b0;
      end
    endcase
  end

  always @(state, done_counting) begin
    case(state)
      10'b1000000011: begin
        if(!done_counting)
          Count_next = 1'b1;
      end
      
      10'b1000000011: begin
        if(done_counting)
          Wait_next = 1'b1;
      end
      
      default: begin
        Count_next = 1'b0;
        Wait_next = 1'b0;
      end
    endcase
  end

  always @(done, ack) begin
    if(done) begin
      if(!ack)
        Wait_next = 1'b1;
    end else if(!done && ack)
      S_next = 1'b1;
  end

  always @(state) begin
    case(state)
      10'b1000000011: counting = 1'b1;
      default: counting = 1'b0;
    endcase
  end

  always @(state) begin
    case(state)
      10'b1000000011: shift_ena = 1'b1;
      default: shift_ena = 1'b0;
    endcase
  end

  always @(state) begin
    case(state)
      10'b1000000100: done = 1'b1;
      default: done = 1'b0;
    endcase
  end
endmodule
