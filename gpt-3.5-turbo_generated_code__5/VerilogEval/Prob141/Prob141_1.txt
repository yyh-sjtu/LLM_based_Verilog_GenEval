module TopModule (

    input reset,
    input ena,
    output reg pm,
    output reg [7:0] hh,
    output reg [7:0] mm,
    output reg [7:0] ss
);

reg [4:0] seconds;
reg [5:0] minutes;
reg [4:0] hours;
reg pm_temp;

always @(posedge clk) begin
    if (reset) begin
        seconds <= 0;
        minutes <= 0;
        hours <= 1; // Start at 12:00 AM

endmodule
