module TopModule (
    input        sel,
    input  [7:0] a,
    input  [7:0] b,
    output       out
);

    reg [7:0] corrected_out;
    
    always @(*) begin
        if(sel)
            corrected_out = b;
        else
            corrected_out = a;
    end
    
    assign out = corrected_out;
    
endmodule
