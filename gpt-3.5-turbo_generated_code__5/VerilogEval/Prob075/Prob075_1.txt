module TopModule (

  input train_valid,

      state <= 2'b01; // Reset to weakly not-taken
    end else if (train_valid) begin

        state <= state + 1;

        state <= state - 1;
      end
    end
  end

endmodule

Otherwise, when not training (`train_valid` deasserted), the counter retains its current value.
