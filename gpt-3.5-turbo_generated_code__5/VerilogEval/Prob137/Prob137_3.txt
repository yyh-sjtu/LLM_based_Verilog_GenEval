use ieee.std_logic_1164.all;

entity TopModule is
    port (
        clk   : in  std_logic;
        reset : in  std_logic;
        in    : in  std_logic;
        done  : out std_logic
    );
end entity TopModule;

architecture rtl of TopModule is
    type state_type is (IDLE, START, DATA, STOP);
    signal state      : state_type;
    signal data_count : integer range 0 to 7;
    signal next_state : state_type;
begin
    process(clk)
    begin
        if rising_edge(clk) then
            if reset = '1' then
                state      <= IDLE;
                data_count <= 0;
                done       <= '0';
            else
                state      <= next_state;
                data_count <= data_count + 1;
                done       <= '0';

                case state is
                    when IDLE =>
                        if in = '0' then
                            next_state <= START;
                        else
                            next_state <= IDLE;
                        end if;

                    when START =>
                        if data_count = 7 then
                            next_state <= STOP;
                        else
                            next_state <= DATA;
                        end if;

                    when DATA =>
                        if data_count = 7 then
                            next_state <= STOP;
                        else
                            next_state <= DATA;
                        end if;

                    when STOP =>
                        if in = '1' then
                            next_state <= IDLE;
                            done       <= '1';
                        else
                            next_state <= STOP;
                        end if;

                    when others =>
                        next_state <= IDLE;
                end case;
            end if;
        end if;
    end process;
end architecture rtl;

This implementation uses a state machine with four states to identify when bytes have been correctly received. The state machine starts in the IDLE state and transitions to the START state when an incoming low signal is detected. It then transitions to the DATA state and increments the data_count signal for each incoming bit until it reaches the 8th bit. At this point, it transitions to the STOP state and checks if the incoming bit is high. If it is high, it considers the byte received correctly and goes back to the IDLE state.

The done signal is set high when a byte has been correctly received. Resetting the module sets all signals to their initial values and clears the done signal. Sequential logic is triggered on the rising edge of the clock.
