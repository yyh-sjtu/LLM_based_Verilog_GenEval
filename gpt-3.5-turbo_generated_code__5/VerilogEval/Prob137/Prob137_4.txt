library ieee;
use ieee.std_logic_1164.all;

entity TopModule is
    port (
        clk   : in  std_logic;
        reset : in  std_logic;
        in    : in  std_logic;
        done  : out std_logic
    );
end entity TopModule;

architecture rtl of TopModule is
    type state_type is (idle, start_bit, data_1, data_2, data_3, data_4, data_5, data_6, data_7, stop_bit);
    signal state : state_type;

begin
    process (clk)
    begin
        if rising_edge(clk) then
            if reset = '1' then
                state <= idle;
                done <= '0';
            else
                case state is
                    when idle =>
                        if in = '0' then
                            state <= start_bit;
                        end if;
                    when start_bit =>
                        if in = '0' then
                            state <= data_1;
                        else
                            state <= idle;
                        end if;
                    when data_1 =>
                        if in = '1' then
                            state <= data_2;
                        else
                            state <= idle;
                        end if;
                    -- Repeat the same pattern for data_2 to data_7 states
                    when data_7 =>
                        if in = '1' then
                            state <= stop_bit;
                        else
                            state <= idle;
                        end if;
                    when stop_bit =>
                        if in = '1' then
                            state <= idle;
                            done <= '1';
                        else
                            state <= idle;
                        end if;
                    when others =>
                        state <= idle;
                end case;
            end if;
        end if;
    end process;

end architecture rtl;

This implementation describes a synchronous finite state machine (FSM) in VHDL. The `state` signal is used to store the current state of the FSM and determine the next state based on the input signal `in`. The `done` output signal is set to '1' when a complete byte has been received correctly.

The FSM starts in the `idle` state, waits for the start bit (logic '0'), transitions through the `data_1` to `data_7` states while waiting for the 8 data bits, and finally checks for the stop bit (logic '1') in the `stop_bit` state. If the stop bit is detected, the FSM goes back to the `idle` state and sets `done` to '1'. Any invalid bit sequence or missing/extra stop bit will cause the FSM to reset to the `idle` state without setting `done` to '1'.
