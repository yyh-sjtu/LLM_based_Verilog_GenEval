  reg [2:0] count;

  always @(posedge clk) begin

      state <= 3'b000;
      consecutive_ones <= 6'b000000;

            flag <= 1'b1;

                     count == 3'b100) begin
            disc <= 1'b1;
          end else if (consecutive_ones == 6'b011111 &&
                     count >= 3'b110) begin
            err <= 1'b1;
          end else if (in == 1'b0) begin
            consecutive_ones <= 6'b000000;
            count <= 3'b000;
            state <= 3'b001;
          end else if (in == 1'b1) begin
            state <= 3'b010;
            consecutive_ones <= 6'b000000;
            count <= 3'b000;
          end
        end

            consecutive_ones <= {consecutive_ones[4:0], in};

            consecutive_ones <= 6'b000000;
