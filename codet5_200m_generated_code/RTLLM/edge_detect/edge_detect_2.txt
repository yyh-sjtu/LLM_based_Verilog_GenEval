module edge_detect (
    input clk,
    input rst_n,
    input a,
    output rise,
    output down
);
    reg [3:0] codes;
    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
            codes <= 0;
        else
            codes <= codes + 1;
    end
    assign rise = codes[0];
    assign down = codes[1];
endmodule
