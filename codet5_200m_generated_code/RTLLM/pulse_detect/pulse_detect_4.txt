module pulse_detect (
    input clk,
    input rst_n,
    input data_in,
    output data_out
);
    reg [1:0] code;
    always @(posedge clk or negedge rst_n)
    begin
        if(!rst_n)
            code <= 0;
        else
            code <= code + 1;
    end
    assign data_out = code[1];
endmodule
