module accu (
	input clk,
	input rst_n,
	input [7:0] data_in,
	input valid_in,
	output valid_out,
	output [9:0] data_out
);
    reg [7:0] data_out_reg;
    reg valid_out_reg;
    always @(posedge clk or negedge rst_n)
    begin
        if(!rst_n)
            valid_out_reg <= 1'b0;
        else
            valid_out_reg <= valid_in;
    end
    always @(posedge clk or negedge rst_n)
    begin
        if(!rst_n)
            data_out_reg <= 8'h00;
        else
            data_out_reg <= data_in;
    end
    assign valid_out = valid_out_reg;
    assign data_out = data_out_reg;
endmodule
