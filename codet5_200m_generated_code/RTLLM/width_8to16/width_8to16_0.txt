module width_8to16 (
    input clk,
    input rst_n,
    input valid_in,
    input [7:0] data_in,
    output valid_out,
    output [15:0] data_out
);
    reg [7:0] data_out_reg;
    reg valid_out_reg;
    always @(posedge clk or negedge rst_n)
    begin
        if(!rst_n)
            valid_out_reg <= 1'b0;
        else
            valid_out_reg <= valid_in;
    end
    assign data_out = data_out_reg;
    assign valid_out = valid_out_reg;
endmodule
