module radix2_div (
	input clk,
	input rst,
	input sign,
	input [7:0] dividend,
	input [7:0] divisor,
	input opn_valid,
	output reg res_valid,
	output reg [15:0] result
);
    reg [15:0] result_reg;
    always @(posedge clk) begin
        if (rst) begin
            res_valid <= 0;
            result <= 0;
        end else begin
            if (opn_valid) begin
                if (sign) begin
                    res_valid <= 1;
                    result <= result_reg;
                end
                else begin
                    res_valid <= 0;
                    result <= result_reg;
                end
            end
        end
    end
endmodule
