module parallel2serial (
    input clk,
    input rst_n,
    input [3:0] d,
    output valid_out,
    output dout
);
    reg [3:0] code;
    assign dout = code[3];
    always @(posedge clk or negedge rst_n)
        if (!rst_n)
            code <= 4'b0000;
        else
            code <= {code[3:0], d};
endmodule
