module parallel2serial (
    input clk,
    input rst_n,
    input [3:0] d,
    output valid_out,
    output dout
);
    reg [3:0] code;
    always @(posedge clk or negedge rst_n)
    begin
        if(!rst_n)
            code <= 4'b0000;
        else
            code <= code + 4'b1;
    end
    assign dout = code[3];
    assign valid_out = 1;
endmodule
