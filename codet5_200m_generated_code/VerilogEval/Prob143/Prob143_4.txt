module TopModule (
	input in,
	input [9:0] state,
	output [9:0] next_state,
	output out1,
	output out2
);
    localparam
        STATE_PIN        = 9'b11111111111111111111111111111111111, 
        STATE_IN_STATE_IN_PIN_ERROR        = 9'b0111111111111111111111111111; 
    wire [5:0] out;
    wire [5:0] in;
    assign out = (state == STATE_IN_STATE_IN_PIN_ERROR)? 4'b1111 :
                             (state == STATE_IN_STATE_IN_PIN_ERROR)? 4'b0000 :
                             (state == STATE_IN_STATE_IN_PIN_ERROR)? 4'b0000 :
                             4'b0000; 
    assign in = (state == STATE_IN_STATE_IN_PIN_ERROR)? in :
                             4'b0000; 
    assign in[1:0] = (state == STATE_IN_STATE_IN_PIN_ERROR)? in :
                             4'b0000; 
    assign out[0] = in[0]; 
    assign out[1] = in[1]; 
    assign out[2] = in[2]; 
    assign out[3] = in[3]; 
    assign out[4] = in[4]; 
    assign out[5] = in[5]; 
    assign out[6] = in[6]; 
    assign out[7] = in[7]; 
    assign out[8] = in[8]; 
    assign out[9] = in[9]; 
endmodule
