module TopModule (
	input in,
	input [9:0] state,
	output [9:0] next_state,
	output out1,
	output out2
);
    parameter A_ONEHOLD    = 0;
    parameter A_ONEHOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD = 1;
    assign out1 = (in)? 1'b1 : 1'b0;
    assign out2 = (in)? 1'b1 : 1'b0;
    reg [9:0] codes;
    initial begin
        codes = 0;
    end
    assign next_state = codes;
    always @(state or in) begin
        case (state)
            A_ONEHOLD: begin
                codes = 1;
            end
            A_ONEHOLD_ONE_HOLD: begin
                codes = 0;
            end
            A_ONEHOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD: begin
                codes = 0;
            end
            A_ONEHOLD_ONE_HOLD_ONE_HOLD_ONE_HOLD: begin
                codes = 1;
            end
        endcase
    end
endmodule
