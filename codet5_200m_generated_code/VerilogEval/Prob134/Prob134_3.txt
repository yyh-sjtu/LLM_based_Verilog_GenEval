module TopModule (
	input x,
	input [2:0] y,
	output Y0,
	output z
);
    parameter [1:0] INIT = 2'b00;
    parameter [1:0] STATE = INIT;
    assign Y0 = y[0];
    assign z = y[1];
    always @(x or y)
    begin
        case (STATE)
            INIT:
                assign z = 1'b1;
            endcase
        end
endmodule
