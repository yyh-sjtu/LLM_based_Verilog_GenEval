module TopModule (
    input a,
    input b,
    input c,
    input d,
    output q
);
    parameter A_WIDTH = 32;
    parameter B_WIDTH = 32;
    parameter C_WIDTH = 32;
    parameter D_WIDTH = 32;
    parameter E_WIDTH = 32;
    parameter H_WIDTH = 32;
    parameter V_WIDTH = 32;
    parameter V_WIDTH = 32;
    parameter V_WIDTH = 32;
    parameter V_WIDTH = 32;
    reg q;
    always @ (a or b or c or d)
    begin
        q <= a ^ b ^ c ^ d;
    end
endmodule
