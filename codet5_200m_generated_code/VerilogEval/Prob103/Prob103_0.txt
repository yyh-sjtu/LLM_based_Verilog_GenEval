module TopModule (
    input a,
    input b,
    input c,
    input d,
    output q
);
    reg [3:0] code;
    initial code = 0;
    always @ (a or b or c or d)
        if (a) code <= 2'b00; 
        else if (b) code <= 2'b01; 
        else code <= 2'b0;
    always @ (a or b or c or d)
        if (a) code <= 2'b10; 
        else code <= 2'b0;
    assign q = code;
endmodule
