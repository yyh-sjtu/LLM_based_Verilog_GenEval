module TopModule (
	input [6:0] y,
	input w,
	output Y1,
	output Y3
);
    parameter [3:0] INIT = 2'b00;
    parameter [3:0] INIT2 = 2'b01;
    parameter [3:0] INIT3 = 2'b10;
    parameter [3:0] INIT4 = 2'b11;
    reg [3:0] state;
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= INIT;
            Y1 <= 1'b0;
            Y3 <= 1'b0;
        end
        else begin
            state <= state + 1;
            Y1 <= 1'b0;
            Y3 <= 1'b0;
        end
    end
endmodule
