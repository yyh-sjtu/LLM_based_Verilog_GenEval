module TopModule (
	input clk,
	input reset,
	input in,
	output out
);
    parameter A = 1;
    parameter B = 2;
    parameter C = 3;
    reg [1:0] state;
    reg [1:0] code;
    reg [3:0] data;
    initial begin
        state = A;
        code = 0;
    end
    always @(posedge clk or posedge reset) begin
        if(reset) begin
            code = 0;
            data = 0;
        end
        else begin
            case(state)
                A: begin
                    if(in) begin
                        code = data;
                        data = 1;
                    end
                    end
                B: begin
                        code = data;
                        data = 0;
                    end
                C: begin
                        code = data;
                        data = 0;
                    end
                default: begin
                        code = 4'b0000;
                    end
            endcase
        end
    end
    assign out = code;
endmodule
