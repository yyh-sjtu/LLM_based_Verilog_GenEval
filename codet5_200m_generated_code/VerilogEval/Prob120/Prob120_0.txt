module TopModule (
	input clk,
	input reset,
	input in,
	output out
);
    reg [1:0] state;
    reg [3:0] nextstate;
    always @(posedge clk or posedge reset)
    begin
        if(reset)
            state <= 3'b000;
        else
            state <= nextstate;
    end
    always @(posedge clk or posedge reset)
    begin
        if(reset)
            nextstate <= 3'b000;
        else
            nextstate <= nextstate;
    end
    always @(posedge clk or posedge reset)
    begin
        if(reset)
            nextstate <= 3'b000;
        else
            nextstate <= nextstate;
    end
    always @(posedge clk or posedge reset)
    begin
        if(reset)
            nextstate <= 3'b000;
        else
            nextstate <= nextstate;
    end
    assign out = state == 3'b000? 3'b000 : 2'b11;
endmodule
