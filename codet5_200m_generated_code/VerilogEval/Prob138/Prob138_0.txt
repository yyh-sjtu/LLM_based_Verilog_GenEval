module TopModule (
    input clk,
    input reset,
    input w,
    output z
);
    parameter [0:0] INIT = 1'b0;
    parameter [0:0] WAIT = 1'b1;
    reg [0:0] state;
    always @(posedge clk or posedge reset)
        if (reset)
            state <= INIT;
        else
            state <= state + 1'b1;
    assign z = state[0];
endmodule
