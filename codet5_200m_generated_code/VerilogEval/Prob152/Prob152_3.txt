module TopModule (
	input clk,
	input areset,
	input bump_left,
	input bump_right,
	input ground,
	input dig,
	output walk_left,
	output walk_right,
	output aaah,
	output digging
);
    parameter DONT_TOUCH = 1'b1;
    parameter VIDEO_DEPTH = 4;
    parameter VIDEO_N = 4;
    parameter VIDEO_STANDARD = "NO";
    parameter VIDEO_CARRY_MODE = "ALL";
    parameter VIDEO_POR'SLEEP_MODE = "SINGLE";
    reg [1:0] current_state;
    always @ (posedge clk)
    begin
        current_state <= current_state + 1'b1;
    end
    always @ (posedge clk)
    begin
        if (reset)
            current_state <= DONT_TOUCH;
        else if (bump_left && dig)
            current_state <= VIDEO_STATE;
        else if (bump_right && dig)
            current_state <= VIDEO_STATE;
    end
    wire [DONT_TOUCH-1:0] current_bump = {current_state[0], bump_left, bump_right};
    assign walk_left = current_bump[1]? 1'b1 : 1'b0;
    assign walk_right = current_bump[0]? 1'b1 : 1'b0;
    assign digging = current_state == VIDEO_STANDARD;
    assign aaah = current_state == VIDEO_POR;
    wire [DONT_TOUCH-1:0] current_bump_next = (current_state == VIDEO_STANDARD)? (bump_left? 1'b1 : 1'b0) :
                                    (current_state == VIDEO_STANDARD)? 1'b1 :
                                    (current_state == VIDEO_STANDARD)? 1'b1 :
                            (current_state == VIDEO_STANDARD)? 1'b1 :
                            1'b0;
    assign digging = current_state == VIDEO_STANDARD;
    always @ (posedge clk)
    begin
        if (reset)
            current_bump <= {DONT_TOUCH_MODE, VIDEO_STANDARD[DONT_TOUCH-1:0]};
        else if (bump_left && dig)
            current_bump <= current_bump[1]? 1'b1 : 1'b0;
    end
endmodule
