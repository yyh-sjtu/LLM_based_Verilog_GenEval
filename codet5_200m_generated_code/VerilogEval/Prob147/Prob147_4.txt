module TopModule (
    input clk,
    input a,
    input b,
    output q,
    output state
);
    reg [3:0] a_reg;
    reg [3:0] b_reg;
    reg [3:0] state_reg;
    always @(posedge clk) begin
        if (a_reg[0] == 1'b1)
            state_reg <= a_reg[1];
        else
            state_reg <= b_reg[1];
    end
    assign q = state_reg;
    always @(posedge clk) begin
        if (a_reg[0] == 1'b1)
            q_reg <= a_reg[1];
        else
            q_reg <= b_reg;
    end
endmodule
