module TopModule (
    input clk,
    input areset,

    input predict_valid,
    input [6:0] predict_pc,
    output predict_taken,
    output [6:0] predict_history,

    input train_valid,
    input train_taken,
    input train_mispredicted,
    input [6:0] train_history,
    input [6:0] train_pc
);
    reg [7:0] pc_reg;
    reg [7:0] global_history;
    always @(posedge clk or negedge areset) begin
        if (!areset) begin
            pc_reg <= 6'b0;
            predict_taken <= 0;
            predict_history <= 0;
        end
        else begin
            pc_reg <= pc_reg ^ {predict_pc, train_mispredicted};
            predict_taken <= predict_taken | (predict_reg >> 1);
            predict_history <= global_history ^ {predict_pc, train_mispredicted};
        end
    end
endmodule
