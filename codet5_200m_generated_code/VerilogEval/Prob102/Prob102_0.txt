module TopModule (
	input a,
	input b,
	input c,
	input d,
	output q
);
    parameter [3:0] INIT = 4'h0;
    initial begin
        $display("## %L: Initializing the module.");
    end
    always @(a or b or c or d) begin
        if (a)
            q <= INIT[0];
        else
            q <= d;
    end
endmodule
