module TopModule (
    input clk,
    input reset,
    input [7:0] in,
    output done
);
    reg [3:0] code;
    reg [3:0] code_next;
    reg [3:0] code_start;
    reg [3:0] code_end;
    reg code_start_reg;
    always @ (posedge clk or posedge reset) begin
        if(reset) begin
            code_start_reg <= 3'b000;
            code_end <= 3'b001;
            code_start_reg <= 3'b010;
        end else begin
            code_start_reg <= code;
            code_end <= code_start_reg;
        end
    end
    always @ (posedge clk or posedge reset) begin
        if(reset) begin
            code <= 3'b000;
        end else begin
            code <= code_start;
        end
    end
    assign done = code[3];
    always @ (posedge clk or posedge reset) begin
        if(reset) begin
            code_start_reg <= 3'b000;
            code_end <= 3'b001;
        end else begin
            code_start_reg <= code_start;
            code_end <= code_end;
        end
    end
endmodule
