module TopModule (
    input clk,
    input reset,
    input [7:0] in,
    output done
);
    parameter RECEIVER_PS2 = 0;
    parameter RECEIVER_PS2 = 1;
    reg [3:0] p_state;
    reg [3:0] p_state_next;
    reg [3:0] p_state_next_value;
    wire [3:0] msg;
    always @(posedge clk) begin
        p_state <= p_state_next;
        p_state_next_value <= p_state_next;
    end
    always @(posedge clk) begin
        p_state_next <= p_state_next_value;
        p_state_next_value <= {p_state_next, p_state_next_value};
    end
    assign msg = p_state_next_value;
    assign done = msg;
endmodule
