module TopModule (
    input clk,
    input reset,
    input s,
    input w,
    output z
);
    parameter A = 0;
    parameter S = 1;
    parameter W = 2;
    reg [1:0] state;
    always @(posedge clk)
        if (reset)
            state <= A;
        else if (s)
            state <= S;
    assign z = state == W;
endmodule
