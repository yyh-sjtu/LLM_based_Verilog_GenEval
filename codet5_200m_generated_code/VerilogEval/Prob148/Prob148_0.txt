module TopModule (
    input  clk,
    input  resetn,
    input  [2:0] r,
    output [2:0] g
);
    reg [1:0] state;
    reg [1:0] next_state;
    always @(posedge clk or negedge resetn)
        if (!resetn)
            state <= 0;
        else
            state <= next_state;
    always @(posedge clk or negedge resetn)
        if (!resetn)
            next_state <= 0;
        else
            next_state <= next_state + 1;
    always @(posedge clk or negedge resetn)
        if (!resetn)
            next_state <= 0;
        else
            next_state <= next_state + 1;

endmodule
