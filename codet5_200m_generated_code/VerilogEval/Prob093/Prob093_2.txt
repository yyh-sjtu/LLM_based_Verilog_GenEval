module TopModule (
	input c,
	input d,
	output [3:0] mux_in
);
    wire [3:0] a = c? d : 4'b0000;
    wire [3:0] b = c? d : 4'b0000;
    wire [3:0] a_out;
    wire [3:0] b_out;
    wire [3:0] mux_out;
    assign mux_out = a_out? {a[0], a[1], a[2], a[3], a[4]} : a_out;
    Karnaugh m(a, b, mux_out);
    assign mux_in = b_out? {b[0], b[1], b[2], b[3], b[4]} : b_out;
endmodule
