module TopModule (
	input clk,
	input reset,
	input [31:0] in,
	output [31:0] out
);
    reg [31:0] codes[31:0];
    assign out = codes[in];
    always @(posedge clk or posedge reset) begin
        if(reset) begin
            codes[31:0] <= 32'b0;
            codes[1:0] <= 32'b0;
        end else begin
            case(codes)
                32'b0: codes[31:0] <= in;
                32'b0: codes[1:0] <= in;
                32'b1: codes[1:0] <= in;
                default: codes[31:0] <= 32'b0;
            endcase
        end
    end
endmodule
