module TopModule (
	input clk,
	input [7:0] in,
	input [7:0] anyedge
);
    wire [7:0] out;
    wire [7:0] out_reg;
    assign out_reg = out[7];
    assign out_reg_last = out[7];
    assign out_reg = out_reg + 1;
    always @ (posedge clk) begin
        if (out_reg_last == 8'b1) begin
            out <= 8'b1;
        end else begin
            out <= out_reg;
        end
    end
endmodule
