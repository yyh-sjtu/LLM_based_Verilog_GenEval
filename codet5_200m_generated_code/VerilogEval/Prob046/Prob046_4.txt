module TopModule (
	input clk,
	input reset,
	input [7:0] d,
	output [7:0] q
);
    reg [7:0] q_reg, q_next;
    always @(negedge clk, posedge reset)
        q_reg <= reset? 8'h34 : q_next;
    always @(negedge clk, posedge reset)
        q_next <= q_reg;
    always @(negedge clk, posedge reset)
        q_next <= q_reg;
    always @(negedge clk, posedge reset)
        q <= q_reg;
endmodule
