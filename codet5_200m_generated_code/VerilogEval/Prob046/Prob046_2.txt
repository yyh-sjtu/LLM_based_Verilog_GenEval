module TopModule (
	input clk,
	input reset,
	input [7:0] d,
	output [7:0] q
);
    genvar i;
    generate
        for (i = 0; i < 8; i = i + 1) begin : gen_dff
            DFF3 FD3 (
                .Q(q[i]),
                .C(clk),
                .D(d[i]),
                .R(reset),
                .S(1'b1)
            );
        end
    endgenerate
endmodule
