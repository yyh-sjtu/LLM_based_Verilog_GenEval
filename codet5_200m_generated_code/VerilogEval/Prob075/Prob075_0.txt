module TopModule (
    input clk,
    input areset,
    input train_valid,
    input train_taken,
    output [1:0] state
);
    reg [1:0] count;
    always @(posedge clk or posedge areset)
    begin
        if(areset)
            count <= 2'b01;
        else
            count <= count + 1'b1;
    end
    assign state = count;
endmodule
