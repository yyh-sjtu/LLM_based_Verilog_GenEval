module TopModule (
    input clk,
    input reset,
    input j,
    input k,
    output out
);
    reg [7:0] b;
    reg [7:0] a;
    reg [7:0] b_reg;
    assign out = b_reg;
    always @(posedge clk, posedge reset) begin
        if(reset) begin
            b <= 0;
            a <= 0;
            b_reg <= 0;
        end else begin
            if(j) begin
                b <= b;
                a <= a;
            end
            if(k) begin
                b <= b_reg;
            end
        end
    end
endmodule
