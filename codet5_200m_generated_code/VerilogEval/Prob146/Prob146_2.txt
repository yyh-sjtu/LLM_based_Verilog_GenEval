module TopModule (
    input clk,
    input in,
    input reset,
    output [7:0] out_byte,
    output done
);
parameter S_START = 8'd0;
parameter S_END = 8'd1;
reg [3:0] state;
reg [7:0] rdata;
reg [7:0] rdata_valid;
reg rdata_valid_l;
reg [1:0] count;
reg [3:0] rbyte_reg;
always @(posedge clk)
begin
    if(reset)
        state <= S_START;
    else
        state <= state + 1;
end
always @(posedge clk)
begin
    rdata_valid <= in;
    rdata_valid_l <= rdata_valid;
    rbyte_reg <= rdata;
end
always @(posedge clk)
begin
    if(reset)
        count <= 3'd0;
    else
        count <= count + 3'd1;
    if(count == 3'd2)
        rdata_valid_l <= 1'b0;
    else
        rdata_valid_l <= 1'b1;
end
always @(posedge clk)
begin
    if(reset)
        rbyte_reg <= 8'd0;
    else
        rbyte_reg <= {rbyte_reg[7:1], 1'b0};
    if(count == 3'd4)
        rbyte_valid_l <= 1'b0;
    else
        rbyte_valid_l <= 1'b0;
end
always @(posedge clk)
begin
    if(reset)
        count <= 3'd0;
    else
        count <= count + 3'd1;
end
always @(posedge clk)
begin
    if(reset)
        rbyte_reg <= 8'd0;
    else
        rbyte_reg <= {rbyte_reg[7:1], 1'b0};
    else
        rbyte_reg <= {rbyte_reg[7:1], 1'b0};
end
always @(posedge clk)
begin
    if(reset)
        count <= 3'd0;
    else
        count <= count - 3'd1;
end
always @(posedge clk)
begin
    if(reset)
        rbyte_reg <= 8'd0;
    else
        rbyte_reg <= {rbyte_reg[7:1], 1'b0};
    else
        rbyte_reg <= {rbyte_reg[7:1], 1'b0};
end
always @(posedge clk)
begin
    if(reset)
        rbyte_reg <= 8'd0;
    else
        rbyte_reg <= {rbyte_reg[7:1], 1'b0};
end
assign out_byte = rbyte_reg;
assign done = rbyte_reg[2];
endmodule
