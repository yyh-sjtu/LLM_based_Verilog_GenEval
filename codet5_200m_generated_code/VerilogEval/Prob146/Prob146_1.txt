module TopModule (
    input clk,
    input in,
    input reset,
    output [7:0] out_byte,
    output done
);
parameter DATA_WIDTH = 8;
parameter START_BIT = 0;
parameter TWO_DATA_BITS = 1;
parameter START_BIT_BITS = 2;
wire [7:0] out_byte_w;
assign out_byte = out_byte_w;
reg [7:0] out_byte_i;
reg [7:0] out_byte_o;
reg [7:0] out_byte_i_d;
reg [7:0] out_byte_o_d;
reg done_i;
always @(posedge clk or negedge reset)
begin
    if (reset == 1'b1) begin
        out_byte_i <= 8'h00;
    end else begin
        if (out_byte_i == 8'h00) begin
            out_byte_i <= out_byte_w;
            done_i <= 1'b1;
        end else if (out_byte_i == 8'h00) begin
            out_byte_i <= out_byte_o;
            done_i <= 1'b1;
        end else if (out_byte_i_d == 8'h00) begin
            out_byte_i <= out_byte_o;
            done_i <= 1'b1;
        end else if (done_i == 1'b0) begin
            done_i <= 1'b0;
        end
    end
end
always @(posedge clk or negedge reset)
begin
    if (reset == 1'b1) begin
        out_byte_o <= 8'h00;
        out_byte_o_d <= 8'h00;
        out_byte_o_i_d <= 8'h00;
    end else begin
        if (out_byte_o == 8'h00) begin
            out_byte_o <= out_byte_o_d;
            out_byte_o_d <= out_byte_o_o_o_d;
        end
        if (out_byte_o_d == 8'h00) begin
            out_byte_o <= out_byte_o_o_o_d;
            out_byte_o_d <= out_byte_o_o_o_o_d;
        end
        if (out_byte_o_d == 8'h00) begin
            out_byte_o <= out_byte_o_o_o_d;
            out_byte_o_d <= out_byte_o_d;
        end
    end
end
always @(posedge clk or negedge reset)
begin
    if (reset == 1'b1) begin
        out_byte_o <= 8'h00;
        out_byte_o_i_d <= 8'h00;
        done_i <= 1'b0;
    end else begin
        if (out_byte_o_d == 8'h00) begin
            out_byte_o <= out_byte_o_o;
            done_i <= 1'b1;
        end else begin
            out_byte_o <= out_byte_o_i;
            done_i <= 1'b0;
        end
    end
end
assign done = done_i;
always @(posedge clk or negedge reset)
begin
    if (reset == 1'b1) begin
        out_byte_o <= 8'h00;
        out_byte_o_i_d <= 8'h00;
        out_byte_o_o_i_d <= 8'h00;
        out_byte_o_o_i_d <= 8'h00;
        done_i <= 1'b0;
    end else begin
        out_byte_o <= out_byte_o_i;
        out_byte_o_i_d <= out_byte_o_o_i_d;
        done_i <= done_i;
    end
end
always @(posedge clk or negedge reset)
begin
    if (reset == 1'b1) begin
        done_i <= 1'b0;
        out_byte_o_i_d <= 8'h00;
        out_byte_o_i_d <= 8'h00;
        out_byte_o_o_o_i_d <= 8'h00;
        out_byte_o_o_i_d <= 8'h00;
    end else begin
        out_byte_o <= out_byte_o_i;
        out_byte_o_i_d <= out_byte_o_o_i_d;
        done_i <= done_i;
    end
end
always @(posedge clk or negedge reset)
begin
    if (reset == 1'b1) begin
        out_byte_o <= 8'h00;
        out_byte_o_i_d <= 8'h00;
        out_byte_o_i_d <= out_byte_o_o_o_o_o_i_d;
        done_i <= 1'b1;
    end else begin
        out_byte_o <= out_byte_o_i;
        out_byte_o_i_d <= out_byte_o_i;
        done_i <= done_i;
    end
end
endmodule
