module TopModule (
    input clk,
    input in,
    input reset,
    output [7:0] out_byte,
    output done
);
    parameter S = 5;
    parameter IDLE = 4;
    parameter EIGHT = 5;
    parameter COUNT = 6;
    parameter COUNT_READ = 4;
    parameter COUNT_WRITE = 4;
    parameter COUNT_WAIT = 5;
    reg [7:0] out_byte_reg;
    reg [7:0] out_byte_next;
    assign done = out_byte_reg == 8'b0;
    wire out_byte_valid;
    assign out_byte = out_byte_next;
    always @(posedge clk or posedge reset)
        if (reset)
            out_byte_reg <= 8'b0;
        else
            if (out_byte_valid)
                out_byte_reg <= {out_byte_reg[7:0], out_byte[7:0]};
    always @(posedge clk or posedge reset)
        if (reset)
            out_byte_next <= 8'b0;
        else
            if (out_byte_valid)
                out_byte_next <= {out_byte_reg[7:0], out_byte[7:0]};
    assign out_byte_valid = (out_byte_reg == 8'b1);
    assign out_byte[7:0] = out_byte_reg;
endmodule
