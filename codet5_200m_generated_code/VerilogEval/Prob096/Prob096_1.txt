module TopModule (
	input clk,
	input reset,
	input data,
	output start_shifting
);
    reg [7:0] data_in;
    reg [7:0] data_out;
    always @(posedge clk or posedge reset)
    begin
        if(reset)
            data_in <= 0;
        else
            data_in <= data;
    end
    assign start_shifting = data_out[7];
endmodule
