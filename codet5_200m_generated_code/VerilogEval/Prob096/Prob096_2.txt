module TopModule (
	input clk,
	input reset,
	input data,
	output start_shifting
);
    reg [4:0] codes;
    reg [7:0] data_in;
    reg start_shifting;
    reg [6:0] code_counter;
    reg [6:0] current_code;
    always @(posedge clk or posedge reset)
    begin
        if (reset) begin
            code_counter <= 6'd0;
            current_code <= 6'd0;
            start_shifting <= 1'b1;
            code_counter <= 6'd0;
        end
        else begin
            code_counter <= code_counter + 6'd1;
            current_code <= current_code + 6'd1;
        end
    end
endmodule
