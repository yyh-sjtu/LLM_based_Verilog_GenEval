module TopModule (
	input clk,
	input reset,
	input data,
	output start_shifting
);
    reg [1:0] state;
    wire [1:0] s1;
    assign s1 = 1'b1;
    assign start_shifting = 1'b1;
    reg [2:0] a;
    always @(posedge clk)
        if (reset)
            a <= 2'b00;
        else
            a <= a - 1;
    always @(posedge clk)
        if (reset)
            state <= 2'b00;
        else
            state <= a;
endmodule
