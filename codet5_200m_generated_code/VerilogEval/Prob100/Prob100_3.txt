module TopModule (
	input in,
	input [1:0] state,
	output [1:0] next_state,
	output out
);
    parameter [1:0] STATE_IDLE = 2'b00,
        STATE_READ = 2'b01,
        STATE_WRITE = 2'b10;
    reg [1:0] state_code;
    reg [1:0] next_state_code;
    always @ (state or in)
    begin
        case (state)
            STATE_IDLE:
                next_state_code = STATE_READ;
            STATE_READ:
                next_state_code = STATE_WRITE;
            STATE_WRITE:
                next_state_code = STATE_IDLE;
            default:
                next_state_code = STATE_IDLE;
        endcase
    end
    always @ (state or in)
    begin
        case (state)
            STATE_IDLE:
                next_state_code = STATE_READ;
            STATE_READ:
                next_state_code = STATE_WRITE;
            STATE_WRITE:
                next_state_code = STATE_IDLE;
            default:
                next_state_code = STATE_IDLE;
        endcase
    end
    always @ (state or in)
    begin
        case (state)
            STATE_IDLE:
                next_state_code = STATE_READ;
            STATE_READ:
                next_state_code = STATE_IDLE;
            STATE_WRITE:
                next_state_code = STATE_IDLE;
            default:
                next_state_code = STATE_IDLE;
        endcase
    end
    assign out = (state_code == STATE_IDLE)? 1'b1 : 1'b0;
endmodule
