module TopModule (
    input clk,
    input load,
    input [511:0] data,
    output [511:0] q
);
    reg [3:0] cnt;
    reg [3:0] cnt2;
    always @(posedge clk) begin
        if (load)
            cnt2 <= cnt2 + 1;
        else
            cnt2 <= 0;
    end
    assign q = cnt2;
    assign q[0] = cnt[3] ^ cnt[2] ^ cnt[1] ^ cnt[0];
    always @(posedge clk) begin
        if (load)
            cnt2[0] <= cnt2[1];
        else
            cnt2[1] <= 0;
    end
endmodule
