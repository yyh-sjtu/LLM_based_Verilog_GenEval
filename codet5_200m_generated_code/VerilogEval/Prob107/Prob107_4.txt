module TopModule (
    input clk,
    input reset,
    input in,
    output out
);
    parameter BUS_0 = 3'b000,
        BUS_1 = 3'b001,
        BUS_2 = 3'b010,
        BUS_3 = 3'b011,
        BUS_4 = 3'b100;
    reg [3:0] code;
    always @(posedge clk)
        code <= {code[3:0], in};
    assign out = code[2];
endmodule
