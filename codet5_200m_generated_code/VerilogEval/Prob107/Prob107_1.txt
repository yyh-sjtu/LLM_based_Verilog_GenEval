module TopModule (
    input clk,
    input reset,
    input in,
    output out
);
    parameter IDLE        = 2'b00, 
        STATE_S_START        = 2'b01, 
        STATE_S_STOP    = 2'b10; 
    parameter STATE_RESET    = 2'b11; 
    reg [1:0] out_status; 
    reg [1:0] next_state; 
    assign out = out_status[1]; 
    always @ (posedge clk or posedge reset)
        if(reset)
            out_status <= IDLE; 
        else
            out_status <= next_state; 
    always @ (posedge clk or posedge reset)
        if(reset)
            next_state <= IDLE; 
        else
            next_state <= next_state + 1; 
    assign out = out_status[0]; 
endmodule
