module TopModule (
	input clk,
	input resetn,
	input x,
	input y,
	output f,
	output g
);
    reg [1:0] state;
    reg [1:0] next_state;
    always @(posedge clk or negedge resetn)
        if (!resetn)
            state <= 0;
        else
            state <= next_state;
    assign f = state[0];
    assign g = state[1];
endmodule
