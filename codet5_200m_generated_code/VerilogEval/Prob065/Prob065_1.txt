module TopModule (
	input p1a,
	input p1b,
	input p1c,
	input p1d,
	input p2a,
	input p2b,
	input p2c,
	input p2d,
	output p1y,
	output p2y
);
    wire [3:0] nand0;
    wire [3:0] nand1;
    nand nand2 (nand0, p1a, p1b, p1c, p1d);
    nand nand3 (nand1, p2a, p2b, p2c, p2d);
    assign p1y = nand0[0];
    assign p2y = nand1[1];
endmodule
