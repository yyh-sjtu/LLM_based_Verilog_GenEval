module left_rotate (
    input clk,
    input reset,
    input [2:0] amount,
    input [7:0] data,
    input load,
    output reg [7:0] out
);
    reg [7:0] data_out;
    always @(posedge clk) begin
        if (reset) begin
            data_out <= 8'b0;
        end else begin
            if (load) begin
                data_out <= data;
            end
        end
    end
    always @(posedge clk) begin
        if (reset) begin
            data_out <= 8'b0;
        end else begin
            if (load) begin
                data_out <= data;
            end
        end
    end
    assign out = data_out;
endmodule
