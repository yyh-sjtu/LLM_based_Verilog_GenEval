module count_slow (
	input clk,
	input slowena,
	input reset,
	output reg [3:0] q
);
    parameter [3:0] DEC_COUNTER_WIDTH = 4;
    reg [3:0] count;
    always @(posedge clk)
        if (reset)
            count <= 0;
        else if (slowena)
            count <= count + 1;
    always @(posedge clk)
        if (reset)
            q <= 0;
        else if (slowena)
            q <= count;
endmodule
