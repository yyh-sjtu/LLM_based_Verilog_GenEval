module advshift (
	input clk,
	input load,
	input ena,
	input [1:0] amount,
	input [63:0] data,
	output reg [63:0] q
);
 
parameter WIDTH = 64; 
parameter DEPTH = 64; 
reg [WIDTH-1:0] shiftreg; 
always @(posedge clk)
  if (load)
    shiftreg <= data; 
always @(posedge clk)
  if (ena)
    q <= shiftreg[amount-1:0]; 
always @(posedge clk)
  if (load)
    shiftreg <= {shiftreg[WIDTH-1:0], data}; 
always @(posedge clk)
  if (ena)
    q <= shiftreg[amount-1:0]; 
endmodule
